#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 · Prüfkanal-Werkzeug (Vorlage: BIOTOP-LAGE-01 arbeit/pk.py und pk_stand.py).
Veröffentlicht Dateien aus <quelle> (Vorgabe /root/biotop-einlagerung/ausgabe) nach laeufe/<ordner>/ (Vorgabe
biotop-einlagerung-01):
1. Geheimnissuche (geheimnis.py: Muster und Werte aus .env ≥ 8 / auth.json ≥ 12 Zeichen) über genau diese Dateien;
   bei einem Treffer: Abbruch, nichts veröffentlicht.
2. exklusiv (O_EXCL): eine vorhandene Datei wird NIE überschrieben; höchstens 40.000 Bytes je Datei (vorab geprüft);
   Unterordner im Namen erlaubt (z. B. RUNDE6-DOKUMENTE-01/x.md).
3. Rechte root:caddy, Ordner 750, Dateien 640; danach Abruf jeder Datei über HTTPS, Vergleich Bytes und SHA-256.
Aufruf: pk.py [--ordner <unterordner von laeufe>] [--quelle <dir>] <name> [<name> ...]
        pk.py [--ordner …] --abruf <name> [...]   (nur abrufen und vergleichen)"""
import grp
import hashlib
import os
import subprocess
import sys
import urllib.error
import urllib.request
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
GID = grp.getgrnam('caddy').gr_gid
MAX = 40_000
GEHEIMNIS = '/root/biotop-einlagerung/arbeit/geheimnis.py'

args = sys.argv[1:]
ordner, quelle, nur_abruf = 'biotop-einlagerung-01', Path('/root/biotop-einlagerung/ausgabe'), False
while args and args[0].startswith('--'):
    if args[0] == '--ordner':
        ordner, args = args[1], args[2:]
    elif args[0] == '--quelle':
        quelle, args = Path(args[1]), args[2:]
    elif args[0] == '--abruf':
        nur_abruf, args = True, args[1:]
    else:
        sys.exit(f'unbekannt: {args[0]}')
LAEUFE = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
ZIEL = LAEUFE / ordner
BASIS = f'https://milieulux-reinigung-muenchen.de/_akte/{K}/laeufe/{ordner}'


def abrufen(name: str):
    q = urllib.request.Request(f'{BASIS}/{name}', headers={'User-Agent': 'biotop-einlagerung-01-abruf'})
    try:
        with urllib.request.urlopen(q, timeout=60) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, b''


def pruefen(name: str, daten: bytes) -> bool:
    h = hashlib.sha256(daten).hexdigest()
    st, ab = abrufen(name)
    gleich = st == 200 and hashlib.sha256(ab).hexdigest() == h
    print(f'{name} | {len(daten)} Bytes | sha256 {h} | HTTP {st} | abgerufen {len(ab)} Bytes | gleich {"ja" if gleich else "NEIN"}')
    print(f'  {BASIS}/{name}')
    return gleich


def main() -> None:
    if not args:
        sys.exit('keine Dateien genannt')
    if nur_abruf:
        ok = sum(pruefen(n, (ZIEL / n).read_bytes()) for n in args)
        print(f'Ergebnis: {ok} von {len(args)} Dateien mit HTTP 200 und gleicher SHA-256')
        sys.exit(0 if ok == len(args) else 1)
    zu_gross = [(n, (quelle / n).stat().st_size) for n in args if (quelle / n).stat().st_size > MAX]
    if zu_gross:
        sys.exit(f'ABBRUCH: über {MAX} Bytes: {zu_gross} — nichts veröffentlicht')
    vorhanden = [n for n in args if (ZIEL / n).exists()]
    if vorhanden:
        sys.exit(f'ABBRUCH: schon im Prüfkanal (nie überschreiben): {vorhanden} — nichts veröffentlicht')
    g = subprocess.run([sys.executable, GEHEIMNIS] + [str(quelle / n) for n in args], capture_output=True, text=True)
    zeile = [z for z in g.stdout.splitlines() if z.startswith('Durchsucht')]
    print('Geheimnissuche: ' + (zeile[0] if zeile else 'KEINE ZÄHLZEILE') + f' · rc {g.returncode}')
    for z in g.stdout.splitlines():
        if z.startswith('TREFFER'):
            print('  ' + z.replace(str(quelle) + '/', ''))
    if g.returncode != 0 or not zeile or not zeile[0].endswith('Treffer: 0') or f'Durchsucht: {len(args)} Dateien' not in zeile[0]:
        sys.exit('ABBRUCH: Geheimnissuche nicht sauber — nichts veröffentlicht')
    for d in [ZIEL] + sorted({(ZIEL / n).parent for n in args} - {ZIEL}):
        if not d.exists():
            d.mkdir(mode=0o750, parents=False)
            os.chown(d, 0, GID)
            os.chmod(d, 0o750)
            print(f'Ordner angelegt: {d.relative_to(LAEUFE)} (root:caddy 750)')
    ok = 0
    for n in args:
        daten = (quelle / n).read_bytes()
        z = ZIEL / n
        fd = os.open(z, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o640)
        with os.fdopen(fd, 'wb') as f:
            f.write(daten)
        os.chown(z, 0, GID)
        os.chmod(z, 0o640)
        ok += pruefen(n, daten)
    print(f'Ergebnis: {ok} von {len(args)} Dateien mit HTTP 200 und gleicher SHA-256')
    sys.exit(0 if ok == len(args) else 1)


if __name__ == '__main__':
    main()
