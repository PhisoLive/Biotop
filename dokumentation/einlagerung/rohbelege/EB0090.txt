#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 · Kosten: Kassenstand vorher (EB<a>) und nachher (EB<b>) vergleichen (nur lesend).
Je Datenbank und Tabelle: Zeilenzahl, jüngster Zeitwert und SHA-256 über alle Zeilen. Gleich = 0 neue Biotop-Buchungen.
Aufruf: kasse_vergleich.py <EB vorher> <EB nachher>"""
import re
import sys
from pathlib import Path

R = Path('/root/biotop-einlagerung/rohbelege')


def lade(nr: str) -> dict:
    t = (R / f'EB{int(nr):04d}.txt').read_text(encoding='utf-8').split('Ausgabe (vollständig, stdout und stderr):', 1)[1]
    werte, db, tab = {}, None, None
    for z in t.splitlines():
        m = re.match(r'^(STAMM|VB1|R02|T3) (/\S+)', z)
        if m:
            db = m.group(1)
            continue
        m = re.match(r'^  (kasse|eichkosten): (.*)$', z)
        if m:
            tab = m.group(1)
            n = re.match(r'(\d+) Zeilen', m.group(2))
            werte[(db, tab, 'Zeilen')] = n.group(1) if n else m.group(2)
            continue
        m = re.match(r'^    (Summe \S+|jüngster Wert \S+|SHA-256 über alle Zeilen[^:]*) = ?:? ?(.*)$', z)
        if m and db and tab:
            werte[(db, tab, m.group(1))] = m.group(2).strip()
        m = re.match(r'^    SHA-256 über alle Zeilen \(repr, nach rowid\): (\S+)', z)
        if m and db and tab:
            werte[(db, tab, 'SHA-256')] = m.group(1)
    return werte


a, b = lade(sys.argv[1]), lade(sys.argv[2])
gleich = 0
for k in sorted(set(a) | set(b)):
    va, vb = a.get(k), b.get(k)
    ok = va == vb
    gleich += ok
    print(f'{"gleich" if ok else "VERSCHIEDEN"} · {k[0]} {k[1]} {k[2]}: vorher {va} · nachher {vb}')
zeilen = [k for k in a if k[2] == 'Zeilen']
neu = sum(int(b.get(k, 0)) - int(a[k]) for k in zeilen if str(a[k]).isdigit() and str(b.get(k, '')).isdigit())
print(f'Werte verglichen: {len(set(a) | set(b))} · gleich: {gleich} · Zeilen nachher minus vorher (Summe aller Tabellen): {neu}')
print('ERGEBNIS: 0 neue Buchungen, Kassen unverändert' if gleich == len(set(a) | set(b)) else 'ERGEBNIS: Kassen NICHT unverändert')
