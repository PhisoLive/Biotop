#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 · Schritt 3: Arbeitsdokumente der Runde 6 (bisher unveröffentlicht) geschwärzt nach
ausgabe/RUNDE6-DOKUMENTE-01/ legen — veröffentlicht wird danach mit pk.py (Geheimnissuche, exklusiv, Abruf).
- Schwärzung: jedes Schlüsselmuster (geheimnis.MUSTER) → „[GESCHWÄRZT: <Muster>]“, jeder Wert aus .env (≥ 8) und
  auth.json (≥ 12) → „[GESCHWÄRZT]“; gezählt je Datei.
- Dateien über 38.000 B werden an Zeilengrenzen in Teile geteilt (…-TEILk…); .py/.sh bekommen „.txt“ angehängt.
- Die Teile tragen keinen Kopf: ihre Verkettung ist die geschwärzte Datei, bei 0 Schwärzungen byte-gleich das Original.
  Quelle, Bytes, SHA-256 und Teile stehen in VERZEICHNIS.md.
- Jede Zieldatei wird exklusiv angelegt."""
import hashlib
import os
import re
import sys
from pathlib import Path

sys.path.insert(0, '/root/biotop-einlagerung/arbeit')
import geheimnis  # noqa: E402

A = Path('/root/biotop-einlagerung/ausgabe/RUNDE6-DOKUMENTE-01')
V6 = Path('/root/biotop-vor-t4-06')
GRENZE = 38_000
WERTE = geheimnis.werte()
QUELLEN = [
    (V6 / 'UEBERGABE.md', 'Übergabe der Runde 6: Auftrag, Plan, Stand (neueste zuletzt), zuletzt geschrieben 25.09. 12:52'),
    (V6 / 'BAU-S1C-UNBEOBACHTET.md', 'Bauauftrag S1c (Klasse „tragfähig ohne Beobachtung“, Teile A–F)'),
    (V6 / 'bau' / 'S1C-PLAN.md', 'Plan- und Standdatei des Baus S1c'),
    (Path('/root/sicherung/biotop-lage-01/arbeit/berichte/BERICHT-S1c-deleg_c88798ee.txt'),
     'Schlussbericht S1c im Wortlaut (Delegation deleg_c88798ee; gesichert in BIOTOP-LAGE-01, R175/R193)'),
    (Path('/root/biotop-vor-t4-05/SPEZIFIKATIONEN-NAECHSTER-AUFTRAG.md'),
     'Spezifikationen für Runde 6 aus Runde 5 (R6-1 … R6-15, U-1 … U-4)'),
    (V6 / 'bau' / 'S1C-BAU-SPEZ.txt', 'Protokoll: rote Spezifikationen S1c geschrieben'),
    (V6 / 'bau' / 'S1B-BAU-SPEZ.txt', 'Protokoll: rote Spezifikationen S1b (1)'),
    (V6 / 'bau' / 'S1B-BAU-SPEZ2.txt', 'Protokoll: rote Spezifikationen S1b (2)'),
    (V6 / 'bau' / 'S1B-BAU-SPEZ3.txt', 'Protokoll: rote Spezifikationen S1b (3)'),
    (V6 / 'bau' / 'R6-S4-SPEZIFIKATION.txt', 'Protokoll: Spezifikation S4'),
    (Path('/root/biotop/tests/test_vor_t4_06.py'),
     'Spezifikationen und Tests der Runde 6, darunter die roten test_w_g_* (Befund G); Stand Commit e8401af'),
    (V6 / 'BAU-S1-SONDE.md', 'Bauauftrag S1'), (V6 / 'BAU-S1-FORTSETZUNG.md', 'Bauauftrag S1-Fortsetzung'),
    (V6 / 'BAU-S1B-M2B.md', 'Bauauftrag S1b'), (V6 / 'BAU-S2-TOR4.md', 'Bauauftrag S2'),
    (V6 / 'BAU-S3-STUFE2.md', 'Bauauftrag S3'), (V6 / 'BAU-S4-VERDRAHTUNG.md', 'Bauauftrag S4'),
]
sha = lambda b: hashlib.sha256(b).hexdigest()


def maske(t: str):
    n = 0
    for name, m in geheimnis.MUSTER.items():
        t, k = re.subn(m, f'[GESCHWÄRZT: {name}]', t)
        n += k
    for v in WERTE:
        if v in t:
            n += t.count(v)
            t = t.replace(v, '[GESCHWÄRZT]')
    return t, n


def teile(b: bytes):
    if len(b) <= GRENZE:
        return [b]
    out, akt = [], b''
    for z in b.splitlines(keepends=True):
        if akt and len(akt) + len(z) > GRENZE:
            out.append(akt)
            akt = b''
        akt += z
    out.append(akt)
    return out


A.mkdir(parents=True, exist_ok=False)
zeilen = ['# BIOTOP-EINLAGERUNG-01 · RUNDE6-DOKUMENTE-01 · Verzeichnis', '',
          'Arbeitsdokumente der Behebungsrunde 6, bisher unveröffentlicht (Auftrag Schritt 3). Geschwärzt: Schlüsselmuster '
          'und Werte aus `.env`/`auth.json` (Werkzeug `arbeit/geheimnis.py`). Teile haben keinen Kopf; ihre Verkettung ist '
          'die geschwärzte Datei, bei 0 Schwärzungen byte-gleich das Original.', '',
          '| Quelle | Zweck | Bytes | SHA-256 Original | geschwärzt | veröffentlichte Teile (Bytes, SHA-256) |', '|---|---|---|---|---|---|']
gesamt = 0
for q, zweck in QUELLEN:
    orig = q.read_bytes()
    text, n = maske(orig.decode('utf-8'))
    neu = text.encode('utf-8')
    tt = teile(neu)
    assert b''.join(tt) == neu
    endung = q.suffix + ('.txt' if q.suffix in ('.py', '.sh') else '')
    namen = [q.stem + endung] if len(tt) == 1 else [f'{q.stem}-TEIL{k}{endung}' for k in range(1, len(tt) + 1)]
    for name, b in zip(namen, tt):
        fd = os.open(A / name, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
        with os.fdopen(fd, 'wb') as f:
            f.write(b)
        gesamt += 1
    gleich = 'Verkettung = Original' if n == 0 and b''.join(tt) == orig else f'{n} Stellen'
    zeilen.append(f'| `{q}` | {zweck} | {len(orig)} | `{sha(orig)}` | {gleich} | '
                  + '; '.join(f'`{nm}` {len(b)} B `{sha(b)[:16]}…`' for nm, b in zip(namen, tt)) + ' |')
    print(f'{q} · {len(orig)} B · SHA-256 {sha(orig)} · geschwärzt {n} · Teile {len(tt)}: {namen}')
text = '\n'.join(zeilen) + '\n'
fd = os.open(A / 'VERZEICHNIS.md', os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
with os.fdopen(fd, 'w') as f:
    f.write(text)
print(f'Quellen {len(QUELLEN)} · Dateien {gesamt} + VERZEICHNIS.md ({len(text.encode())} B)')
