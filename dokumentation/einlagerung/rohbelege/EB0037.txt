def zaun(text):
    lauf = max((len(m.group(0)) for m in re.finditer(r'`+', text)), default=0)
    return '`' * max(3, lauf + 1)


def stuecke(text, platz):
    """Text in Stücke ≤ platz Bytes, an Zeilengrenzen, sonst an Zeichengrenzen."""
    aus, akt = [], ''
    for z in text.splitlines(keepends=True):
        while len(z.encode()) > platz:
            b = z.encode()[:platz].decode('utf-8', errors='ignore')
            if akt:
                aus.append(akt)
                akt = ''
            aus.append(b)
            z = z[len(b):]
        if len((akt + z).encode()) > platz and akt:
            aus.append(akt)
            akt = ''
        akt += z
    if akt or not aus:
        aus.append(akt)
    return aus
