#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 · Schritt 4: Masterprompt v1–v8 mit Änderungsverzeichnissen, weitere Biotop-Dokumente aus der
Sitzungsdatenbank, Aufträge und alle Verfassungen im Wortlaut nach paket/dokumentation/.
Quellen:
- Wortlaut v1–v8: /root/sicherung/biotop-lage-01/arbeit/dokumente/ (aus state.db, BIOTOP-LAGE-01 R15/R17). Volle
  Fassungen ≥ 10.000 B nach masterprompt/, Bruchstücke (< 10.000 B, aus Verdichtungen) nach masterprompt/bruchstuecke/.
  Abgleich je Fassung mit INVENTAR-TEIL2-01 §5 (SHA-256 ohne abschließenden Zeilenumbruch).
- Änderungsverzeichnisse: …/arbeit/fassungen/aenderungen-v1..v8.md (BIOTOP-LAGE-01, FK15).
- übrige Dokumente dieses Ordners mit „biotop“ im Inhalt → dokumente/ (andere Vorhaben bleiben draußen).
- Aufträge: BIOTOP-EINLAGERUNG-01, BIOTOP-LAGE-01 (…/auftrag/) → auftraege/.
- Verfassungen: /root/biotop/verfassungen/ (ganzer Ordner), /root/biotop-t3/VERFASSUNG-T3-*.md, dazu die
  zusammengestellten Wortlaute VERFASSUNGEN-TEIL1–4-01.md aus BIOTOP-LAGE-01 → verfassungen/.
Jede Kopie bytegleich (SHA-256 verglichen), exklusiv angelegt."""
import hashlib
import os
import re
import shutil
from pathlib import Path

D = Path('/root/biotop-einlagerung/paket/dokumentation')
L1 = Path('/root/sicherung/biotop-lage-01')
I4 = {'v1': '93d358cdda6473f91fda795392e509212f75f1f9678bef31593cad9104fe1ea8',
      'v2': 'acc42ad73f6111a3922993aaadb94ebb12aaeeb995f445f3fe7e8abaab1b4b9d',
      'v3': 'ca74ed53c33fd68f1ce1a5b172262197a173c118b1099a991a7dac0f350373a2',
      'v4': 'ff1cda438d09c1b440d32700d1723b8e8df0daf6f45a7e2251372e035042e886',
      'v5': 'a34d7e1ad4532fc1ea11fb805b3ef56d2ec09b0b1febad3dc3af1854f9f5035e',
      'v6': '3f63dec2a7d6f82e932699907dab42d71148fe84ba907a13c505fa2c010bd483',
      'v7': '0d3a07327b05da232adc49d56cf98e1369f04693c9e857b9e8199d558d38c6c1',
      'v8': '4168a35ede38c31fe4fad21844105a148da5b5aa0d69d2672f29b774b7a67cc9'}
sha = lambda b: hashlib.sha256(b).hexdigest()
n_kopien = {'gleich': 0, 'gesamt': 0}


def kopie(q: Path, z: Path) -> None:
    z.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(z, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
    os.close(fd)
    shutil.copyfile(q, z)
    shutil.copystat(q, z)
    os.chmod(z, 0o644)
    n_kopien['gesamt'] += 1
    n_kopien['gleich'] += sha(q.read_bytes()) == sha(z.read_bytes())


print('== Masterprompt')
for q in sorted((L1 / 'arbeit' / 'dokumente').glob('*_INFINITE-EVOLUTION-BIOTOP-Masterprompt-v*.md')):
    v = re.search(r'-(v\d)\.md$', q.name).group(1)
    b = q.read_bytes()
    voll = len(b) >= 10_000
    z = D / 'masterprompt' / ('' if voll else 'bruchstuecke') / q.name
    kopie(q, z)
    ohne = sha(b.rstrip(b'\n'))
    print(f'{v} · {q.name} · {len(b)} B · SHA-256 {sha(b)} · ohne abschl. NL {ohne} · '
          + (f'gleich INVENTAR I4: {ohne == I4[v] or sha(b) == I4[v]}' if voll else 'Bruchstück (< 10.000 B)'))
for n in range(1, 9):
    q = L1 / 'arbeit' / 'fassungen' / f'aenderungen-v{n}.md'
    kopie(q, D / 'masterprompt' / f'AENDERUNGEN-v{n}.md')
    print(f'Änderungsverzeichnis v{n}: {q.stat().st_size} B · SHA-256 {sha(q.read_bytes())}')
print('== weitere Dokumente aus der Sitzungsdatenbank')
for q in sorted((L1 / 'arbeit' / 'dokumente').glob('*.md')):
    if 'Masterprompt-v' in q.name and 'INFINITE-EVOLUTION' in q.name:
        continue
    b = q.read_bytes()
    if b'biotop' not in b.lower():
        print(f'ausgelassen (kein Biotop-Bezug im Inhalt, anderes Vorhaben): {len(b)} B')
        continue
    kopie(q, D / 'dokumente' / q.name)
    print(f'{q.name} · {len(b)} B · SHA-256 {sha(b)}')
print('== Aufträge')
for q in [Path('/root/biotop-einlagerung/auftrag/BIOTOP-EINLAGERUNG-01-Auftrag.md')] + sorted((L1 / 'auftrag').iterdir()):
    kopie(q, D / 'auftraege' / q.name)
    print(f'{q.name} · {q.stat().st_size} B · SHA-256 {sha(q.read_bytes())}')
print('== Verfassungen')
for q in sorted(p for p in Path('/root/biotop/verfassungen').rglob('*') if p.is_file()):
    kopie(q, D / 'verfassungen' / 'biotop-verfassungen' / q.relative_to('/root/biotop/verfassungen'))
    print(f'/root/biotop/verfassungen/{q.relative_to("/root/biotop/verfassungen")} · {q.stat().st_size} B · SHA-256 {sha(q.read_bytes())}')
for q in sorted(Path('/root/biotop-t3').glob('VERFASSUNG-T3-*.md')) + sorted((L1 / 'ausgabe').glob('VERFASSUNGEN-TEIL*-01.md')):
    kopie(q, D / 'verfassungen' / q.name)
    print(f'{q} · {q.stat().st_size} B · SHA-256 {sha(q.read_bytes())}')
print(f"Kopien: {n_kopien['gesamt']} · bytegleich {n_kopien['gleich']} von {n_kopien['gesamt']}")
