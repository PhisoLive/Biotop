#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 (nur lesend): Wo steht in der Datei aus P2 ein Wert aus einer .env-Datei (≥ 8 Zeichen)?
Die Datei wird hier im Skript genannt, nicht auf der Befehlszeile (Vorsorge SPERRE-01 aus BIOTOP-LAGE-01).
Ausgabe je Treffer: Zeile, Variablenname der Quelle, Länge; die Zeile selbst mit geschwärztem Wert, gekürzt auf 200 Zeichen.
Dazu: Rechte, Besitzer, Größe, SHA-256 der Datei und die übrigen Dateien des Ordners."""
import hashlib
import os
import stat
import sys
from pathlib import Path

sys.path.insert(0, '/root/sicherung/biotop-lage-01/arbeit')
import geheimnis  # noqa: E402  (werte(): .env-Werte ≥ 8 Zeichen und auth.json-Werte ≥ 12 Zeichen)

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
ORD = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe/biotop-lage-01/zwischenbericht-20260925-163807-r114')
p = ORD / 'ZWISCHENBERICHT.md'
wt = geheimnis.werte()
d = p.read_bytes()
st = p.stat()
print(f'Datei: laeufe/biotop-lage-01/zwischenbericht-20260925-163807-r114/ZWISCHENBERICHT.md · {len(d)} B · SHA-256 '
      f'{hashlib.sha256(d).hexdigest()} · Rechte {stat.filemode(st.st_mode)} · uid {st.st_uid} gid {st.st_gid}')
for i, z in enumerate(d.decode('utf-8', 'replace').splitlines(), 1):
    for v, quelle in wt.items():
        if v in z:
            print(f'Zeile {i}: Wert aus {quelle.split(":")[0].replace("/root/", "~/")}:{quelle.split(":")[-1]} '
                  f'({len(v)} Zeichen) · Zeile geschwärzt: {z.replace(v, "[GESCHWÄRZT]")[:200]}')
print('Ordnerinhalt:')
for q in sorted(ORD.iterdir()):
    s = q.stat()
    print(f'  {q.name} · {s.st_size} B · {stat.filemode(s.st_mode)} · uid {s.st_uid} gid {s.st_gid}')
