#!/bin/bash
# BIOTOP-EINLAGERUNG-01 · Schritt 6.5 (nur lesend): Git im entpackten Repository <pfad>.
# Erwartet: Markierungen runde-5 → 0c2f058…, eingelagert-20260925 → e8401af…, Zweig einlagerung/runde-6-stand ausgecheckt,
# main → 0c2f058…, git status sauber, git fsck ohne Fehler. Vergleich mit dem Original (Refs und Objektzahl).
set -u
R=$1
g() { git -C "$R" --no-optional-locks -c safe.directory='*' "$@"; }
o() { git -C /root/biotop --no-optional-locks "$@"; }
echo "== $(date '+%Y-%m-%d %H:%M:%S %Z') · $R"
echo "HEAD: $(g rev-parse HEAD) · Zweig: $(g symbolic-ref --short HEAD)"
echo "runde-5 → $(g rev-parse runde-5^{commit}) · eingelagert-20260925 → $(g rev-parse eingelagert-20260925^{commit})"
echo "main → $(g rev-parse main) · einlagerung/runde-6-stand → $(g rev-parse einlagerung/runde-6-stand)"
echo "-- Zweige"; g branch -a --format='%(refname) %(objectname)'
echo "-- Markierungen"; g tag -l --format='%(refname) %(objectname)'
st=$(g status --short --untracked-files=all | wc -l)
echo "git status --short (mit unverfolgten): $st Einträge"; g status --short --untracked-files=all | head -20
echo "-- ignorierte, aber vorhandene Dateien (Datenbanken u. a.): $(g status --short --ignored | grep -c '^!!')"
echo "-- git fsck --full"; g fsck --full 2>&1 | tail -5; echo "fsck rc=${PIPESTATUS[0]}"
echo "-- Vergleich mit dem Original /root/biotop"
a=$(g for-each-ref --format='%(refname) %(objectname)' | sort | sha256sum | cut -c1-64)
b=$(o for-each-ref --format='%(refname) %(objectname)' | sort | sha256sum | cut -c1-64)
echo "Refs gleich: $([ "$a" = "$b" ] && echo ja || echo NEIN) ($a)"
echo "Objekte entpackt: $(g count-objects -v | tr '\n' ' ')"
echo "Objekte Original: $(o count-objects -v | tr '\n' ' ')"
[ "$st" -eq 0 ] && [ "$(g rev-parse runde-5^{commit})" = 0c2f058d57f68b8891b535d6e00214172c82fefa ] \
  && [ "$(g rev-parse eingelagert-20260925^{commit})" = e8401afda5907a997eae96537394f7dbf4e8af3f ] \
  && [ "$(g symbolic-ref --short HEAD)" = einlagerung/runde-6-stand ] && echo "ERGEBNIS: Git im entpackten Stand wie verlangt" \
  || echo "ERGEBNIS: Git im entpackten Stand NICHT wie verlangt"
