#!/bin/bash
# BIOTOP-EINLAGERUNG-01 Schritt 1: Sicherungskopien vor jeder Änderung (nichts wird überschrieben: cp --no-clobber,
# Ziel unter /root/biotop-einlagerung/vorher/). Gesichert: MEMORY.md (vor dem Stillstandsvermerk) und auftrag.json des
# Zwischenbericht-Wächters (vor der Umstellung auf diesen Auftrag). Ausgabe: SHA-256 von Original und Kopie.
set -eu
V=/root/biotop-einlagerung/vorher
mkdir -p $V
date '+Zeitpunkt %Y-%m-%d %H:%M:%S %Z (UTC%z)'
cp -p --no-clobber /root/.hermes/memories/MEMORY.md $V/MEMORY.md.vor-einlagerung
cp -p --no-clobber /root/.hermes/zwischenbericht/auftrag.json $V/auftrag.json.vor-biotop-einlagerung-01
# zweite Kopie der auftrag.json am Ort (wie in BIOTOP-LAGE-01, waechter_umstellen.sh), exklusiv
cp -p --no-clobber /root/.hermes/zwischenbericht/auftrag.json /root/.hermes/zwischenbericht/auftrag.json.vor-biotop-einlagerung-01
sha256sum /root/.hermes/memories/MEMORY.md $V/MEMORY.md.vor-einlagerung /root/.hermes/zwischenbericht/auftrag.json \
  $V/auftrag.json.vor-biotop-einlagerung-01 /root/.hermes/zwischenbericht/auftrag.json.vor-biotop-einlagerung-01
wc -c /root/.hermes/memories/MEMORY.md
ls -la --time-style='+%Y-%m-%d %H:%M:%S' /root/.hermes/zwischenbericht/
