#!/bin/bash
# BIOTOP-EINLAGERUNG-01 · Stillstandsprüfung (nur lesend): Läuft etwas mit Biotop-Bezug?
# Geprüft: Prozesse, Container, Kammer-Sperre, Cron (root, /etc), systemd-Zeitgeber und -Dienste, at-Aufträge,
# Hermes-Cron-Jobs (mit Skriptinhalt auf Biotop-Bezug), Delegationen (Subagenten) laut Manifest, Wächter-Auftrag,
# Hermes-Hintergrundprozesse (Wecker) über die Prozessliste.
# Aufruf: s1_stillstand.sh <anlass>
echo "== Anlass: ${1:-?}"
date '+== Zeitpunkt %Y-%m-%d %H:%M:%S %Z (UTC%z)'
MUSTER='biotop|pytest|kammer|vitest|tsc |wecker|nachspiel|unshare|docker run|sleep '
echo "== 1. Prozesse, deren Befehlszeile auf /${MUSTER}/ passt (PID, Start, Laufzeit, Befehl; ohne diese Prüfung selbst)"
ps -eo pid,ppid,lstart,etime,args --sort=start_time | grep -E "$MUSTER" | grep -v -E "grep -E|s1_stillstand|eb.py" > /root/biotop-einlagerung/tmp/ps-treffer.txt
if [ -s /root/biotop-einlagerung/tmp/ps-treffer.txt ]; then cat /root/biotop-einlagerung/tmp/ps-treffer.txt; else echo "(keine)"; fi
echo "Anzahl Treffer: $(wc -l < /root/biotop-einlagerung/tmp/ps-treffer.txt)"
echo "-- Einordnung je Treffer: Biotop-Bezug, wenn die Befehlszeile /root/biotop, biotop-, kammer, pytest, nachspiel, wecker oder unshare nennt"
grep -c -E "/root/biotop|biotop-|kammer|pytest|nachspiel|wecker|unshare" /root/biotop-einlagerung/tmp/ps-treffer.txt | sed 's/^/Prozesse mit Biotop-Bezug: /'
echo "== 2. Container (docker ps -a: laufende und gestoppte)"
docker ps -a --no-trunc --format '{{.ID}} | {{.Image}} | {{.Status}} | {{.Names}}'
echo "laufend: $(docker ps -q | wc -l) · gesamt (auch gestoppt): $(docker ps -aq | wc -l)"
echo "== 3. Kammer-Sperrdatei /tmp/biotop-kammer-exklusiv"
ls -ld /tmp/biotop-kammer-exklusiv 2>&1
echo "== 4. Cron: crontab root, /var/spool/cron/crontabs, /etc/crontab, /etc/cron.d"
crontab -l 2>&1
ls -la /var/spool/cron/crontabs 2>&1
grep -v -E '^\s*(#|$)' /etc/crontab 2>&1
ls -la /etc/cron.d 2>&1
grep -l -i -E 'biotop|hermes' /etc/cron.d/* /etc/crontab /var/spool/cron/crontabs/* 2>/dev/null | sed 's/^/Datei mit biotop|hermes: /' || true
echo "== 5. systemd-Zeitgeber (alle) und Dienste/Zeitgeber mit biotop im Namen oder im Befehl"
systemctl list-timers --all --no-pager 2>&1
for t in $(systemctl list-timers --all --no-legend --no-pager | grep -o '[A-Za-z0-9@_.-]*\.timer' | sort -u); do
  s=$(systemctl show -p Unit --value "$t"); e=$(systemctl show -p ExecStart --value "$s" | grep -o 'argv\[\]=[^;]*' | head -1)
  echo "$t -> $s -> $e" | grep -i -E 'biotop|hermes|kammer' && echo "  ^ Biotop/Hermes-Bezug" || true
done
systemctl list-units --all --no-pager --no-legend 2>&1 | grep -i -E 'biotop|kammer' || echo "(keine systemd-Einheit mit biotop/kammer im Namen)"
echo "== 6. at-Aufträge"
command -v atq >/dev/null && atq || echo "(at nicht installiert)"
echo "== 7. Hermes-Cron-Jobs (jobs.json) mit Skript und Biotop-Bezug des Skripts"
python3 /root/biotop-einlagerung/arbeit/s1_cronjobs.py
echo "== 8. Subagenten: Delegations-Manifeste mit laufendem Status"
python3 /root/biotop-einlagerung/arbeit/s1_delegationen.py
echo "== 9. Zwischenbericht-Wächter: auftrag.json (Ziel der Zwischenberichte) und Herzschlag"
cat /root/.hermes/zwischenbericht/auftrag.json; echo
python3 -c "import json;d=json.load(open('/root/.hermes/zwischenbericht/herzschlag.json'));print('Herzschlag', d.get('zeit'), 'Eltern:', d.get('eltern','')[:80])"
echo "== 10. Hermes: pending/, pending_messages/, hooks/ (könnten etwas später auslösen)"
find /root/.hermes/pending /root/.hermes/pending_messages /root/.hermes/hooks -maxdepth 3 2>&1 | head -30
echo "== 11. kanban.db (Hermes-Aufgabentafel): Tabellen und Zeilen (nur lesend, immutable, keine -wal vorhanden?)"
ls -la /root/.hermes/kanban.db* 2>&1
python3 /root/biotop-einlagerung/arbeit/s1_kanban.py
echo "== Ende $(date '+%Y-%m-%d %H:%M:%S %Z')"
