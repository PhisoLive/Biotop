#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 (nur lesend): Welche Werte sucht die Geheimnissuche, und welcher Art sind sie?
Ausgegeben werden NUR Variablennamen, Länge und Art — nie ein Wert.
Art (Auslegung Hermes, für die Entscheidung, was in welches Archiv darf):
  ZUGANG  — Name enthält KEY, TOKEN, SECRET, PASSWORD, PASSWD, PASS, AUTH, COOKIE, CREDENTIAL, PRIVATE, SESSION, oder der
            Wert passt auf ein Schlüsselmuster (sk-…, gh?_…, AKIA…, Bot-Token), oder der Wert stammt aus auth.json;
  KENNUNG — Wert besteht nur aus Ziffern (auch mit Komma/Minus), z. B. Telegram-Chat- oder Benutzerkennung;
  SONST   — übrige .env-Werte ≥ 8 Zeichen (Modellnamen, Adressen, Einstellungen).
"""
import re
import sys
from collections import Counter

sys.path.insert(0, '/root/biotop-einlagerung/arbeit')
import geheimnis  # noqa: E402

ZUGANG_NAME = re.compile(r'KEY|TOKEN|SECRET|PASSWORD|PASSWD|PASS|AUTH|COOKIE|CREDENTIAL|PRIVATE|SESSION', re.I)
MUSTER = [re.compile(m) for m in geheimnis.MUSTER.values()]


def art(wert: str, quelle: str) -> str:
    datei, name = quelle.rsplit(':', 1)
    if datei.endswith('auth.json') or 'auth.json' in datei:
        return 'ZUGANG'
    if ZUGANG_NAME.search(name) or any(m.search(wert) for m in MUSTER):
        return 'ZUGANG'
    if re.fullmatch(r'[\d,\-\s]+', wert):
        return 'KENNUNG'
    return 'SONST'


if __name__ == '__main__':
    w = geheimnis.werte()
    z = Counter()
    for v, q in sorted(w.items(), key=lambda x: x[1]):
        a = art(v, q)
        z[a] += 1
        datei, name = q.rsplit(':', 1)
        print(f'{a:8} {len(v):4} Zeichen · {datei.replace("/root/", "~/")} : {name}')
    print(f'Werte gesamt {len(w)} · je Art {dict(z)}')
