#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 · Schritt 4: Kopien aller Biotop-Ordner des Prüfkanals (laeufe/biotop-* und laeufe/gedaechtnis/)
nach paket/dokumentation/pruefkanal/laeufe/…, dabei Geheimnissuche über JEDE Datei des Prüfkanals (Muster und Werte aus
.env ≥ 8 / auth.json ≥ 12 Zeichen).
- Datei ohne Treffer: bytegleich kopiert (Zeit übernommen), SHA-256 verglichen.
- Textdatei mit Treffer: in der Kopie geschwärzt („[GESCHWÄRZT]“ bzw. „[GESCHWÄRZT: <Muster>]“), in der Liste der
  Schwärzungen geführt (Datei, Art, Anzahl) — die Datei im Prüfkanal selbst bleibt unverändert (sie ist ein Befund).
- Binärdatei mit Treffer: nicht kopiert, nur genannt.
- Nicht lesbare Dateien (Rechte): Kopie trotzdem als root; gezählt.
Ausgabe: je Ordner Dateien/Bytes/geschwärzt; alle Fundstellen im Prüfkanal (ohne Werte); Summe. Exklusiv geschrieben."""
import hashlib
import os
import re
import shutil
import sys
from pathlib import Path

sys.path.insert(0, '/root/biotop-einlagerung/arbeit')
import geheimnis  # noqa: E402

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
Z = Path('/root/biotop-einlagerung/paket/dokumentation/pruefkanal/laeufe')
WERTE = geheimnis.werte()
MU = {n: re.compile(m) for n, m in geheimnis.MUSTER.items()}
sha = lambda b: hashlib.sha256(b).hexdigest()
Z.mkdir(parents=True, exist_ok=False)
ordner = sorted(p for p in L.iterdir() if p.is_dir() and (p.name.startswith('biotop-') or p.name == 'gedaechtnis'))
funde, gesamt = [], {'dateien': 0, 'bytes': 0, 'geschwaerzt': 0, 'nicht_kopiert': 0, 'gleich': 0}
for o in ordner:
    z = {'dateien': 0, 'bytes': 0, 'geschwaerzt': 0, 'nicht_kopiert': 0, 'gleich': 0}
    for p in sorted(o.rglob('*')):
        if p.is_dir() or p.is_symlink():
            continue
        rel = p.relative_to(L)
        d = p.read_bytes()
        try:
            t = d.decode('utf-8')
            text = True
        except UnicodeDecodeError:
            t = d.decode('latin-1')
            text = False
        treffer = []
        for name, m in MU.items():
            k = len(m.findall(t))
            if k:
                treffer.append((f'Muster {name}', k))
        for v, q in WERTE.items():
            if v in t:
                treffer.append((f'Wert {q.split(":")[-1]}', t.count(v)))
        ziel = Z / rel
        ziel.parent.mkdir(parents=True, exist_ok=True)
        z['dateien'] += 1
        if treffer and not text:
            z['nicht_kopiert'] += 1
            funde.append(f'{rel} · BINÄR, nicht kopiert · {treffer}')
            continue
        if treffer:
            for name, m in MU.items():
                t = m.sub(f'[GESCHWÄRZT: {name}]', t)
            for v in WERTE:
                t = t.replace(v, '[GESCHWÄRZT]')
            d2 = t.encode('utf-8')
            z['geschwaerzt'] += 1
            funde.append(f'{rel} · Text, in der Kopie geschwärzt · {treffer}')
        else:
            d2 = d
        fd = os.open(ziel, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
        with os.fdopen(fd, 'wb') as f:
            f.write(d2)
        shutil.copystat(p, ziel)
        os.chmod(ziel, 0o644)
        z['bytes'] += len(d2)
        z['gleich'] += (not treffer) and sha(ziel.read_bytes()) == sha(d)
    for k in gesamt:
        gesamt[k] += z[k]
    print(f"{o.name} | Dateien {z['dateien']} | Bytes der Kopie {z['bytes']} | bytegleich kopiert {z['gleich']} | "
          f"geschwärzt {z['geschwaerzt']} | nicht kopiert {z['nicht_kopiert']}")
print(f"SUMME: Ordner {len(ordner)} · Dateien {gesamt['dateien']} · Bytes {gesamt['bytes']} · bytegleich {gesamt['gleich']} · "
      f"geschwärzt {gesamt['geschwaerzt']} · nicht kopiert {gesamt['nicht_kopiert']}")
print(f'Fundstellen im Prüfkanal (Dateien mit Treffer): {len(funde)}')
for f in funde:
    print('  ' + f)
