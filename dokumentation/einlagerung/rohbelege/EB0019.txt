def jetzt() -> str:
    return time.strftime('%Y-%m-%d %H:%M:%S %Z (UTC%z)')


def sha_datei(p) -> str:
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for block in iter(lambda: f.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def exklusiv(p: Path, daten: bytes) -> None:
    p.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, 'wb') as f:
        f.write(daten)


def git(repo, *args, schreibend=False, pruefen=True) -> str:
    cmd = ['git'] + ([] if schreibend else ['--no-optional-locks']) + ['-C', str(repo)] + list(args)
    p = subprocess.run(cmd, capture_output=True)
    if pruefen and p.returncode != 0:
        raise SystemExit(f'ABBRUCH: {" ".join(cmd)} rc={p.returncode}: {p.stderr.decode(errors="replace")[:800]}')
    return p.stdout.decode('utf-8', errors='surrogateescape')


# ------------------------------------------------------------------ Prüfsummenliste
def liste_zeilen(wurzel: str):
    zeilen, n_dat, n_ver, n_son, byt = [], 0, 0, 0, 0
    for d, ds, fs in os.walk(wurzel):
        if d == wurzel:
            ds[:] = [x for x in ds if x != '.git']
            fs = [x for x in fs if x != '.git']      # .git als Datei (Arbeitsbaum-Verweis) zählt wie der Ordner nicht
        ds.sort()
        for x in list(ds):
            p = os.path.join(d, x)
            if os.path.islink(p):
                fs.append(x)
                ds.remove(x)
        for f in sorted(fs):
            p = os.path.join(d, f)
            rel = os.path.relpath(p, wurzel)
            st = os.lstat(p)
            if stat.S_ISLNK(st.st_mode):
                zeilen.append(f'VERWEIS -> {os.readlink(p)}  {rel}')
                n_ver += 1
            elif stat.S_ISREG(st.st_mode):
                zeilen.append(f'{sha_datei(p)}  {rel}')
                n_dat += 1
                byt += st.st_size
            else:
                zeilen.append(f'SONDERDATEI  {rel}')
                n_son += 1
    zeilen.sort(key=lambda z: z.split('  ', 1)[1])
    return zeilen, n_dat, n_ver, n_son, byt


def liste(wurzel: str, name: str) -> None:
    zeilen, n_dat, n_ver, n_son, byt = liste_zeilen(wurzel)
    text = '\n'.join(zeilen) + '\n'
    ziel = PS / f'{name}.txt'
    exklusiv(ziel, text.encode('utf-8', errors='surrogateescape'))
    print(f'Prüfsummenliste {name}: {wurzel} ohne die oberste .git/ (verschachtelte .git-Ordner zählen mit)')
    print(f'Zeitpunkt: {jetzt()}')
    print(f'Dateien: {n_dat} · Verweise: {n_ver} · Sonderdateien: {n_son} · Einträge gesamt: {len(zeilen)} · Bytes der Dateien: {byt}')
    print(f'Liste: {ziel} · {len(text.encode(errors="surrogateescape"))} Bytes · SHA-256 der Liste '
          f'{hashlib.sha256(text.encode(errors="surrogateescape")).hexdigest()}')


def lade(n: str):
    p = Path(n) if n.startswith('/') else PS / f'{n}.txt'
    t = p.read_text(errors='surrogateescape')
    m = {z.split('  ', 1)[1]: z.split('  ', 1)[0] for z in t.splitlines() if z}
    return m, hashlib.sha256(t.encode(errors='surrogateescape')).hexdigest(), p


def unterschied(ma, mb):
    nur_a = sorted(set(ma) - set(mb))
    nur_b = sorted(set(mb) - set(ma))
    anders = sorted(k for k in set(ma) & set(mb) if ma[k] != mb[k])
    return nur_a, nur_b, anders


def vergleich(a: str, b: str) -> None:
    ma, ha, pa = lade(a)
    mb, hb, pb = lade(b)
    nur_a, nur_b, anders = unterschied(ma, mb)
    print(f'A = {pa}: {len(ma)} Einträge, SHA-256 der Liste {ha}')
    print(f'B = {pb}: {len(mb)} Einträge, SHA-256 der Liste {hb}')
    print(f'nur in A: {len(nur_a)} · nur in B: {len(nur_b)} · Inhalt verschieden: {len(anders)}')
    for k in nur_a[:60]:
        print(f'  NUR A: {k}')
    for k in nur_b[:60]:
        print(f'  NUR B: {k}')
    for k in anders[:60]:
        print(f'  VERSCHIEDEN: {k}  {ma[k][:16]}… → {mb[k][:16]}…')
    print('ERGEBNIS: Unterschied leer' if not (nur_a or nur_b or anders) else 'ERGEBNIS: Unterschied NICHT leer')
    # Gegenprobe: dieselbe Vergleichsfunktion gegen eine Kopie von A mit genau einer geänderten Zeile
    kopie = dict(ma)
    schluessel = sorted(k for k, v in kopie.items() if len(v) == 64)[len(kopie) // 2]
    kopie[schluessel] = ('0' if kopie[schluessel][0] != '0' else '1') + kopie[schluessel][1:]
    g = unterschied(ma, kopie)
    ok = g == ([], [], [schluessel])
    print(f'Gegenprobe (Kopie von A, SHA-256 von „{schluessel}“ um ein Zeichen geändert): gemeldet {sum(map(len, g))} '
          f'Abweichung(en) → {"angeschlagen: genau dieser Pfad" if ok else "NICHT wie erwartet: " + str(g)[:300]}')

def status_eintraege(repo) -> list:
    roh = git(repo, 'status', '--porcelain=v1', '-z')
    teile = [t for t in roh.split('\0') if t]
    eintraege, i = [], 0
    while i < len(teile):
        t = teile[i]
        code, pfad = t[:2], t[3:]
        if code[0] in 'RC':      # Umbenennung: der nächste Teil ist der alte Pfad
            eintraege.append((code, pfad, teile[i + 1]))
            i += 2
            continue
        eintraege.append((code, pfad, None))
        i += 1
    return eintraege


def festschreiben(repo: str, basis: str, zweig: str, erwartet: str, nachricht: str, runde5: bool) -> None:
    r = Path(repo)
    print(f'Beginn {jetzt()} · Repository {repo}')

    print(f'HEAD {head} · Basis {basis} = {basis_voll}')
    if head != basis_voll:
        raise SystemExit('ABBRUCH: HEAD ist nicht die Basis — nichts geschrieben')
    alt_zweig = git(r, 'symbolic-ref', '-q', 'HEAD', pruefen=False).strip()

    schon = [x for x in fuer if x in refs]
    if schon:
        raise SystemExit(f'ABBRUCH: besteht schon: {schon} — nichts geschrieben')
    gd = Path(git(r, 'rev-parse', '--absolute-git-dir').strip())
    haken = [h.name for h in (gd / 'hooks').iterdir() if not h.name.endswith('.sample')] if (gd / 'hooks').is_dir() else []
    print(f'Haken außer *.sample: {haken or "keine"}')
    if haken:
        raise SystemExit('ABBRUCH: Haken vorhanden (könnten beim Commit etwas ändern) — nichts geschrieben')
    ein = status_eintraege(r)
    print(f'git status: {len(ein)} Einträge')
    for code, pfad, alt in ein:
        print(f'  {code!r} {pfad}' + (f' (vorher {alt})' if alt else ''))
    if erwartet != '-' and len(ein) != int(erwartet):
        raise SystemExit(f'ABBRUCH: {len(ein)} Einträge statt {erwartet} — nichts geschrieben')
    geaendert = [p for p in git(r, 'diff', '--name-only', '-z', 'HEAD').split('\0') if p]
    unverfolgt = [p for p in git(r, 'ls-files', '--others', '--exclude-standard', '-z').split('\0') if p]
    soll = sorted(set(geaendert) | set(unverfolgt))
    print(f'Dateien zum Festschreiben: {len(soll)} (geändert/vorgemerkt gegen HEAD {len(geaendert)}, unverfolgt {len(unverfolgt)})')
    verschachtelt = [p for p in unverfolgt if p.endswith('/.git') or '/.git/' in p or p == '.git']
    for code, pfad, _ in ein:
        if code == '??' and pfad.endswith('/') and any((r / pfad).rglob('.git')):
            verschachtelt.append(pfad)
    if verschachtelt:
        raise SystemExit(f'ABBRUCH: verschachteltes .git in Unverfolgtem: {verschachtelt[:5]} — nichts geschrieben')
    vorhanden = [str(r / p) for p in soll if (r / p).is_file() and not (r / p).is_symlink()]
    verweise = [p for p in soll if (r / p).is_symlink()]
    geloescht = [p for p in soll if not os.path.lexists(r / p)]
    g = subprocess.run([sys.executable, str(W / 'arbeit' / 'geheimnis.py')] + vorhanden, capture_output=True, text=True)

          f'{zz[0] if zz else "ohne Zählzeile"} · rc {g.returncode}')

        raise SystemExit('ABBRUCH: Geheimnissuche nicht sauber — nichts geschrieben')
    msg = Path(nachricht).read_text(encoding='utf-8')
    print(f'Commit-Nachricht ({nachricht}, {len(msg.encode())} B, SHA-256 {hashlib.sha256(msg.encode()).hexdigest()}):')
    print(msg.rstrip('\n'))
    # ---- Schreiben
    if runde5:
        git(r, 'tag', 'runde-5', basis_voll, schreibend=True)
        print(f'Markierung runde-5 → {git(r, "rev-parse", "runde-5").strip()}')
    git(r, 'switch', '-c', zweig, schreibend=True)

    print(f'vorgemerkt: {len(vorgemerkt)} Pfade · gleich den nicht eingecheckten Dateien: {vorgemerkt == soll}')

    git(r, *ICH, 'commit', '-q', '-F', nachricht, schreibend=True)

    print(f'git status nach dem Commit: {len(rest)} Einträge {"(sauber)" if not rest else rest[:10]}')
    if alt_zweig:
        nachher = git(r, 'rev-parse', alt_zweig).strip()
        print(f'bestehender Zweig {alt_zweig}: vorher {alt_spitze} · nachher {nachher} · unverändert: {nachher == alt_spitze}')
    print('--- Zweige und Markierungen')
    print(git(r, 'for-each-ref', '--format=%(refname) %(objectname) %(objecttype)').rstrip())
