#!/bin/bash
# BIOTOP-EINLAGERUNG-01 · Schritt 6 (Weg B, Teil „Abhängigkeiten einrichten“, im Kleinen):
# 1. Python-Umgebung der Regression im entpackten Stand neu bauen, aus der gesicherten Paketliste, ohne Netz (uv --offline,
#    Pakete aus dem uv-Zwischenspeicher des Servers). Das Archiv enthält keine venv (Abhängigkeitsordner).
# 2. Kammer-Abbilder aus dem Archiv laden (docker load) und die Kennungen mit dem Stand vor dem Laden vergleichen.
# Aufruf: s6_einrichten.sh <baum>
set -u
B=$1
V=$B/root/biotop-t3/venv
F=$B/root/biotop-einlagerung/paket/dokumentation/umgebung/pip-freeze-root-biotop-t3-venv.txt
T=$B/root/biotop-einlagerung/kammer-abbilder/biotop-kammer-1-und-ts1.docker-save.tar
echo "== $(date '+%Y-%m-%d %H:%M:%S %Z') · Baum $B"
[ -e "$V" ] && { echo "ABBRUCH: $V besteht schon"; exit 3; }
echo "-- 1. venv aus $F ($(wc -l < "$F") Pakete, SHA-256 $(sha256sum "$F" | cut -c1-64))"
cat "$F"
/root/.hermes/bin/uv venv --python /usr/bin/python3.12 "$V" 2>&1
/root/.hermes/bin/uv pip install --offline --python "$V/bin/python" -r "$F" 2>&1
echo "uv pip install rc=$?"
/root/.hermes/bin/uv pip freeze --python "$V/bin/python" > "$B/../venv-freeze-neu.txt" 2>/dev/null
echo "freeze neu gleich freeze gesichert: $(cmp -s "$B/../venv-freeze-neu.txt" "$F" && echo ja || echo NEIN)"
"$V/bin/python" -m pytest --version 2>&1
echo "-- 2. Kammer-Abbilder aus dem Archiv: $T ($(stat -c %s "$T") B, SHA-256 $(sha256sum "$T" | cut -c1-64))"
cat "$B/root/biotop-einlagerung/kammer-abbilder/ABBILDER.txt"
vor=$(docker image inspect --format '{{index .RepoTags 0}} {{.Id}}' biotop-kammer:1 biotop-kammer:ts1 2>&1)
echo "vor dem Laden:"; echo "$vor" | sed 's/^/  /'
docker load -i "$T" 2>&1
echo "docker load rc=$?"
nach=$(docker image inspect --format '{{index .RepoTags 0}} {{.Id}}' biotop-kammer:1 biotop-kammer:ts1 2>&1)
echo "nach dem Laden:"; echo "$nach" | sed 's/^/  /'
echo "Kennungen gleich den gesicherten (ABBILDER.txt): $(for l in $(echo "$nach" | awk '{print $2}'); do grep -q "$l" "$B/root/biotop-einlagerung/kammer-abbilder/ABBILDER.txt" && echo -n j || echo -n N; done)"
echo "Konfigurations-Digests im Tar (manifest.json):"; tar -xOf "$T" manifest.json | python3 -c 'import json,sys; [print("  ", m["RepoTags"], m["Config"]) for m in json.load(sys.stdin)]'
