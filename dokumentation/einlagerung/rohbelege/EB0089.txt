#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 (nur lesend): Kassenstand der vier Kassen-Datenbanken (BIOTOP-LAGE-01 AUFBAU FK38 „Kasse“):
Zeilen, jüngste Buchung, Summe der Kostenspalten in den Tabellen `kasse` und `eichkosten`.
Gelesen wird nie das Original selbst: Datei und -wal werden in einen eigenen Ordner tmp/kasse-<marke>/ kopiert (exklusiv),
die Kopie wird gelesen. Aufruf: kasse_stand.py <marke>   (z. B. vorher, nachher)"""
import datetime as dt
import hashlib
import os
import shutil
import sqlite3
import sys
from pathlib import Path

DBS = {'STAMM': '/root/biotop/daten/biotop.db', 'VB1': '/root/phase-4-vorbedingung-1/privat/biotop.db',
       'R02': '/root/reparatur-02/privat-cli/privat.db', 'T3': '/root/biotop-t3/t3.db'}
marke = sys.argv[1]
ziel = Path('/root/biotop-einlagerung/tmp') / f'kasse-{marke}'
ziel.mkdir(parents=True, exist_ok=False)
print(f'Zeitpunkt: {dt.datetime.now():%Y-%m-%d %H:%M:%S} CEST · Marke {marke}')
for name, src in DBS.items():
    vor = {e: (os.path.getsize(src + e) if os.path.exists(src + e) else None) for e in ('', '-wal', '-shm')}
    k = ziel / f'{name}.db'
    shutil.copy2(src, k)
    if vor['-wal']:
        shutil.copy2(src + '-wal', str(k) + '-wal')
    c = sqlite3.connect(k)
    zeile = [f'{name} {src} · Bytes {vor[""]} · -wal {vor["-wal"]} · -shm {vor["-shm"]}']
    for tab in ('kasse', 'eichkosten'):
        try:
            spalten = [r[1] for r in c.execute(f'PRAGMA table_info({tab})')]
        except sqlite3.Error as e:
            zeile.append(f'  {tab}: Fehler {e}')
            continue
        if not spalten:
            zeile.append(f'  {tab}: Tabelle fehlt')
            continue
        n = c.execute(f'SELECT count(*) FROM {tab}').fetchone()[0]
        teile = [f'  {tab}: {n} Zeilen · Spalten {spalten}']
        for s in spalten:
            if any(x in s.lower() for x in ('eur', 'kosten', 'betrag', 'usd')):
                v = c.execute(f'SELECT round(sum({s}), 9) FROM {tab}').fetchone()[0]
                teile.append(f'    Summe {s} = {v}')
            if any(x in s.lower() for x in ('zeit', 'ts', 'datum', 'am')) and not s.lower().startswith('stat'):
                v = c.execute(f'SELECT max({s}) FROM {tab}').fetchone()[0]
                teile.append(f'    jüngster Wert {s} = {v}')
        h = hashlib.sha256()
        for r in c.execute(f'SELECT * FROM {tab} ORDER BY rowid'):
            h.update(repr(r).encode())
        teile.append(f'    SHA-256 über alle Zeilen (repr, nach rowid): {h.hexdigest()}')
        zeile += teile
    c.close()
    print('\n'.join(zeile))
print(f'Kopien (bleiben liegen, eigener Arbeitsordner): {ziel}')
