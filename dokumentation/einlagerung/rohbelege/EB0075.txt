def sha_datei(p) -> str:
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for block in iter(lambda: f.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def exklusiv_schreiben(p: Path, text: str) -> None:
    p.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, 'w', encoding='utf-8', errors='surrogateescape') as f:
        f.write(text)

def aus_grund(p: str):
    if p in AUS_PFADE:
        return AUS_PFADE[p]
    for m, g in AUS_MUSTER:
        if m.search(p):
            return g
    return None


def laufe(wurzel: Path):
    """Liefert (art, pfad, grund) für alles im Umfang."""
    w = str(wurzel)
    if wurzel.is_symlink():
        yield 'verweis', w, None
        return
    if wurzel.is_file():
        if geheim(wurzel.name) or w in GEHEIM_PFADE:
            yield 'aus-geheim', w, 'Geheimnisdatei, nur Pfad'
        else:
            yield 'datei', w, None
        return
    for d, ds, fs in os.walk(w):
        yield 'ordner', d, None
        behalten = []
        for x in sorted(ds):
            p = os.path.join(d, x)
            if os.path.islink(p):
                yield 'verweis', p, None
            elif x in AUSGELASSENE_ORDNER:
                yield 'aus-ordner', p, 'Abhängigkeits- oder Zwischenspeicherordner'
            elif aus_grund(p):
                yield 'aus-pfad', p, aus_grund(p)
            else:
                behalten.append(x)
        ds[:] = behalten
        for f in sorted(fs):
            p = os.path.join(d, f)
            if geheim(f) or p in GEHEIM_PFADE:
                yield 'aus-geheim', p, 'Geheimnisdatei, nur Pfad'
                continue
            if aus_grund(p):
                yield 'aus-pfad', p, aus_grund(p)
                continue
            try:
                st = os.lstat(p)
            except FileNotFoundError:
                yield 'verschwunden', p, 'verschwunden vor dem Kopieren'
                continue
            if stat.S_ISLNK(st.st_mode):
                yield 'verweis', p, None
            elif stat.S_ISREG(st.st_mode):
                yield 'datei', p, None
            else:
                yield 'sonder', p, 'Sonderdatei: Socket, FIFO oder Gerät'


def ist_sqlite(p: str) -> bool:
    try:
        if os.path.getsize(p) < 100:
            return False
        with open(p, 'rb') as h:
            return h.read(16) == b'SQLite format 3\x00'
    except OSError:
        return False

def sichere_db(src: str, dst: str) -> str:
    wal = src + '-wal'
    if src.startswith('/root/.hermes/'):
        weg = 'mode=ro am Original (Hermes schreibt laufend; backup liest einen Schnappschuss)'
        s = sqlite3.connect(f'file:{src}?mode=ro', uri=True)
    elif os.path.exists(wal) and os.path.getsize(wal) > 0:
        TMP.mkdir(parents=True, exist_ok=True)
        td = tempfile.mkdtemp(dir=TMP, prefix='db-')
        k = os.path.join(td, 'k.db')
        shutil.copy2(src, k)
        shutil.copy2(wal, k + '-wal')
        weg = f'Kopie von Datei und -wal ({td}), Backup aus der Kopie (Original unberührt)'
        s = sqlite3.connect(k)
    else:
        weg = 'mode=ro&immutable=1 (keine -wal-Daten; Original und -shm unberührt)'
        s = sqlite3.connect(f'file:{src}?mode=ro&immutable=1', uri=True)
    d = sqlite3.connect(dst)
    with d:
        s.backup(d)
    d.close()
    s.close()
    return weg

                st = os.lstat(p)

def integritaet(wurzel: str = None) -> None:
    dbs = json.loads((P / 'dbliste.json').read_text())
    basis = Path(wurzel) if wurzel else STUFE
    ok, schlecht, fehlt, wichtig = 0, [], [], []
    for p in sorted(dbs):
        k = basis / p.lstrip('/')
        if not k.exists():
            fehlt.append(p)
            continue
        c = sqlite3.connect(f'file:{k}?mode=ro&immutable=1', uri=True)
        r = [x[0] for x in c.execute('PRAGMA integrity_check').fetchall()]
        tab = c.execute("SELECT count(*) FROM sqlite_master WHERE type='table'").fetchone()[0]
        c.close()
        if r == ['ok']:
            ok += 1
        else:
            schlecht.append((p, r[:5]))
        if ('/pytest' not in p and '/tmp' not in p and '/eltern-temp/' not in p) or dbs[p]['bytes'] > 20_000_000:
            wichtig.append(f'  {r[0] if len(r) == 1 else r[:3]} · {tab} Tabellen · Kopie {k.stat().st_size} B · {p}')
    print(f'{jetzt()} · PRAGMA integrity_check über {len(dbs)} Datenbankkopien in {basis}')
    print(f'Ergebnis „ok“: {ok} von {len(dbs)} · nicht ok: {len(schlecht)} · Kopie fehlt: {len(fehlt)}')
    for p, r in schlecht:
        print(f'  NICHT OK: {p} {r}')
    for p in fehlt:
        print(f'  FEHLT: {p}')
    print(f'Einzeln (alle außer Testzwischenständen unter pytest/tmp, und alle > 20 MB): {len(wichtig)}')
    print('\n'.join(wichtig))


def geheimsuche(wurzel: str = None, mit_db: str = '') -> None:
    """Geheimnissuche mit Gegenprobe. Werte aus .env/auth.json nach Art eingeordnet (EB24, EB33, EB34):
    ÖFFENTLICH = auth.active_provider (öffentlicher Anbietername, EB33) und die zwei credential_pool-„source“-Werte
    (Form „env:NAME“, nur Variablennamen, EB34) — gezählt, aber kein Befund; KENNUNG = TELEGRAM_ALLOWED_USERS
    (Philipps Telegram-Kennung); SONST = TERMINAL_MODAL_IMAGE; ZUGANG = alle übrigen (Schlüssel, Token).
    Gegenprobe im Speicher (kein Geheimnis wird auf die Platte geschrieben): an den Inhalt der ersten durchsuchten Datei
    wird ein echter ZUGANG-Wert und ein erfundener OpenRouter-Schlüssel gehängt; beide müssen gemeldet werden."""
    sys.path.insert(0, str(E / 'arbeit'))
    import geheimnis
    werte = geheimnis.werte()
    basis = Path(wurzel) if wurzel else STUFE
    muster = {n: re.compile(m.encode()) for n, m in geheimnis.MUSTER.items()}

def manifest_zeilen(root: Path, ohne: str = None) -> list:
    """ohne: Dateiname auf der obersten Ebene, der nicht ins Manifest kommt (das Manifest selbst liegt im Archiv)."""
    r = str(root)
    zeilen = []
    for d, ds, fs in os.walk(r):
        zeilen.append(f'D\t\t\t{os.path.relpath(d, r)}')
        for x in list(ds):
            if os.path.islink(os.path.join(d, x)):
                fs.append(x)
                ds.remove(x)
        ds.sort()
        for f in fs:
            if ohne and d == r and f == ohne:
                continue
            p = os.path.join(d, f)
            rel = os.path.relpath(p, r)
            st = os.lstat(p)
            if stat.S_ISLNK(st.st_mode):
                zeilen.append(f'L\t{os.readlink(p)}\t\t{rel}')
            elif stat.S_ISREG(st.st_mode):
                zeilen.append(f'F\t{sha_datei(p)}\t{st.st_size}\t{rel}')
            else:
                zeilen.append(f'S\t\t\t{rel}')
    zeilen.sort(key=lambda z: z.split('\t')[3])
    return zeilen


def manifest(baum: str, ziel: str, ohne: str = None) -> dict:
    t0 = time.time()
    z = manifest_zeilen(Path(baum), ohne)
    text = '\n'.join(z) + '\n'
    exklusiv_schreiben(Path(ziel), text)
    arten = {a: sum(1 for x in z if x.startswith(a + '\t')) for a in 'DFLS'}
    byt = sum(int(x.split('\t')[2]) for x in z if x.startswith('F\t'))
    info = {'baum': baum, 'ziel': ziel, 'eintraege': len(z), 'verzeichnisse': arten['D'], 'dateien': arten['F'],
            'verweise': arten['L'], 'sonder': arten['S'], 'bytes': byt,
            'sha256_manifest': hashlib.sha256(text.encode(errors='surrogateescape')).hexdigest(), 'dauer_s': round(time.time() - t0, 1)}
    print(json.dumps(info, ensure_ascii=False))
    return info


def lade_manifest(p: Path) -> dict:
    m = {}
    for z in p.read_text(errors='surrogateescape').splitlines():
        a, x, y, rel = z.split('\t')
        m[rel] = (a, x, y)

def vergleiche_manifeste(a: Path, b: Path, name_a: str, name_b: str) -> list:
    ma, mb = lade_manifest(a), lade_manifest(b)
    nur_a = sorted(set(ma) - set(mb))
    nur_b = sorted(set(mb) - set(ma))
    anders = sorted(k for k in set(ma) & set(mb) if ma[k] != mb[k])
    zf = lambda m: sum(1 for v in m.values() if v[0] == 'F')
    print(f'{name_a}: {len(ma)} Einträge, davon {zf(ma)} Dateien · {name_b}: {len(mb)} Einträge, davon {zf(mb)} Dateien')
    print(f'nur {name_a}: {len(nur_a)} · nur {name_b}: {len(nur_b)} · verschieden: {len(anders)}')
    for k in nur_a[:50]:
        print(f'  NUR {name_a}: {k}')
    for k in nur_b[:50]:
        print(f'  NUR {name_b}: {k}')
    for k in anders[:50]:
        print(f'  VERSCHIEDEN: {k} · {ma[k][1][:16]}… → {mb[k][1][:16]}…')
    leer = not (nur_a or nur_b or anders)
    print('ERGEBNIS: Unterschied leer' if leer else f'ERGEBNIS: Unterschied NICHT leer ({len(nur_a) + len(nur_b) + len(anders)} Einträge)')

def _tar_pigz(quelle_c: Path, eintrag: str, ziel: Path) -> None:
    if ziel.exists():
        raise SystemExit(f'{ziel} besteht schon — nichts überschrieben')
    ziel.parent.mkdir(exist_ok=True)
    t0 = time.time()
    fd = os.open(ziel, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, 'wb') as aus:
        tar = subprocess.Popen(['tar', '--numeric-owner', '--sort=name', '-C', str(quelle_c), '-cf', '-', eintrag],
                               stdout=subprocess.PIPE)
        gz = subprocess.Popen(['pigz', '-6', '-p', '4'], stdin=tar.stdout, stdout=aus)
        tar.stdout.close()
        rc_gz = gz.wait()
        rc_tar = tar.wait()
    os.chmod(ziel, 0o600)
    print(f'tar rc={rc_tar} · pigz rc={rc_gz} · Dauer {time.time() - t0:.1f} s')
    print(f'Archiv {ziel}: {ziel.stat().st_size} Bytes · SHA-256 {sha_datei(ziel)} · Rechte {oct(stat.S_IMODE(ziel.stat().st_mode))}')
    t = subprocess.run(f'pigz -dc {ziel} | tar -tf - | wc -l', shell=True, capture_output=True, text=True)
    print(f'Einträge laut tar -t: {t.stdout.strip()}')


def _manifest_ins_archiv(baum: Path, name: str, kopie: Path) -> dict:

    ziel = baum / name

    with os.fdopen(fd, 'wb') as f:

    print(f'Manifest liegt im Archiv als {name} ({ziel.stat().st_size} B, SHA-256 {sha_datei(ziel)}); '
          f'erwartete Einträge im Archiv: {info["eintraege"]} + 1 (Manifest) = {info["eintraege"] + 1}')
