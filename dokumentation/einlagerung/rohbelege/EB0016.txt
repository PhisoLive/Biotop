#!/bin/bash
# BIOTOP-EINLAGERUNG-01 · Selbsttest von s2_git.py an einem erfundenen Repository (vor dem echten Lauf).
# Fälle: (1) sauberer Durchlauf mit geändert/unverfolgt/Unterordner/ignoriert/vorgemerkt → Commit, Markierungen,
# Status leer, Baum vorher = nachher, alter Zweig unverändert; (2) Geheimnis in einer unverfolgten Datei → Abbruch,
# NICHTS geschrieben (keine Zweige, keine Markierungen, Index unverändert); (3) falsche Anzahl → Abbruch;
# (4) Gegenprobe des Vergleichs schlägt an. Alles unter /root/biotop-einlagerung/tmp/selbsttest-git/.
set -u
S=/root/biotop-einlagerung/tmp/selbsttest-git
P="python3 /root/biotop-einlagerung/arbeit/s2_git.py"
[ -e $S ] && { echo "ABBRUCH: $S besteht"; exit 3; }
mkdir -p $S/a $S/b
for R in $S/a $S/b; do
  git -C $R init -q -b main
  git -C $R -c user.name=t -c user.email=t@t commit -q --allow-empty -m leer
  printf 'eins\n' > $R/eins.txt; printf 'zwei\n' > $R/zwei.txt; printf '*.log\n' > $R/.gitignore
  git -C $R add -A; git -C $R -c user.name=t -c user.email=t@t commit -q -m basis
  printf 'eins geändert\n' > $R/eins.txt
  printf 'neu\n' > $R/neu.txt
  mkdir -p $R/ordner/unter; printf 'a\n' > $R/ordner/unter/a.txt; printf 'b\n' > $R/ordner/b.txt
  printf 'ignoriert\n' > $R/x.log
  printf 'vorgemerkt\n' > $R/vorgemerkt.txt; git -C $R add vorgemerkt.txt
  ln -s eins.txt $R/verweis
done
printf 'sk-or-v1-%s\n' "$(python3 -c 'import secrets;print(secrets.token_hex(32))')" > $S/b/ordner/geheim.txt
B=$(git -C $S/a rev-parse HEAD)
echo "== Fall 1: sauber (erwartet 5 Status-Einträge: M eins, A vorgemerkt, ?? neu, ?? ordner/, ?? verweis)"
$P liste $S/a selbsttest-a-vorher | head -3
$P festschreiben $S/a $B einlagerung/test 5 /root/biotop-einlagerung/arbeit/commit-nachricht-weitere.txt --runde5; echo "rc=$?"
$P liste $S/a selbsttest-a-nachher | head -3
$P vergleich selbsttest-a-vorher selbsttest-a-nachher
echo "-- Kontrolle: x.log nicht im Commit: $(git -C $S/a show --name-only --format= HEAD | grep -c 'x.log')"
echo "-- Kontrolle: Verweis im Commit als Verweis (Modus 120000): $(git -C $S/a ls-tree HEAD verweis | cut -c1-6)"
echo "== Fall 2: Geheimnis in ordner/geheim.txt → Abbruch ohne Schreiben"
B2=$(git -C $S/b rev-parse HEAD); vor=$(sha256sum $S/b/.git/index | cut -c1-16); refs_vor=$(git -C $S/b for-each-ref | wc -l)
$P festschreiben $S/b $B2 einlagerung/test 5 /root/biotop-einlagerung/arbeit/commit-nachricht-weitere.txt --runde5; echo "rc=$?"
echo "-- Index vorher $vor nachher $(sha256sum $S/b/.git/index | cut -c1-16) · Referenzen vorher $refs_vor nachher $(git -C $S/b for-each-ref | wc -l) · Zweig $(git -C $S/b symbolic-ref HEAD)"
echo "== Fall 3: falsche Anzahl (7 statt 5) → Abbruch ohne Schreiben"
rm $S/b/ordner/geheim.txt
$P festschreiben $S/b $B2 einlagerung/test 7 /root/biotop-einlagerung/arbeit/commit-nachricht-weitere.txt; echo "rc=$?"
echo "-- Referenzen nachher $(git -C $S/b for-each-ref | wc -l) · Zweig $(git -C $S/b symbolic-ref HEAD)"
