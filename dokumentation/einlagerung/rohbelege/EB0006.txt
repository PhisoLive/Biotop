#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 Schritt 1 (nur lesend): Wartende Gedächtnis-Änderungen (/root/.hermes/pending/memory/*.json)
mit Feldern id, subsystem, action, summary, origin, created_at; payload nur als Länge und erste 300 Zeichen, Kennungen
ab 8 Ziffern geschwärzt. Dazu: gibt es /opt/hermes/kaltstart.py und /opt/hermes/melden.py, und nennen sie „biotop“?"""
import glob
import json
import os
import re
import time

def schwarz(s: str) -> str:
    return re.sub(r'\d{8,}', '[KENNUNG GESCHWÄRZT]', s)

for p in sorted(glob.glob('/root/.hermes/pending/memory/*.json')):
    d = json.load(open(p))
    pl = json.dumps(d.get('payload'), ensure_ascii=False)
    print(f"--- {p}")
    for k in ('id', 'subsystem', 'action', 'origin'):
        print(f"  {k}: {schwarz(str(d.get(k)))[:200]}")
    ca = d.get('created_at')
    try:
        ca = time.strftime('%Y-%m-%d %H:%M:%S %Z', time.localtime(float(ca)))
    except (TypeError, ValueError):
        pass
    print(f"  created_at: {ca}")
    print(f"  summary: {schwarz(str(d.get('summary')))[:600]}")
    print(f"  payload: {len(pl)} Zeichen · Anfang: {schwarz(pl)[:300]}")
for f in ('/opt/hermes/kaltstart.py', '/opt/hermes/melden.py'):
    if os.path.isfile(f):
        t = open(f, errors='replace').read()
        print(f"{f}: vorhanden, {len(t)} Zeichen, Zeilen mit 'biotop': {sum('biotop' in z.lower() for z in t.splitlines())}; "
              f"Docstring-Anfang: {t[:400]!r}")
    else:
        print(f"{f}: NICHT GEFUNDEN (os.path.isfile falsch; ls: {os.path.exists(f)})")
