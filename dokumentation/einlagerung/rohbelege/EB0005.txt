#!/bin/bash
# BIOTOP-EINLAGERUNG-01 Schritt 1 (nur lesend): Einordnung der Zeitgeber und Warteschlangen, die EB2 fand.
# - je systemd-Zeitgeber: Befehl des Dienstes und Anzahl der Zeilen mit „biotop“ in den aufgerufenen Skriptdateien;
# - /root/.hermes/pending/memory/*.json: Größe, Schlüssel, Treffer „biotop“ (kein Inhalt ausgegeben);
# - Hermes-Konfiguration: Wiederaufnahme nach Neustart (Schlüssel mit resume/restart), .restart_last_processed.json.
date '+== Zeitpunkt %Y-%m-%d %H:%M:%S %Z (UTC%z)'
echo "== systemd-Zeitgeber: Dienst | Befehl | aufgerufene Dateien mit Anzahl Zeilen 'biotop' (Groß/klein egal)"
for t in $(systemctl list-timers --all --no-legend --no-pager | grep -o '[A-Za-z0-9@_.-]*\.timer' | sort -u); do
  s=$(systemctl show -p Unit --value "$t")
  e=$(systemctl show -p ExecStart --value "$s" | grep -o 'argv\[\]=[^;]*' | head -1 | sed 's/argv\[\]=//')
  z=""
  for w in $e; do
    if [ -f "$w" ] && [ "$(stat -c %s "$w")" -lt 5000000 ]; then z="$z $w:$(grep -c -i biotop "$w")"; fi
  done
  echo "$t | $s | $e |$z"
done
echo "== /root/.hermes/pending/memory/*.json"
python3 - <<'EOF'
import json, glob, os
for p in sorted(glob.glob('/root/.hermes/pending/memory/*.json')):
    t = open(p, errors='replace').read()
    try:
        d = json.loads(t)
        keys = list(d) if isinstance(d, dict) else type(d).__name__
    except ValueError:
        keys = 'kein JSON'
    print(p, os.path.getsize(p), 'B · geändert', os.path.getmtime(p), '· Schlüssel', keys, '· Treffer biotop:', t.lower().count('biotop'))
EOF
echo "== config.yaml: Schlüssel mit resume/restart/auto (nur Schlüsselnamen und Werte ohne Geheimnis)"
grep -n -i -E 'resume|restart|auto_|autostart|wake' /root/.hermes/config.yaml | cut -c1-160
echo "== .restart_last_processed.json"
cat /root/.hermes/.restart_last_processed.json; echo
echo "== Skripte in /root/.hermes/scripts mit Biotop-Bezug (Anzahl Zeilen 'biotop')"
grep -c -i biotop /root/.hermes/scripts/* 2>/dev/null
