#!/bin/bash
# BIOTOP-EINLAGERUNG-01 Schritt 2 (nur lesend): jedes Git-Repository mit Biotop-Bezug — HEAD, Zweig, Zweige, Markierungen,
# git status. Alle git-Befehle mit --no-optional-locks (schreiben den Index nicht). Gesucht wird nach „.git“ (Ordner oder
# Datei) in den Orten der Fundstellenliste von BIOTOP-LAGE-01 (AUFBAU FK38, INVENTAR I4/I5, sich.py ARBEITSORDNER) und
# in /tmp. Getrennt gezählt: Kopien in den Probe-/Stufenordnern von BIOTOP-LAGE-01 (sicherung/, testlauf/).
date '+== Zeitpunkt %Y-%m-%d %H:%M:%S %Z (UTC%z)'
ORTE="/root/biotop /root/biotop-t3 /root/biotop-vor-t4 /root/biotop-vor-t4-02 /root/biotop-vor-t4-03 /root/biotop-vor-t4-04
/root/biotop-vor-t4-05 /root/biotop-vor-t4-06 /root/phase-4-vorbedingung-1 /root/reparatur-01 /root/reparatur-02
/root/uebungswerk /root/uebungswerk2 /root/uebungswerk-nachlauf /root/uebungswerk-p2 /root/uebungswerk-p3 /root/uebungswerk-p3b
/root/uebungswerk-p3c /root/uebungswerk-p3d /root/versandwerk-t3 /root/versandwerk-t3-02 /root/gesamtstand-01
/root/wiedereinstieg-01 /root/biotop-einstieg-20260921-0122 /root/traeger-umstellung-2026-09-20_22-32-03Z /root/versiegelt
/root/w7 /root/sicherung/biotop-lage-01 /tmp"
L=/root/biotop-einlagerung/tmp/git-fundorte.txt
: > $L
for o in $ORTE; do
  find "$o" \( -name node_modules -o -name venv -o -name .venv \) -prune -o -name .git -print 2>/dev/null >> $L
done
sort -u -o $L $L
echo "== .git-Fundorte gesamt: $(wc -l < $L)"
echo "   davon in /root/sicherung/biotop-lage-01/{sicherung,testlauf} (Kopien der Probe von BIOTOP-LAGE-01): $(grep -c -E '^/root/sicherung/biotop-lage-01/(sicherung|testlauf)/' $L)"
echo "   davon in /tmp: $(grep -c '^/tmp/' $L)"
echo "== Je Repository außerhalb der Probe-Kopien: Pfad | HEAD | Zweig | Zweige | Markierungen | Status-Zeilen (M/??/sonst)"
grep -v -E '^/root/sicherung/biotop-lage-01/(sicherung|testlauf)/' $L | while read g; do
  r=$(dirname "$g")
  head=$(git --no-optional-locks -C "$r" rev-parse --short=12 HEAD 2>&1 | head -1)
  zw=$(git --no-optional-locks -C "$r" symbolic-ref -q --short HEAD 2>/dev/null || echo "(losgelöst)")
  zws=$(git --no-optional-locks -C "$r" for-each-ref --format='%(refname:short)=%(objectname:short=12)' refs/heads 2>/dev/null | tr '\n' ' ')
  tags=$(git --no-optional-locks -C "$r" tag -l 2>/dev/null | tr '\n' ' ')
  st=$(git --no-optional-locks -C "$r" status --porcelain 2>&1)
  n=$(printf '%s' "$st" | grep -c .)
  nm=$(printf '%s' "$st" | grep -c '^ M\|^M')
  nu=$(printf '%s' "$st" | grep -c '^??')
  top=$(git --no-optional-locks -C "$r" rev-parse --show-toplevel 2>&1)
  echo "$r | HEAD $head | Zweig $zw | Zweige: ${zws:-–} | Markierungen: ${tags:-keine} | Status $n Zeilen ($nm geändert, $nu unverfolgt) | Wurzel $top"
done
echo "== /root/biotop im Einzelnen"
git --no-optional-locks -C /root/biotop log -1 --format='%H %ci %s'
git --no-optional-locks -C /root/biotop for-each-ref --format='%(refname) %(objectname)'
git --no-optional-locks -C /root/biotop status --porcelain=v1 --untracked-files=all
git --no-optional-locks -C /root/biotop stash list | sed 's/^/stash: /'; echo "(stash-Einträge: $(git --no-optional-locks -C /root/biotop stash list | wc -l))"
git --no-optional-locks -C /root/biotop worktree list
echo "== Liste der .git-Fundorte (vollständig)"
cat $L
