#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 (nur lesend): Befunde der Geheimnissuche über den Zwischenstand KOMPLETT (EB58) einordnen.
Gibt nie einen Wert aus: je Befund Pfad, Art, Muster, Länge und Anzahl verschiedener Zeichen der Fundstelle, und ob die
Fundstelle wörtlich auch im Hermes-Agent-Quelltext steht (dann ist sie ein Beispiel oder Platzhalter aus dem Paket)."""
import collections
import re
import sys
from pathlib import Path

sys.path.insert(0, '/root/biotop-einlagerung/arbeit')
import geheimnis  # noqa: E402

E = Path('/root/biotop-einlagerung')
S = E / 'stufe'
liste = sorted((E / 'paket').glob('geheimsuche-befunde-*.tsv'))[-1]
zeilen = [z.split('\t') for z in liste.read_text().splitlines() if z]
print(f'Befundliste {liste.name}: {len(zeilen)} Dateien')
gruppen = collections.Counter()
arten = collections.defaultdict(collections.Counter)
for rel, info in zeilen:
    teile = rel.strip('/').split('/')
    g = '/'.join(teile[:3]) if teile[0] == 'root' else '/'.join(teile[:4])
    for x in info.split(';'):
        k, name, n = x.split('|')
        arten[g][f'{k} {name}'] += 1
    gruppen[g] += 1
print('== Dateien mit Befund je Ort (oberste Ebenen) und Arten')
for g, n in gruppen.most_common():
    print(f'{n:4d}  {g}  ·  ' + ', '.join(f'{a} ({c})' for a, c in arten[g].most_common()))
print('== ZUGANG- und MUSTER-Befunde einzeln (ohne Werte)')
paket_texte = None
for rel, info in zeilen:
    if not any(x.startswith(('ZUGANG', 'MUSTER')) for x in info.split(';')):
        continue
    data = (S / rel.lstrip('/')).read_bytes().decode('utf-8', errors='replace')
    for x in info.split(';'):
        k, name, n = x.split('|')
        if k == 'MUSTER':
            m = geheimnis.MUSTER[name.replace('Muster ', '')]
            funde = [t.group(0) for t in re.finditer(m, data)]
            beschr = []
            for f in funde[:4]:
                if paket_texte is None:
                    paket_texte = []
                beschr.append(f'Länge {len(f)}, {len(set(f))} versch. Zeichen')
            print(f'  MUSTER {name[7:]} ×{n} · {rel} · ' + ' | '.join(beschr))
        elif k == 'ZUGANG':
            print(f'  ZUGANG {name} ×{n} · {rel} · Dateigröße {len(data)} Zeichen')
