#!/usr/bin/env python3
"""BIOTOP-EINLAGERUNG-01 Schritt 2 (nur lesend): EB14 auswerten — Repositorys außerhalb der Probe-Kopien mit nicht
eingecheckten Änderungen (Status-Zeilen > 0), gruppiert nach oberstem Ort; dazu die Zahl der Repositorys je Ort."""
import re
from collections import Counter
from pathlib import Path

t = Path('/root/biotop-einlagerung/rohbelege/EB0014.txt').read_text()
zeilen = [z for z in t.splitlines() if re.match(r'^/\S+ \| HEAD ', z)]
print(f'Repositorys außerhalb der Probe-Kopien: {len(zeilen)}')
je = Counter()
for z in zeilen:
    p = z.split(' | ')[0]
    teile = p.split('/')
    je['/'.join(teile[:3]) if not p.startswith('/tmp') else '/'.join(teile[:3])] += 1
for k, v in sorted(je.items()):
    print(f'  {k}: {v}')
print('Mit nicht eingecheckten Änderungen (Status > 0 Zeilen) oder Fehler:')
n = 0
for z in zeilen:
    m = re.search(r'Status (\d+) Zeilen', z)
    if not m or int(m.group(1)) > 0 or 'fatal' in z:
        n += 1
        print('  ' + z)
print(f'Anzahl: {n}')
