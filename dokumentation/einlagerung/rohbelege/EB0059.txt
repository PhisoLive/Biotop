#!/bin/bash
# BIOTOP-EINLAGERUNG-01 (nur lesend): Vorschau 5 für Schritt 6 (Probe) — uv-Zwischenspeicher (Pakete der Test-venv),
# Tests mit Bezug auf .env/anbieter/node/Netz, Marker, Größe der Testsammlung, laufende Container.
set -u
echo "== Zeitpunkt $(date '+%Y-%m-%d %H:%M:%S %Z (UTC%z)')"
echo "== 1. uv-Zwischenspeicher: Einträge zu den Paketen der Test-venv"
for p in pytest iniconfig packaging pluggy pygments; do
  printf '%-10s ' "$p"; find /root/.cache/uv -maxdepth 4 -iname "${p}-*" 2>/dev/null | head -3 | tr '\n' ' '; echo
done
echo "== 2. Tests/Code mit Bezug auf .env, anbieter, schluessel_laden, socket, urllib, requests"
grep -rn -E "\.env\b|anbieter\.|from biotop import anbieter|from biotop.anbieter|schluessel|urllib|requests\.|httpx" /root/biotop/tests/*.py | grep -v -E "^\S+:\s*#" | cut -c1-200 | head -30
echo "== 3. Testdateien und Testfunktionen (def test_) je Datei, Summe"
cd /root/biotop/tests && for f in test_*.py; do printf '%5d %s\n' "$(grep -c -E '^(async )?def test_' "$f")" "$f"; done | sort -rn | head -60
echo "Summe def test_: $(cat /root/biotop/tests/test_*.py | grep -c -E '^(async )?def test_')"
echo "== 4. Marker integration je Datei"
grep -l -E "mark.integration|pytestmark" /root/biotop/tests/*.py | xargs -r -n1 basename
echo "== 5. Container und Kammer-Sperre"
docker ps -a --format '{{.ID}} {{.Names}} {{.Status}}' | head; echo "Container gesamt: $(docker ps -aq | wc -l)"
ls -la /tmp/biotop-kammer-exklusiv 2>&1 | head -1
echo "== 6. unshare/nsenter vorhanden, Kernel erlaubt Mount-Namensräume"
command -v unshare nsenter; unshare --mount --net --propagation private --fork true && echo "unshare mount+net: ok"
