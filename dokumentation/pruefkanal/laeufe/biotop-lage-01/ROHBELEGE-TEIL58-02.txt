#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil C (nur lesend): Fundstellen zu Messgerätfehlern, Befund-Kennungen und Nummernkreisen.
Durchsucht (nur lesen) die *.md-Dateien der Biotop-Abschlussordner im Prüfkanal (oberste und zweite Ebene) und die
Arbeitsordner /root/biotop-vor-t4*/ (oberste Ebene und bau/*.md), jeweils nach:
  (a) 'Messgerätfehler' / 'Messgerätbefund' / 'Messgerät-Fehler' / 'Messfehler'  -> Zeilen mit Datei:Zeile (gekürzt 260)
  (b) Kennungsmuster: R6-\\d+, U-\\d, M\\d(\\(b\\))?, 'Review ?\\d+ Nr\\. ?\\d+', 'Befund S\\d', 'Vorbedingung \\d+ ' (T4)
      -> Zählung je Muster und Datei
Ausgabe gekürzt; keine Inhalte anderer Vorhaben (nur biotop-* Ordner)."""
import re
from collections import Counter, defaultdict
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
dateien = []
for o in sorted(L.iterdir()):
    if o.is_dir() and o.name.startswith('biotop') and o.name != 'biotop-lage-01':
        dateien += sorted(o.glob('*.md')) + sorted(o.glob('*/*.md'))
for r in sorted(Path('/root').glob('biotop-vor-t4*')):
    if r.is_dir():
        dateien += sorted(r.glob('*.md')) + sorted(r.glob('bau/*.md'))
print(f'Dateien durchsucht: {len(dateien)}')
MF = re.compile(r'Messger(ä|ae)t(fehler|befund|-Fehler|-Funde|ausfall)|Messfehler', re.I)
print('\n== (a) Zeilen mit Messgerätfehler/-befund/Messfehler (je Datei höchstens 6)')
for p in dateien:
    t = p.read_text(errors='replace').splitlines()
    treffer = [(i + 1, z.strip()) for i, z in enumerate(t) if MF.search(z)]
    if treffer:
        rel = str(p).replace(str(L) + '/', 'PK:')
        print(f'-- {rel}: {len(treffer)} Zeilen')
        for n, z in treffer[:6]:
            print(f'   {n}: {z[:260]}')
muster = {
    'R6-n': re.compile(r'\bR6-\d+\b'),
    'U-n': re.compile(r'\bU-[1-9]\b'),
    'Review-Nr': re.compile(r'Review(?:\s*0?\d+)?\s*Nr\.\s*\d+'),
    'M-n': re.compile(r'\bM[1-9](?:\([a-z]\))?\b'),
    'Befund S': re.compile(r'Befund(?:e)?\s+S\d'),
    'VB-T4 n': re.compile(r'Vorbedingung(?:en)?\s+(?:für T4\s+)?\d+'),
}
print('\n== (b) Kennungsmuster je Datei (nur Dateien mit Treffern)')
for p in dateien:
    t = p.read_text(errors='replace')
    z = {k: sorted(set(m.group(0) for m in rx.finditer(t))) for k, rx in muster.items()}
    z = {k: v for k, v in z.items() if v}
    if z:
        rel = str(p).replace(str(L) + '/', 'PK:')
        print(f'-- {rel}')
        for k, v in z.items():
            print(f'   {k}: {v[:40]}{" …" if len(v) > 40 else ""}')
