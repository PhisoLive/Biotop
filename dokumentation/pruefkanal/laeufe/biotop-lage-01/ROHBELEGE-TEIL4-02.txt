#!/usr/bin/env python3
"""BIOTOP-LAGE-01: Rohbelege R<von>..R<bis> zu ausgabe/ROHBELEGE-TEIL<k>-<NN>.txt bündeln (je höchstens 40.000 Bytes).
Geteilt wird an Belegen; ein einzelner Beleg über der Grenze wird an Zeilengrenzen geteilt und als Fortsetzung
markiert. Jede Zieldatei wird exklusiv angelegt (nie überschrieben). Aufruf: rb_buendeln.py <NN> <von> <bis>"""
import os
import sys
from pathlib import Path

W = Path('/root/sicherung/biotop-lage-01')
MAX = 40_000
KOPF_RESERVE = 400


def stuecke(nr: int, text: str):
    """Ein Beleg als Liste von Stücken (je < MAX - KOPF_RESERVE Bytes), an Zeilengrenzen geteilt."""
    grenze = MAX - KOPF_RESERVE - 200
    b = text.encode('utf-8')
    if len(b) <= grenze:
        return [text]
    teile, akt, groesse = [], [], 0
    for z in text.splitlines(keepends=True):
        zb = len(z.encode('utf-8'))
        if zb > grenze:  # überlange Zeile hart an Zeichen teilen
            while z:
                stueck = z[: grenze // 4]
                z = z[len(stueck):]
                if groesse + len(stueck.encode('utf-8')) > grenze and akt:
                    teile.append(''.join(akt)); akt, groesse = [], 0
                akt.append(stueck); groesse += len(stueck.encode('utf-8'))
            continue
        if groesse + zb > grenze and akt:
            teile.append(''.join(akt)); akt, groesse = [], 0
        akt.append(z); groesse += zb
    if akt:
        teile.append(''.join(akt))
    return [t if i == 0 else f'[R{nr} · Fortsetzung, Stück {i + 1} von {len(teile)}]\n' + t for i, t in enumerate(teile)]


def main() -> None:
    nn, von, bis = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
    alle = []
    for n in range(von, bis + 1):
        p = W / 'rohbelege' / f'R{n:04d}.txt'
        if not p.exists():
            sys.exit(f'ABBRUCH: {p} fehlt')
        alle += [(n, s) for s in stuecke(n, p.read_text(encoding='utf-8'))]
    dateien, akt, groesse = [], [], 0
    for n, s in alle:
        sb = len(s.encode('utf-8')) + 2
        if groesse + sb > MAX - KOPF_RESERVE and akt:
            dateien.append(akt); akt, groesse = [], 0
        akt.append((n, s)); groesse += sb
    if akt:
        dateien.append(akt)
    (W / 'ausgabe').mkdir(exist_ok=True)
    for k, inhalt in enumerate(dateien, 1):
        name = f'ROHBELEGE-TEIL{k}-{nn}.txt'
        rs = sorted({n for n, _ in inhalt})
        kopf = (f'BIOTOP-LAGE-01 · ROHBELEGE Stand {nn} · Teil {k} von {len(dateien)} · enthält R{rs[0]} bis R{rs[-1]}\n'
                f'Jeder Beleg: Befehl wörtlich (mit dem Inhalt aufgerufener Skriptdateien) und vollständige Ausgabe.\n'
                f'Zeiten in CEST (UTC+0200), soweit nicht anders angegeben.\n\n')
        daten = (kopf + '\n\n'.join(s for _, s in inhalt)).encode('utf-8')
        fd = os.open(W / 'ausgabe' / name, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
        with os.fdopen(fd, 'wb') as f:
            f.write(daten)
        print(f'{name}: {len(daten)} Bytes, R{rs[0]}–R{rs[-1]}')


if __name__ == '__main__':
    main()

#!/usr/bin/env python3
"""BIOTOP-LAGE-01: Prüfkanal-Werkzeug.
Veröffentlicht Dateien aus /root/sicherung/biotop-lage-01/ausgabe/ nach laeufe/biotop-lage-01/ im Prüfkanal:
- exklusiv (O_EXCL): eine vorhandene Datei wird NIE überschrieben (Regel 2.5), der Aufruf bricht dann ab;
- höchstens 40.000 Bytes je Datei (Abschnitt 8), sonst Abbruch vor dem Schreiben;
- Rechte root:caddy, Ordner 750, Dateien 640 (wie alle Biotop-Ordner);
- danach Abruf jeder Datei über HTTPS von außen, Vergleich Bytes und SHA-256.
Adressen werden aus /etc/pruefkanal/schluessel gebaut.
Aufruf: python3 pk.py <name> [<name> ...]      (Namen relativ zu ausgabe/)
        python3 pk.py --abruf <name> [...]      (nur abrufen und vergleichen, nichts schreiben)

import grp
import hashlib
import os
import sys
import urllib.error
import urllib.request
from pathlib import Path

AUSGABE = Path('/root/sicherung/biotop-lage-01/ausgabe')
K = Path('/etc/pruefkanal/schluessel').read_text().strip()
ZIEL = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe/biotop-lage-01')
BASIS = f'https://milieulux-reinigung-muenchen.de/_akte/{K}/laeufe/biotop-lage-01'
GID = grp.getgrnam('caddy').gr_gid
MAX = 40_000


def url(name: str) -> str:
    return f'{BASIS}/{name}'


def abrufen(name: str):
    q = urllib.request.Request(url(name), headers={'User-Agent': 'biotop-lage-01-abruf'})
    try:
        with urllib.request.urlopen(q, timeout=60) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, b''


def pruefen(name: str, daten: bytes) -> bool:
    h = hashlib.sha256(daten).hexdigest()
    st, ab = abrufen(name)
    gleich = st == 200 and hashlib.sha256(ab).hexdigest() == h
    print(f'{name} | {len(daten)} Bytes | sha256 {h} | HTTP {st} | abgerufen {len(ab)} Bytes | '
          f'gleich {"ja" if gleich else "NEIN"}')
    print(f'  {url(name)}')
    return gleich


def veroeffentlichen(name: str) -> bool:
    daten = (AUSGABE / name).read_bytes()
    if len(daten) > MAX:
        sys.exit(f'ABBRUCH {name}: {len(daten)} Bytes > {MAX} (Abschnitt 8) — nichts geschrieben')
    if not ZIEL.exists():
        ZIEL.mkdir(mode=0o750)
        os.chown(ZIEL, 0, GID)
        os.chmod(ZIEL, 0o750)
        print(f'Ordner angelegt: {ZIEL.name} (root:caddy 750)')
    z = ZIEL / name
    fd = os.open(z, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o640)
    with os.fdopen(fd, 'wb') as f:
        f.write(daten)
    os.chown(z, 0, GID)
    os.chmod(z, 0o640)
    return pruefen(name, daten)


def main() -> None:
    args = sys.argv[1:]
    nur_abruf = bool(args) and args[0] == '--abruf'
