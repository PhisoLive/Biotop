#!/usr/bin/env python3
"""BIOTOP-LAGE-01 · Einsetzen der Entwurfsteile in die Dateien des Abschlussordners (Abschnitt 8), Stand 03.
Je Zeile der Tafel TEILE: Entwurfsdatei (oder ein Zeilenbereich daraus) → ausgabe/<ziel>.
- Kopf (nur .md): Kennung, Fragen, Stichzeitpunkt, Verfasser und Arbeitszeit, Quelle mit Bytes und SHA-256,
  Änderungen beim Einsetzen, Verweis auf die Prüfung, Nachtrag Hermes (entwurf/hermes/nachtrag/<ziel>, sonst
  entwurf/hermes/nachtrag/ALLGEMEIN.md).
- .txt über 40.000 B nach dem Einsetzen: an Zeilengrenzen geteilt (je ≤ 38.000 B), die folgenden Nummern rücken nach;
  jeder Teil beginnt mit einer Zeile „# <Quelle> Zeilen a–b von n (geteilt beim Einsetzen)“.
- `{LAEUFE}` → volle Adresse; jeder andere Platzhalter `{GROSSBUCHSTABEN}` bricht ab.
- Prüfungen: Größe ≤ 40.000 B; jede `{LAEUFE}/…`-Adresse im Prüfkanal vorhanden (Musterangaben mit …, *, < oder
  endendem - werden als Muster gezählt, nicht geprüft); jede Rohbeleg-Nennung R<n> als rohbelege/R<nnnn>.txt vorhanden.
- Schreibt exklusiv. Liegt das Ziel schon in ausgabe/: Abbruch, außer mit --ersetzen und nur, wenn es noch nicht im
  Prüfkanal liegt. --probe: nur prüfen und zählen, nichts schreiben.
Aufruf: einsetzen.py [--probe] [--ersetzen] <zielpräfix> [...]"""
import datetime as dt
import hashlib
import os
import re
import sys
from pathlib import Path

W = Path('/root/sicherung/biotop-lage-01')
E = W / 'entwurf'
A = W / 'ausgabe'
K = Path('/etc/pruefkanal/schluessel').read_text().strip()
PK = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
LAEUFE = f'https://milieulux-reinigung-muenchen.de/_akte/{K}/laeufe'
ORDNER = f'{LAEUFE}/biotop-lage-01'
MAX, TEIL_MAX = 40_000, 38_000
START_TEILE = dt.datetime(2026, 9, 25, 16, 10, 47)
DAUER = {'A': 5774.98, 'B': 5524.72, 'C': 4285.17, 'D': 5904.11, 'E': 9885.15, 'F': 7214.53, 'G': 7523.87,
         'H': 5767.94}
ersetze = lambda t: t.replace('{LAEUFE}', LAEUFE)

TEILE = []  # (teil, quelle, ziel, k, n, fragen, zeilenbereich oder None)
def teile(teil, quelle, ziel, n, fragen, nummern=None):
    for k in range(1, n + 1):
        TEILE.append((teil, quelle.format(k=k), ziel.format(k=(nummern[k - 1] if nummern else k)), k, n, fragen, None))
def teile_geteilt(teil, quelle, ziel, n, fragen):
    """.txt-Rohlisten: zu große Teile an Zeilengrenzen teilen, Zielnummern fortlaufend."""
    z = 0
    for k in range(1, n + 1):
        q = quelle.format(k=k)
        zeilen = ersetze((E / q).read_text(encoding='utf-8')).splitlines(keepends=True)
        if sum(len(s.encode()) for s in zeilen) <= MAX:
            z += 1; TEILE.append((teil, q, ziel.format(k=z), k, n, fragen, None)); continue
        a, groesse = 0, 0
        for i, s in enumerate(zeilen + [None]):
            b = len(s.encode()) if s is not None else 0
            if s is None or groesse + b > TEIL_MAX - 200:
                z += 1; TEILE.append((teil, q, ziel.format(k=z), k, n, fragen, (a, i, len(zeilen))))
                a, groesse = i, 0
            groesse += b
teile('A', 'entscheidungen/ENTSCHEIDUNGEN-TEIL{k}.md', 'ENTSCHEIDUNGEN-TEIL{k}-01.md', 4, 'FK15–FK19, FK21')
teile('B', 'verfassungen/VERFASSUNGEN-TEIL{k}.md', 'VERFASSUNGEN-TEIL{k}-01.md', 4, 'FK22')
teile('C', 'tore/TORE-TEIL{k}.md', 'TORE-TEIL{k}-01.md', 4, 'FK24, FK27–FK29')
teile('D', 'aufbau/AUFBAU-TEIL{k}.md', 'AUFBAU-TEIL{k}-01.md', 3, 'FK31–FK38')
teile('D', 'aufbau/GLOSSAR.md', 'GLOSSAR-TEIL1-01.md', 1, 'FK50')
teile('E', 'soll-ist/SOLL-IST-TEIL{k}.md', 'SOLL-IST-TEIL{k}-01.md', 4, 'FK30')
teile('F', 'chronik/CHRONIK-TEIL{k}.md', 'CHRONIK-TEIL{k}-01.md', 3, 'FK10, FK12, FK20')
teile('G', 'lage-werke/LAGE-BAUSTEINE-G-TEIL{k}.md', 'LAGE-TEIL{k}-01.md', 2,
      'Abgleich des Wissensstands WS01–WS19; FK23, FK25, FK26, FK44, FK45', nummern=[2, 3])
for k in (1, 2, 3):
    TEILE.append(('H', f'lage-offen/LAGE-BAUSTEINE-H-TEIL{k}.md', f'LAGE-TEIL{k + 3}-01.md', k, 3,
                  ['FK08, FK09, FK13', 'FK39', 'FK40–FK43'][k - 1], None))
teile('I', 'inventar/lauf-stich-v3/INVENTAR-TEIL{k}.md', 'INVENTAR-TEIL{k}-01.md', 3, 'Abschnitt 5')
teile_geteilt('I', 'inventar/lauf-stich-v3/LISTEN-TEIL{k}.txt', 'LISTEN-TEIL{k}-01.txt', 6, 'Abschnitt 5, Rohlisten')
teile('I', 'inventar/lauf-stich-v3/KASSE-TEIL{k}.csv', 'KASSE-TEIL{k}-01.csv', 28, 'Abschnitt 5, Rohlisten')
for name in sorted(p.name for p in (E / 'hermes').glob('*.md')) if (E / 'hermes').exists() else []:
    TEILE.append(('Hermes', f'hermes/{name}', name, 0, 0, '', None))

sha = lambda b: hashlib.sha256(b).hexdigest()
R_MUSTER = re.compile(r'(?<![\w\-/.])R0*(\d{1,4})(?![\w\-])')
L_MUSTER = re.compile(r'\{LAEUFE\}/([^\s)`\'"|<>\]\[,;]+)')
MAX_R = max(int(p.stem[1:]) for p in (W / 'rohbelege').glob('R*.txt'))


def kopf(teil, quelle, ziel, k, n, fragen, roh, stellen):
    if teil == 'Hermes':
        return ''
    ende = START_TEILE + dt.timedelta(seconds=DAUER[teil]) if teil in DAUER else None
    arbeit = (f'Subagent Teil {teil}, Arbeitszeit 25.09.2026 {START_TEILE:%H:%M:%S}–{ende:%H:%M:%S} CEST '
              f'(Delegationsbuch, R193)' if ende else
              'Skript `entwurf/inventar/inventar.py` (Subagent Teil I), ausgeführt von Hermes zum Stichzeitpunkt (R191)')
    n_p = E / 'hermes' / 'nachtrag' / ziel
    n_p = n_p if n_p.exists() else E / 'hermes' / 'nachtrag' / 'ALLGEMEIN.md'
    nachtrag = n_p.read_text(encoding='utf-8').strip() if n_p.exists() else 'keiner.'
    z = [f'# BIOTOP-LAGE-01 · Fassung 2 · `{ziel}`', '',
         f'- **Inhalt:** {fragen}' + (f' · Teil {k} von {n}' if n > 1 else ''),
         '- **Stichzeitpunkt:** 25.09.2026 17:29:59 CEST (Ende des Baus S1c). Der Entwurf beschreibt den Stand zur '
         'Zeit seines Lesens; was sich bis zum Stich geändert hat, steht im Nachtrag unten.',
         f'- **Verfasser:** {arbeit}. Eingesetzt von Hermes.',
         f'- **Quelle:** `/root/sicherung/biotop-lage-01/entwurf/{quelle}` · {len(roh)} B · SHA-256 `{sha(roh)}`. '
         f'Änderungen beim Einsetzen: nur dieser Kopf und `{{LAEUFE}}` → `{LAEUFE}` ({stellen} Stellen).',
         f'- **Prüfung durch Hermes und Rohbelege je Datei:** {ORDNER}/VOLLSTAENDIGKEIT-TEIL1-01.md',
         f'- **Nachtrag Hermes:** {nachtrag}', '', '---', '']
    return '\n'.join(z)


def main():
    args = sys.argv[1:]
    probe, ersetzen = '--probe' in args, '--ersetzen' in args
    praefixe = [a for a in args if not a.startswith('--')]
    auswahl = [t for t in TEILE if any(t[2].startswith(p) for p in praefixe)]
    print(f'{dt.datetime.now():%Y-%m-%d %H:%M:%S} · Auswahl {praefixe}: {len(auswahl)} von {len(TEILE)} Zeilen der Tafel '
          f'· Modus {"PROBE" if probe else "SCHREIBEN"} · höchster Rohbeleg R{MAX_R}')
    fehler, adr_ok, muster, adr_fehlt, r_fehlt, summe = 0, 0, 0, [], [], 0
    for teil, quelle, ziel, k, n, fragen, bereich in auswahl:
        q = E / quelle
        if not q.exists():
            print(f'FEHLT Quelle: {quelle}'); fehler += 1; continue
        roh = q.read_bytes()
        text = roh.decode('utf-8')
        if bereich:
            a, b, gesamt = bereich
            teil_zeilen = text.splitlines(keepends=True)[a:b]
            text = f'# {Path(quelle).name} Zeilen {a + 1}–{b} von {gesamt} (geteilt beim Einsetzen, Grenze 40.000 B)\n' \
                   + ''.join(teil_zeilen)
        andere = sorted(set(re.findall(r'\{[A-ZÄÖÜ_]{3,}\}', text)) - {'{LAEUFE}'})
        pfade = [p.rstrip('.:') for p in L_MUSTER.findall(text)]
        ist_muster = lambda p: any(c in p for c in '…*<{') or p.endswith('-') or not p
        muster += sum(map(ist_muster, pfade))
        echt = [p for p in pfade if not ist_muster(p)]
        da = [p for p in echt if (PK / p).exists()]
        weg = sorted(set(p for p in echt if not (PK / p).exists()))
        adr_ok += len(da); adr_fehlt += [f'{ziel}: {p}' for p in weg]
        rs = sorted({int(m) for m in R_MUSTER.findall(text)})
        rw = [r for r in rs if not (W / 'rohbelege' / f'R{r:04d}.txt').exists()]
        r_fehlt += [f'{ziel}: R{r}' for r in rw]
        stellen = text.count('{LAEUFE}')
        neu = ersetze(text)
        if ziel.endswith('.md'):
            neu = kopf(teil, quelle, ziel, k, n, fragen, roh, stellen) + neu
        b = neu.encode('utf-8')
        summe += len(b)
        zeile = (f'{ziel} ← {quelle}{f" Z. {bereich[0] + 1}–{bereich[1]}" if bereich else ""} · Teil {teil} · {len(b)} B · '
                 f'{{LAEUFE}} {stellen} · Adressen {len(da)} da, {len(weg)} fehlen · Rohbelege {len(rs)}, fehlen {len(rw)}')
        if andere:
            zeile += f' · ANDERE PLATZHALTER {andere}'; fehler += 1
        if len(b) > MAX:
            zeile += f' · ZU GROSS (> {MAX})'; fehler += 1
        print(zeile)
        if probe or andere or len(b) > MAX:
            continue
        z = A / ziel
        if z.exists():
            if (PK / 'biotop-lage-01' / ziel).exists() or not ersetzen:
                print(f'   ABBRUCH für {ziel}: liegt schon in ausgabe/' +
                      (' und im Prüfkanal' if (PK / 'biotop-lage-01' / ziel).exists() else ' (ohne --ersetzen)'))
                fehler += 1; continue
            z.unlink()
        fd = os.open(z, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
        with os.fdopen(fd, 'wb') as f:
            f.write(b)
        print(f'   geschrieben: ausgabe/{ziel} · SHA-256 {sha(b)}')
    print(f'Dateien {len(auswahl)} · Bytes zusammen {summe}')
    print(f'Adressen {{LAEUFE}}/…: {adr_ok} geprüft und vorhanden, {len(adr_fehlt)} fehlen, {muster} Musterangaben (nicht geprüft)')
    for a in adr_fehlt:
        print(f'   FEHLT im Prüfkanal: {a}')
    print(f'Rohbeleg-Nennungen ohne Datei: {len(r_fehlt)}')
    for r in r_fehlt:
        print(f'   {r}')
    print(f'Fehler: {fehler}')
    sys.exit(1 if fehler else 0)


if __name__ == '__main__':
    main()

#!/usr/bin/env python3
"""BIOTOP-LAGE-01: Stand der OpenRouter-Nutzungsanzeige (kein Modellaufruf; Regel 2.3 erlaubt die Abfrage).
Der Schlüssel wird aus /root/.hermes/.env gelesen und nie ausgegeben. Ausgegeben werden nur die Zahlen der Anzeige:
/api/v1/key (usage gesamt, heute, Woche, Monat; Limit) und /api/v1/credits (gekauft, verbraucht).
Aufruf: python3 or_stand.py <marke>   — hängt eine JSON-Zeile an /root/sicherung/biotop-lage-01/arbeit/OR-STAND.jsonl."""
import json
import sys
import time
import urllib.error
import urllib.request
from pathlib import Path

schluessel = None
for z in Path('/root/.hermes/.env').read_text().splitlines():
    if z.startswith('OPENROUTER_API_KEY='):
        schluessel = z.split('=', 1)[1].strip().strip('"').strip("'")
if not schluessel:
    sys.exit('OPENROUTER_API_KEY fehlt in /root/.hermes/.env')


def hole(url):
    q = urllib.request.Request(url, headers={'Authorization': f'Bearer {schluessel}'})
    try:
        with urllib.request.urlopen(q, timeout=60) as r:
            return r.status, json.loads(r.read())
    except urllib.error.HTTPError as e:
        return e.code, {'fehler': e.read().decode('utf-8', 'replace')[:300]}


marke = sys.argv[1] if len(sys.argv) > 1 else 'stand'
zeile = {'zeit': time.strftime('%Y-%m-%dT%H:%M:%S%z'), 'marke': marke}
st, k = hole('https://openrouter.ai/api/v1/key')
d = k.get('data') or {}
zeile['key_http'] = st
for f in ('usage', 'usage_daily', 'usage_weekly', 'usage_monthly', 'byok_usage', 'limit', 'limit_remaining',
          'limit_reset', 'is_free_tier', 'label'):
    if f in d:
        zeile['key_' + f] = d[f] if f != 'label' else '[GESCHWÄRZT: Schlüsselbezeichnung]'
st, c = hole('https://openrouter.ai/api/v1/credits')
zeile['credits_http'] = st
cd = c.get('data') or {}
zeile['credits_total_credits'] = cd.get('total_credits')
zeile['credits_total_usage'] = cd.get('total_usage')
with open('/root/sicherung/biotop-lage-01/arbeit/OR-STAND.jsonl', 'a') as f:
    f.write(json.dumps(zeile, ensure_ascii=False) + '\n')
print(json.dumps(zeile, ensure_ascii=False, indent=1))
