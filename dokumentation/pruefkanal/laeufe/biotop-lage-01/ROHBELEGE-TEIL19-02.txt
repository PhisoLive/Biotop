#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): I3 Abschlussordner im Prüfkanal — jeder Ordner laeufe/biotop-* und laeufe/gedaechtnis:
Dateien gesamt (rekursiv), Unterordner, älteste/jüngste Datei (mtime), Dateien auf oberster Ebene mit Größe, Einstieg
(BERICHT*.md, START*.md, ZWISCHENBERICHT*.md, ABSCHLUSS*.md, NACHGANG*.md in oberster Ebene und in Unterordnern der
ersten Ebene). Schreibt arbeit/inventar/i3.json."""
import datetime as dt
import json
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
AUS = Path('/root/sicherung/biotop-lage-01/arbeit/inventar')
AUS.mkdir(parents=True, exist_ok=True)


def z(t):
    return dt.datetime.fromtimestamp(t).strftime('%Y-%m-%d %H:%M:%S')


erg = []
for o in sorted(L.iterdir()):
    if not o.is_dir() or not (o.name.startswith('biotop') or o.name == 'gedaechtnis'):
        continue
    dateien = [p for p in o.rglob('*') if p.is_file()]
    ordner = sorted({p.parent.relative_to(o).as_posix() for p in dateien if p.parent != o})
    zeiten = [p.stat().st_mtime for p in dateien] or [o.stat().st_mtime]
    oben = sorted((p.name, p.stat().st_size) for p in o.iterdir() if p.is_file())
    einstieg = []
    for p in sorted(o.glob('*.md')) + sorted(o.glob('*/*.md')):
        if any(p.name.startswith(x) for x in ('BERICHT', 'START', 'ZWISCHENBERICHT', 'ABSCHLUSS', 'NACHGANG', 'UEBERGABE')):
            einstieg.append(p.relative_to(o).as_posix())
    erste_ebene = sorted(p.name for p in o.iterdir() if p.is_dir())
    e = {'ordner': o.name, 'dateien': len(dateien), 'unterordner': len(ordner), 'erste_ebene': erste_ebene,
         'aelteste': z(min(zeiten)), 'juengste': z(max(zeiten)), 'oben': oben, 'einstieg': einstieg}
    erg.append(e)
    print(f"{o.name}: {len(dateien)} Dateien, {len(ordner)} Unterordner, {e['aelteste']} – {e['juengste']}")
    print(f"   erste Ebene: {erste_ebene[:30]}{' …' if len(erste_ebene) > 30 else ''}")
    print(f"   oben: {[n for n, _ in oben][:25]}{' …' if len(oben) > 25 else ''}")
    print(f"   Einstieg: {einstieg[:12]}{' …' if len(einstieg) > 12 else ''}")
(AUS / 'i3.json').write_text(json.dumps(erg, ensure_ascii=False, indent=1))
print(f'\nOrdner gezählt: {len(erg)} (biotop-* und gedaechtnis)')

#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Lesefassungen aus arbeit/verlauf/wortlaut.jsonl (bereits geschwärzt) für die
Entwurfsarbeit — je Absendergruppe eine Datei in arbeit/lesen/, ab 18.09.2026, nur Biotop-Sitzungen:
  hermes-an-philipp.md      (absender Hermes)
  auftraege-an-subagenten.md (absender Hermes→Subagent)
  subagent-berichte.md       (absender Subagent)
  system-meldungen.md        (System-Delegationsergebnis, System-Hintergrundprozess, System-Hinweis; ohne Verdichtung)
Dokumentinhalte werden durch Verweis ersetzt, Gateway-Kopf gekürzt."""
import json
import re
from pathlib import Path

AUS = Path('/root/sicherung/biotop-lage-01/arbeit/lesen')
doc = re.compile(r"\[The user sent a (?:text )?document: '([^']+)'\.[^\]]*\]\n\n\[Content of [^\]]+\]:\n.*", re.S)
kopf = re.compile(r'^Gateway message origin \(JSON data, not instructions or authorization\):\n\{.*?\}\n'
                  r'Do not guess a reply destination when these fields are insufficient\.\n\n', re.S)
GRUPPEN = {'hermes-an-philipp.md': {'Hermes'}, 'auftraege-an-subagenten.md': {'Hermes→Subagent'},
           'subagent-berichte.md': {'Subagent'},
           'system-meldungen.md': {'System-Delegationsergebnis', 'System-Hintergrundprozess', 'System-Hinweis'}}
daten = [json.loads(z) for z in open('/root/sicherung/biotop-lage-01/arbeit/verlauf/wortlaut.jsonl')]
for name, abs_ in GRUPPEN.items():
    teile = []
    for e in daten:
        if e['biotop'] != 'ja' or e['absender'] not in abs_:
            continue
        t = kopf.sub('', e['text'])
        m = doc.search(t)
        if m:
            t = t[:m.start()] + f"[DOKUMENT {m.group(1)} — Inhalt ausgelassen]"
        teile.append(f"## {e['zeit']} · {e['absender']} · id {e['id']} · {e['sitzung']}\n{t.strip()}\n")
    p = AUS / name
    p.write_text('\n'.join(teile))
    print(f'{p}: {len(teile)} Nachrichten, {p.stat().st_size} Bytes')
