#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Abhängigkeiten zum Wiederherstellen (Abschnitt 4 Umfang 1 und Vorgehen 9).
1. Python-Umgebungen unter /root: Pakete mit Version aus den *.dist-info-Ordnern (gelesen, nichts ausgeführt).
2. Sperr- und Anforderungsdateien in /root/biotop und den Werk-/Zweigordnern: Pfad, Bytes, SHA-256.
3. Node: Version jedes vorhandenen node-Programms.
4. Docker: Abbilder (docker images) und je Abbild nur die Versionsvariablen aus docker image inspect."""
import glob
import hashlib
import os
import subprocess
from pathlib import Path

sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
run = lambda a: subprocess.run(a, capture_output=True, text=True, timeout=60)

print('== 1. Python-Umgebungen')
venvs = sorted(set(glob.glob('/root/*/venv') + glob.glob('/root/*/.venv') + glob.glob('/root/*/*/venv')))
for v in venvs:
    cfg = Path(v) / 'pyvenv.cfg'
    info = ' · '.join(z.strip() for z in cfg.read_text().splitlines() if z.strip()) if cfg.exists() else 'pyvenv.cfg fehlt'
    pakete = sorted({Path(d).name[:-len('.dist-info')] for d in glob.glob(f'{v}/lib/python*/site-packages/*.dist-info')},
                    key=str.lower)
    print(f'{v}: {len(pakete)} Pakete · {info}')
    for i in range(0, len(pakete), 6):
        print('   ' + ' · '.join(pakete[i:i + 6]))

print('== 2. Sperr- und Anforderungsdateien')
NAMEN = {'requirements.txt', 'requirements-dev.txt', 'pyproject.toml', 'setup.cfg', 'setup.py', 'Pipfile', 'Pipfile.lock',
         'poetry.lock', 'uv.lock', 'package.json', 'package-lock.json', 'pnpm-lock.yaml', 'yarn.lock', 'tsconfig.json',
         'pytest.ini', 'tox.ini', '.python-version', '.nvmrc', 'Dockerfile', 'vitest.config.ts', 'vitest.config.js'}
AUS = {'node_modules', 'venv', '.venv', '.git', '__pycache__', '.pytest_cache', '.mypy_cache', 'dist', 'build'}
WURZELN = ['/root/biotop', '/root/biotop-t3', '/root/versandwerk-t3-02', '/root/versandwerk-t3', '/root/reparatur-01',
           '/root/reparatur-02', '/root/phase-4-vorbedingung-1']
n = 0
for w in WURZELN:
    if not os.path.isdir(w):
        print(f'{w}: fehlt')
        continue
    treffer = []
    for wurzel, dirs, files in os.walk(w):
        dirs[:] = sorted(d for d in dirs if d not in AUS)
        treffer += [os.path.join(wurzel, f) for f in sorted(files)
                    if f in NAMEN or f.startswith('requirements') and f.endswith('.txt') or f.startswith('Dockerfile')]
    print(f'{w}: {len(treffer)} Dateien')
    for p in treffer[:40]:
        print(f'   {p} · {os.path.getsize(p)} B · {sha(p)}')
    if len(treffer) > 40:
        print(f'   … {len(treffer) - 40} weitere (Werkkopien), nicht einzeln aufgeführt')
    n += len(treffer)
print(f'zusammen: {n} Dateien')

print('== 3. Node')
kandidaten = sorted({p for p in ['/usr/bin/node', '/usr/local/bin/node', '/opt/node22/bin/node'] + glob.glob('/opt/*/bin/node')
                     + glob.glob('/root/.nvm/versions/node/*/bin/node') if os.path.exists(p)})
for p in kandidaten:
    r = run([p, '--version'])
    print(f'{p} → {os.path.realpath(p)} · {r.stdout.strip() or r.stderr.strip()}')
for p in ['/usr/bin/npm', '/usr/local/bin/npm', '/opt/node22/bin/npm']:
    if os.path.exists(p):
        r = run([p, '--version'])
        print(f'{p} · {r.stdout.strip() or r.stderr.strip()[:200]}')
print(f'python3: {run(["/usr/bin/python3", "--version"]).stdout.strip()}')

print('== 4. Docker')
r = run(['docker', 'images', '--format', '{{.Repository}}:{{.Tag}}\t{{.ID}}\t{{.Size}}\t{{.CreatedAt}}'])
print(r.stdout.rstrip() or r.stderr.rstrip())
for zeile in r.stdout.splitlines():
    bild = zeile.split('\t')[1]
    e = run(['docker', 'image', 'inspect', '--format', '{{range .Config.Env}}{{println .}}{{end}}', bild])
    versionen = [z for z in e.stdout.splitlines() if 'VERSION' in z.split('=')[0]]
    c = run(['docker', 'image', 'inspect', '--format', '{{json .Config.Cmd}} {{json .Config.Entrypoint}} {{.Os}}/{{.Architecture}}', bild])
    print(f'{zeile.split(chr(9))[0]} ({bild}): {" · ".join(versionen) or "keine Versionsvariable"} · {c.stdout.strip()}')
