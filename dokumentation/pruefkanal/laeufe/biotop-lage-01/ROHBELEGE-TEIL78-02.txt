#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Deckt die Biotop-Kette der Sitzungsdatenbank alle Biotop-Nachrichten ab?
Für jede Sitzung ab 18.09.2026 (Quelle telegram, cli, cron, subagent …): in der Kette ja/nein, Eltern, Beginn,
Nachrichtenzahl, und — nur für Sitzungen AUSSERHALB der Kette — wie viele Nachrichten „biotop“ oder Biotop-Kennwörter
enthalten (Zählung, kein Inhalt; Regel 2.4: Inhalte anderer Vorhaben bleiben draußen). Titel nur für Kettensitzungen."""
import datetime as dt
import re
import sqlite3

AB = dt.datetime(2026, 9, 18).timestamp()
START = '20260918_205818_3767d1c9'
con = sqlite3.connect('file:/root/.hermes/state.db?mode=ro', uri=True)
sess = {r[0]: {'source': r[1], 'title': r[2], 'parent': r[3], 'started': r[4]} for r in con.execute(
    'SELECT id, source, title, parent_session_id, started_at FROM sessions')}
kinder = {}
for sid, s in sess.items():
    kinder.setdefault(s['parent'], []).append(sid)
kette, stapel = set(), [START]
while stapel:
    x = stapel.pop()
    if x in kette:
        continue
    kette.add(x)
    stapel += kinder.get(x, [])
WORT = re.compile(r'biotop|masterprompt|lagerhaus|logmuehle|versandwerk|drahtwerk|prüfkanal|pruefkanal', re.I)
zahl = {}
for sid, rolle, inhalt, ts in con.execute('SELECT session_id, role, content, timestamp FROM messages WHERE timestamp >= ?', (AB,)):
    z = zahl.setdefault(sid, {'n': 0, 'user': 0, 'treffer': 0, 'user_treffer': 0, 'erste': ts, 'letzte': ts})
    z['n'] += 1
    z['erste'] = min(z['erste'], ts)
    z['letzte'] = max(z['letzte'], ts)
    t = inhalt or ''
    hit = bool(WORT.search(t))
    z['treffer'] += hit
    if rolle == 'user':
        z['user'] += 1
        z['user_treffer'] += hit
f = lambda ts: dt.datetime.fromtimestamp(ts).strftime('%m-%d %H:%M')
print(f'Sitzungen mit Nachrichten ab 18.09.: {len(zahl)} · davon in der Kette ab {START}: {len(set(zahl) & kette)}')
print('Sitzung | Quelle | Kette | Eltern | erste–letzte Nachricht | Nachrichten | davon user | Treffer Biotop-Wörter (alle/user) | Titel (nur Kette)')
for sid in sorted(zahl, key=lambda s: zahl[s]['erste']):
    s, z = sess.get(sid, {}), zahl[sid]
    ink = sid in kette
    print(f"{sid} | {s.get('source')} | {'ja' if ink else 'NEIN'} | {s.get('parent') or '-'} | {f(z['erste'])}–{f(z['letzte'])} | "
          f"{z['n']} | {z['user']} | {z['treffer']}/{z['user_treffer']} | {(s.get('title') or '')[:70] if ink else '(nicht ausgegeben)'}")
aus = [s for s in zahl if s not in kette and zahl[s]['user_treffer'] > 0]
print(f'\nSitzungen AUSSERHALB der Kette mit Biotop-Wörtern in user-Nachrichten: {len(aus)}')
for sid in aus:
    print(f"  {sid} | {sess.get(sid, {}).get('source')} | user-Treffer {zahl[sid]['user_treffer']} von {zahl[sid]['user']}")
