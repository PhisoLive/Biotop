def jetzt():

def zeit(s):
    """ISO-Zeit → datetime mit Zone. Zeiten ohne Zone gelten als UTC (so führt die Kasse sie)."""
    if s is None or s == '':
        return None
    if isinstance(s, (int, float)):
        return dt.datetime.fromtimestamp(float(s), TZ)
    s = str(s).strip()
    if re.match(r'\d{4}-\d\d-\d\d \d', s):
        s = s.replace(' ', 'T', 1)
    s = re.sub(r'([+-]\d\d)(\d\d)$', r'\1:\2', s)
    if s.endswith('Z'):
        s = s[:-1] + '+00:00'
    try:
        d = dt.datetime.fromisoformat(s)
    except ValueError:
        return None
    return d if d.tzinfo else d.replace(tzinfo=UTC)

def sha_datei(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    return h.hexdigest()


def zelle(x):

def laufnummern(t):
    s = {int(m.group(1)) for m in LAUF_EIN.finditer(t)}
    for m in LAUF_MEHR.finditer(t):
        a = int(m.group(1))
        s.add(a)
        if m.group(2):
            b = int(m.group(2))
            s.update(range(a, b + 1) if a < b <= a + 60 else [b])
        s.update(int(x) for x in re.findall(r'\d{1,3}', m.group(3) or ''))
    return s

def kern_text(t):
    """Nachrichtentext ohne vorangestellte Herkunftsdaten (JSON-Block) und ohne Antwortzitat."""
    t = t or ''
    m = re.match(r'\s*Gateway message origin[^\n]*\n', t)
    if m:
        rest, tiefe = t[m.end():], 0
        for i, ch in enumerate(rest):
            if ch == '{':
                tiefe += 1
            elif ch == '}':
                tiefe -= 1
                if tiefe == 0:
                    t = rest[i + 1:]
                    break
    return REPLY_RE.sub('', t.lstrip(), count=1)


def art_von(e, text):
    a = e.get('absender', '')
    if a.startswith('System'):
        return 'Meldung'
    if a.endswith('Zwischentext'):
        return 'Zwischentext'
    if a == 'Hermes→Subagent':
        return 'Auftrag'
    if a == 'Subagent':
        return 'Meldung'
    if a == 'CLI-Probe':
        return 'Anweisung'
    kern = kern_text(text)
    letzte = next((x.strip() for x in reversed(kern.strip().splitlines()) if x.strip()), '')
    frage = letzte.endswith('?')
    if a == 'Hermes':
        return 'Frage' if frage else 'Meldung'
    if a == 'Philipp':
        if DOK_RE.search(kern) or '[Content of ' in (text or '') or '[The user sent a' in (text or ''):
            return 'Auftrag'
        if len(kern.encode()) >= 1500:
            return 'Auftrag'
        return 'Frage' if frage else 'Anweisung'
    return 'UNBEKANNT'

        info['werke'] = con.execute('SELECT id, name, pfad, quelle, aufgenommen_am, sprache, status FROM werke '
                                    'ORDER BY id').fetchall()

        con.close()
        info['wal_nachher'] = wal_groesse(pfad)
        Z[name] = info

def weitere_kassen(vier_rows, vier_pfade):
    """Suche nach weiteren SQLite-Dateien *.db unter /root (Tiefe <= 5; ohne .hermes, sicherung, node_modules, .git,
    venv, .venv, .cache, .npm, __pycache__) mit Tabelle kasse; zaehlt Zeilen, die in keinem der vier Zweige stehen.
    Oeffnen: immutable=1, wenn -wal fehlt oder 0 Bytes hat; sonst mode=ro, im Baum /root/biotop gar nicht (nur gezaehlt)."""
    aus = {'mit_kasse': 0, 'mit_zeilen': 0, 'fremd': 0, 'fremd_eur': 0.0, 'liste': [], 'nicht': [], 'fehler': []}
    weg = {'.hermes', 'sicherung', 'node_modules', '.git', 'venv', '.venv', '.cache', '.npm', '__pycache__'}
    for wurzel, dirs, files in os.walk('/root'):
        tiefe = wurzel.count('/') - 1
        dirs[:] = sorted(d for d in dirs if d not in weg and tiefe < 5)
        for f in sorted(files):
            if not f.endswith('.db'):
                continue
            p = os.path.join(wurzel, f)
            if p in vier_pfade:
                continue
            w = p + '-wal'
            try:
                wal = os.path.getsize(w) if os.path.exists(w) else None
                if wal in (None, 0):
                    c = sqlite3.connect(f'file:{p}?mode=ro&immutable=1', uri=True)
                elif p.startswith('/root/biotop/'):
                    aus['nicht'].append((p, wal))
                    continue
                else:
                    c = sqlite3.connect(f'file:{p}?mode=ro', uri=True)
                try:
                    hat = c.execute("SELECT 1 FROM sqlite_master WHERE type='table' AND name='kasse'").fetchone()
                    rows = [tuple(x) for x in c.execute('SELECT * FROM kasse')] if hat else None
                finally:
                    c.close()
            except (sqlite3.Error, OSError) as e:
                aus['fehler'].append((p, str(e)[:80]))
                continue
            if rows is None:
                continue
            aus['mit_kasse'] += 1
            if not rows:
                continue
            aus['mit_zeilen'] += 1
            fremd = [x for x in rows if x not in vier_rows]
            eur_f = sum(x[7] for x in fremd if len(x) > 7 and isinstance(x[7], (int, float)))
            aus['fremd'] += len(fremd)
            aus['fremd_eur'] += eur_f
            zt = sorted(str(x[2]) for x in fremd if len(x) > 2)
            aus['liste'].append({'pfad': p, 'bytes': os.path.getsize(p), 'wal': wal, 'zeilen': len(rows),
                                 'fremd': len(fremd), 'fremd_eur': eur_f, 'von': zt[0] if zt else '-',
                                 'bis': zt[-1] if zt else '-'})
    return aus


def lade_ordner(L, eigener, commit_kurz):
    erg, andere = [], 0
    for o in sorted(L.iterdir()):
        if not o.is_dir():
            continue
        if not (o.name.startswith('biotop') or o.name == 'gedaechtnis'):
            andere += 1
            continue
        dateien, zeiten, laeufe, commits, fass = 0, [], set(), set(), Counter()
        einstieg, oben, texte_bytes = [], [], 0
        for wurzel, dirs, files in os.walk(o):
            for f in files:
                p = Path(wurzel) / f
                try:
                    st = p.lstat()
                except OSError:
                    continue
                dateien += 1
                zeiten.append(st.st_mtime)
                rel = p.relative_to(o).as_posix()
                laeufe |= laufnummern(rel)
                if p.parent == o:
                    oben.append(f)
                if p.suffix == '.md' and rel.count('/') <= 1 and any(
                        f.startswith(x) for x in ('BERICHT', 'START', 'ZWISCHENBERICHT', 'ABSCHLUSS', 'NACHGANG',
                                                  'UEBERGABE')):
                    einstieg.append(rel)
                if p.suffix in ('.md', '.txt') and st.st_size <= 1_000_000:
                    try:
                        t = p.read_text(errors='replace')
                    except OSError:
                        continue
                    texte_bytes += len(t)
                    laeufe |= laufnummern(t)
                    commits |= {h[:7] for h in HEX_RE.findall(t)} & commit_kurz
                    fass.update(int(x) for x in FASSUNG_RE.findall(t))
        zt = zeiten or [o.stat().st_mtime]
        erg.append({'name': o.name, 'dateien': dateien, 'aelteste': dt.datetime.fromtimestamp(min(zt), TZ),
                    'juengste': dt.datetime.fromtimestamp(max(zt), TZ), 'einstieg': sorted(einstieg),
                    'oben': len(oben), 'laeufe_genannt': laeufe, 'commits_genannt': commits, 'fassungen': fass,
                    'texte_bytes': texte_bytes, 'eigener': o.name == eigener, 'erfunden': False})
    return erg, andere

                                   if x.startswith('Stand')), '')})

    for d in dateien:
        for x in d.read_text(errors='replace').splitlines():
            m = re.search(r'\b(C\d{3})\b', x)
            if not m or not x.lstrip().startswith('|'):
                continue
            st = re.findall(r'(?i)\b(erledigt|teilweise|offen|zurückgezogen|ersetzt|abgebrochen|läuft|laeuft)\b', x)
            zt = None
            m2 = re.search(r'(2026)-(\d\d)-(\d\d)[ T](\d\d):(\d\d)', x) or None
            if m2:
                zt = dt.datetime(int(m2.group(1)), int(m2.group(2)), int(m2.group(3)), int(m2.group(4)),
                                 int(m2.group(5)), tzinfo=TZ)
            else:
                m3 = re.search(r'\b(\d{1,2})\.(\d{1,2})\.(?:2026)?,?\s+(\d{1,2}):(\d\d)', x)
                if m3:
                    zt = dt.datetime(2026, int(m3.group(2)), int(m3.group(1)), int(m3.group(3)), int(m3.group(4)),
                                     tzinfo=TZ)
            zeilen.append({'id': m.group(1), 'laeufe': laufnummern(x), 'status': st[-1].lower() if st else None,
                           'ordner': set(re.findall(r'\{LAEUFE\}/([\w.-]+)', x)) | set(re.findall(r'\b(biotop-[\w.-]*\w)', x)),
                           'zeit': zt, 'ids': {int(i) for i in re.findall(r'\bid\s*(\d{3,6})\b', x)}})
    return dateien, zeilen


def lade_r12_r14():
    sitz, verd = {}, []
    p = W / 'rohbelege' / 'R0012.txt'
    if p.exists():
        for x in p.read_text(errors='replace').splitlines():
            m = re.match(r'^(\d{8}_\d{6}_[0-9a-f]+) \| (\S+ \S+) – (\S+ \S+) \| (\d+) Nachrichten, davon user (\d+) \| '
                         r'(\{.*\})$', x)
            if m:
                try:
                    d = ast.literal_eval(m.group(6))
                except (ValueError, SyntaxError):
                    d = {}
                sitz[m.group(1)] = {'end_reason': d.get('end_reason'), 'source': d.get('source'),
                                    'ended_at': d.get('ended_at')}
    p = W / 'rohbelege' / 'R0014.txt'
    if p.exists():
        for x in p.read_text(errors='replace').splitlines():
            m = re.match(r'^\s+active (\d) compacted (\d) Anzahl (\d+)$', x)
            if m:
                verd.append(tuple(map(int, m.groups())))
    return sitz, verd


# ----------------------------------------------------------------------------------------------- Abgleich 1
def abgleich1(laeufe, ordner, chronik, nennung_ordner):
    """Lauf → Ordner und Ordner → Lauf/Auftrag. Gibt (offene Läufe, offene Ordner) als Mengen von Kennungen."""
    ords = [o for o in ordner if not o['eigener'] and o['name'] != 'gedaechtnis']
    zweig_dir = {n: p.parts[2] for n, _, p in ZWEIGE if n != 'STAMM'}
    for l in laeufe:
        kand = [o for o in ords if l['nr'] in o['laeufe_genannt']]
        zo = None
        if 'S' not in l['zweige'] and not l.get('erfunden'):
            d = zweig_dir.get(l['herkunft'])
            treffer = [o for o in ords if d and d in o['name']]
            if len(treffer) == 1:
                zo = treffer[0]
        if zo:
            l['ort'], l['ort_art'] = zo['name'], 'Zweig-Arbeitsordner' + (' + Nennung' if zo in kand else '')
        elif kand:
            b = l['beginn']
            drin = [o for o in kand if o['aelteste'] <= b <= o['juengste']]
            if drin:
                o = min(drin, key=lambda x: x['juengste'] - x['aelteste'])
                art = 'Nennung + Zeitfenster'
            else:
                spaeter = [o for o in kand if o['aelteste'] >= b]
                if spaeter:
                    o, art = min(spaeter, key=lambda x: x['aelteste']), 'Nennung, nächster Ordner danach'
                else:
                    o, art = max(kand, key=lambda x: x['aelteste']), 'Nennung, letzter Ordner davor'
            l['ort'], l['ort_art'] = o['name'], art
        else:
            cz = [c for c in (chronik or []) if l['nr'] in c['laeufe'] and c['ordner']]
            if cz and not l.get('erfunden'):
                l['ort'], l['ort_art'] = sorted(cz[0]['ordner'])[0], f'Chronik {cz[0]["id"]}'
            else:
                l['ort'], l['ort_art'] = None, 'offen'
        l['chronik'] = sorted({c['id'] for c in (chronik or []) if l['nr'] in c['laeufe']})
    offen_l = {l['schluessel'] for l in laeufe if l['ort'] is None}
    offen_o = set()
    for o in ordner:
        hier = [l['schluessel'] for l in laeufe if l.get('ort') == o['name']]
        n = nennung_ordner.get(o['name'], [])
        chron = sorted({c['id'] for c in (chronik or []) if o['name'] in c['ordner']})
        o['laeufe_hier'] = hier
        o['nennung'] = n[:1]
        o['chronik'] = chron
        if hier:
            o['zuordnung'] = f'Lauf ({len(hier)})'
        elif n or chron:
            o['zuordnung'] = 'Auftrag (Nennung)'
        else:
            o['zuordnung'] = 'offen'
            offen_o.add(o['name'])
    return offen_l, offen_o
