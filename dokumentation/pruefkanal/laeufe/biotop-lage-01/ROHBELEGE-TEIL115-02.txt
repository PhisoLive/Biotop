def sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def eigener_rohbeleg() -> str:
    pid = os.getppid()
    for _ in range(6):
        try:
            for fd in os.listdir(f'/proc/{pid}/fd'):
                try:
                    z = os.readlink(f'/proc/{pid}/fd/{fd}')
                except OSError:
                    continue
                m = re.search(r'/rohbelege/R(\d{4})\.txt$', z)
                if m:
                    return f'R{int(m.group(1))}'
            with open(f'/proc/{pid}/stat') as f:
                pid = int(f.read().rsplit(')', 1)[1].split()[1])
        except (OSError, ValueError, IndexError):
            break
    return 'UNBEKANNT (Nummer nicht ermittelt)'


def bloecke(text: str):
    z = text.split('\n')
    i = 0
    while i < len(z):
        if (z[i].startswith('Quelle: ') and i + 2 < len(z) and z[i + 1] == ''
                and z[i + 2].startswith(ZAUN) and len(z[i + 2]) > len(ZAUN)):
            quelle, b_s, s_s, e_s = z[i][len('Quelle: '):].rsplit(' · ', 3)
            j = i + 3
            while z[j] != ZAUN:
                j += 1
            schluss = e_s.strip().endswith('ja')
            inhalt = ('\n'.join(z[i + 3:j]) + ('\n' if schluss else '')).encode('utf-8')
            yield quelle, int(b_s.split()[0].replace('.', '')), s_s.split('`')[1], inhalt
            i = j + 1
        else:
            i += 1


def original(quelle: str) -> bytes:
    pfade = re.findall(r'`([^`]+)`', quelle)
    if quelle.startswith('Konstante '):
        name, pfad = pfade[0], pfade[1]
        baum = ast.parse(Path(pfad).read_text(encoding='utf-8'))
        for n in baum.body:
            if isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id == name for t in n.targets):
                wert = ast.literal_eval(n.value)
                return wert.encode('utf-8') if isinstance(wert, str) else \
                    json.dumps(wert, ensure_ascii=False, indent=1).encode('utf-8')
        raise KeyError(name)
    pfad = pfade[0].replace('{LAEUFE}', LAEUFE)
    roh = Path(pfad).read_bytes()
    m = re.search(r'Zeilen (\d+)–(\d+)', quelle)
    if m:
        zeilen = roh.decode('utf-8').split('\n')
        return ('\n'.join(zeilen[int(m.group(1)) - 1:int(m.group(2))]) + '\n').encode('utf-8')
    return roh
