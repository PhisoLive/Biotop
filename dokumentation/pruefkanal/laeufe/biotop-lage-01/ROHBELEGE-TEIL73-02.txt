#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil B (nur lesend): Verfassungen in den Masterprompt-Fassungen und in Berichten.

1. Je Masterprompt-Fassung (Dokumente aus der Sitzungsdatenbank, arbeit/dokumente/, ohne die zwei Bruchstücke vom
   19.09. 14:01): Abschnitt „## Anhang A“ und „## Anhang B“ — vorhanden ja/nein, Bytes, SHA-256 des Abschnitts
   (von der Überschrift bis vor die nächste „## “-Überschrift oder das Dokumentende, ohne Trennlinien am Ende),
   Satzanfänge. Gleichheit zwischen den Fassungen.
2. Kopfzeilen (Herkunft/Bestätigung) von Anhang A und B je Fassung im Wortlaut.
3. Vergleich Anhang (v8) ↔ Datei (verfassungen/*.md) nach Entfernen von Markdown-Auszeichnung (**, `), satzweise;
   jede Abweichung im Wortlaut (difflib, wortweise).
4. Prüfkanal: Abschnitt 9 des Berichts biotop-phase-3-nachlauf (Vorlage logmuehle an Chat-Claude): Verfassungsteil
   satzweise gegen die Datei.
Schreibt nichts. Führt keinen Biotop-Code aus.
"""

import hashlib

def h(s: str) -> str:
    return hashlib.sha256(s.encode()).hexdigest()


def abschnitt(text: str, kopf: str) -> str | None:
    m = re.search(rf'^## {re.escape(kopf)}.*$', text, re.M)
    if not m:
        return None
    rest = text[m.start():]
    n = re.search(r'^## ', rest[3:], re.M)
    teil = rest[: n.start() + 3] if n else rest
    teil = re.sub(r'(\n---\s*)+\s*$', '\n', teil.rstrip() + '\n')
    teil = re.sub(r'\n\*Ende des Dokuments.*$', '\n', teil, flags=re.S)
    return teil.rstrip('\n') + '\n'


def norm(s: str) -> str:
    s = s.replace('**', '').replace('`', '')
    return re.sub(r'\s+', ' ', s).strip()


def saetze(text: str) -> dict:
    out = {}
    for z in text.splitlines():
        m = re.match(r'^\s*(\d+)\.\s+(.*)$', z)
        if m:
            out[int(m.group(1))] = norm(m.group(2))
    return out


def wortdiff(a: str, b: str) -> str:
    aw, bw = a.split(' '), b.split(' ')
    teile = []
    for op, i1, i2, j1, j2 in difflib.SequenceMatcher(a=aw, b=bw, autojunk=False).get_opcodes():
        if op != 'equal':
            teile.append(f'{op}: [{" ".join(aw[i1:i2])}] → [{" ".join(bw[j1:j2])}]')
    return '; '.join(teile) if teile else 'gleich'

print('\n== Gleichheit der Anhänge zwischen Fassungen (SHA-256 des Abschnitts)')

    for f, _ in FASSUNGEN:
        a = abschn[(f, x)]

            gruppen.setdefault(h(a), []).append(f)
    print(f'Anhang {x}: ' + ' | '.join(f'{k[:16]}… = {v}' for k, v in gruppen.items()))
    # satzweise Gleichheit über Fassungen
    basis = None
    for f, _ in FASSUNGEN:
        a = abschn[(f, x)]
        if a is None:
            continue
        s = saetze(a)
        if basis is None:
            basis = (f, s)
            continue
        for nr in sorted(set(s) | set(basis[1])):
            if s.get(nr) != basis[1].get(nr):
                print(f'   Anhang {x} Satz {nr}: {basis[0]} ≠ {f}: {wortdiff(basis[1].get(nr, ""), s.get(nr, ""))}')
        basis = (f, s)

print('\n== 3. Anhang (v8) gegen Datei, satzweise nach Entfernen von ** und `')

    for nr in sorted(set(sd) | set(sa)):

    print(f'   Kopfzeile Datei: {datei.splitlines()[0]}')
