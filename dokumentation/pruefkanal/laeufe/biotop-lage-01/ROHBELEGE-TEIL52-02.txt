import re
from pathlib import Path

L = Path('/root/sicherung/biotop-lage-01/arbeit/lesen')


def bloecke(p: Path):
    t = p.read_text(errors='replace')
    teile = re.split(r'(?m)^(## 2026-\d\d-\d\d \d\d:\d\d:\d\d · [^\n]*)\n', t)
    # teile: [vorspann, kopf1, text1, kopf2, text2, ...]
    for i in range(1, len(teile) - 1, 2):
        yield teile[i][3:], teile[i + 1]


for name, kurz, lang in (('hermes-an-philipp.md', 110, 330), ('auftraege-an-subagenten.md', 90, 200)):
    print(f'===== {name}')
    n = 0
    for kopf, text in bloecke(L / name):

        flach = ' '.join(text.split())
        grenze = lang if kopf >= '2026-09-21' else kurz
        # Zusammenfassungsblöcke der Verdichtung (Historical Task Snapshot) nur kennzeichnen

            flach = '[Verdichtungs-Zusammenfassung, ausgelassen]'
        print(f'{kopf} | {flach[:grenze]}')
    print(f'-- {n} Blöcke')
