   sonst mode=ro; Dateien unter /root/biotop mit -wal > 0 werden nicht geöffnet.
STAMM nur immutable=1 bei 0 Bytes -wal. Schreibt nichts."""
import datetime as dt
import os
import sqlite3
from collections import Counter, defaultdict
from pathlib import Path

if os.path.exists('/tmp/biotop-kammer-exklusiv'):
    print('ABBRUCH: Kammer-Sperre besteht')
    raise SystemExit(3)
ZW = [('STAMM', '/root/biotop/daten/biotop.db'), ('VB1', '/root/phase-4-vorbedingung-1/privat/biotop.db'),
      ('R02', '/root/reparatur-02/privat-cli/privat.db'), ('T3', '/root/biotop-t3/t3.db')]


def oeffnen(p):
    w = p + '-wal'
    wal = os.path.getsize(w) if os.path.exists(w) else None
    if wal in (None, 0):
        return sqlite3.connect(f'file:{p}?mode=ro&immutable=1', uri=True), wal
    if p.startswith('/root/biotop/'):
        return None, wal
    return sqlite3.connect(f'file:{p}?mode=ro', uri=True), wal


zeilen, eich = {}, {}
for n, p in ZW:
    c, wal = oeffnen(p)
    if c is None:
        print(n, 'NICHT GEÖFFNET wal', wal)
        continue
    for r in c.execute('SELECT * FROM kasse'):
        zeilen.setdefault(tuple(r), n)
    for r in c.execute('SELECT * FROM eichkosten WHERE modellaufrufe > 0'):
        eich.setdefault(tuple(r), (n, p))
    c.close()
print('== 1. Kasse entdoppelt:', len(zeilen))
a = Counter()
s = defaultdict(float)
zw = Counter()
zs = defaultdict(float)
for r in zeilen:
    a[(r[3], r[9])] += 1
    s[(r[3], r[9])] += r[7] or 0
    k = (r[8] or '').split('.')[0]
    zw[k] += 1
    zs[k] += r[7] or 0
for k, v in sorted(a.items(), key=lambda x: -x[1]):
    print(f'   anbieter {k[0]!r} kontingent {k[1]!r}: {v} Buchungen, {s[k]:.6f} €')
for k, v in zw.most_common():
    print(f'   zweck {k!r}: {v}, {zs[k]:.6f} €')
print('\n== 2. eichkosten mit Modellaufrufen:', len(eich))
for r, (n, p) in sorted(eich.items(), key=lambda x: x[0][8]):

    c, _ = oeffnen(p)
    t = dt.datetime.fromisoformat(r[8])
    a0 = (t - dt.timedelta(minutes=15)).isoformat(timespec='milliseconds')
    a1 = (t + dt.timedelta(minutes=2)).isoformat(timespec='milliseconds')
    for b in c.execute('SELECT id, zeit, kosten_eur, zweck, modell FROM kasse WHERE lauf_id=? AND zeit BETWEEN ? AND ? '
                       'ORDER BY zeit', (r[1], a0, a1)):
        print('      ', b)
    c.close()
print('\n== 3. weitere *.db unter /root mit Tabelle kasse')
vier = {p for _, p in ZW}
for wurzel, dirs, files in os.walk('/root'):

    dirs[:] = [d for d in dirs if d not in ('.hermes', 'sicherung', 'node_modules', '.git', 'venv', '.venv', '.cache',
                                            '.npm', '__pycache__') and tiefe < 5]
    for f in files:

                continue
            tabs = {x[0] for x in c.execute("SELECT name FROM sqlite_master WHERE type='table'")}
            if 'kasse' not in tabs:
                c.close()
                continue
            rows = [tuple(x) for x in c.execute('SELECT * FROM kasse')]
            fremd = [x for x in rows if x not in zeilen]
            nl = c.execute('SELECT count(*), max(id) FROM laeufe').fetchone() if 'laeufe' in tabs else None

                  f'Zeit fremd {min((x[2] for x in fremd), default="-")} … {max((x[2] for x in fremd), default="-")}')
            c.close()
        except sqlite3.Error as e:
            print(f'   {p}: Fehler {e}')
