#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil G (nur lesend): Abgleichsbelege für WS03, WS07, WS11, WS16, WS18, FK23, FK25, FK44/FK45.
Alle Dateien werden intern geöffnet (Vorsorge nach SPERRE-01); kein Biotop-Code wird importiert oder ausgeführt.
(a) Nachrichten Philipps (berichtigtes Verzeichnis v2) mit „abgenommen|abgeschlossen|Abnahme“: id, Zeit, ±110 Zeichen
    je Treffer; Dokumente (Text beginnt mit „#“ und > 20.000 Zeichen) übersprungen; höchstens 3 Treffer je Nachricht.
(b) id 11505: ±320 Zeichen um „9,09“.
(c) Masterprompt v8: Zeilen 1–81 (Änderungsverzeichnis) mit „Grundsatz|Grundsätze“ (Zeilennummer, 260 Zeichen).
(d) Masterprompt v8: Zeilen 169–197 (Abschnitt 3.5), je 260 Zeichen.
(e) Masterprompt v8: Zeilen mit „überschrieben|Ende des Vorhabens|nach Phase 5|Phase 6“.
(f) ABNAHME Vorbedingung 1 (20.09.): Abschnitt „## 4“ bis zum nächsten „## “.
(g) {LAEUFE}/biotop-phase-2/BERICHT.md: Zeilen mit „Hersteller|Anbieter“ (höchstens 8, je 220 Zeichen).
(h) Kopf (erste 14 Zeilen) von /root/biotop/tests/drahtwerk.py und speicherwerk.py; Treffer „grosswerk“ in /root/biotop/tests.
"""
import json
import re
from pathlib import Path

W = Path('/root/sicherung/biotop-lage-01')
D = W / 'arbeit/dokumente'
V8 = D / '2026-09-23_155353_INFINITE-EVOLUTION-BIOTOP-Masterprompt-v8.md'
K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')


def eng(s):
    return s.replace('\n', ' ')


nachr = [json.loads(z) for z in (W / 'arbeit/verlauf/v2/wortlaut.jsonl').open(encoding='utf-8')]
ph = [d for d in nachr if d.get('absender') == 'Philipp']
print('== (a) Philipp: abgenommen/abgeschlossen/Abnahme')
for d in ph:
    t = d.get('text') or ''
    if t.lstrip().startswith('#') and len(t) > 20000:
        continue
    n = 0
    for m in re.finditer(r'abgenommen|abgeschlossen|Abnahme', t):
        a = max(0, m.start() - 110)
        print(f"id {d['id']} · {d['zeit']}: …{eng(t[a:m.end() + 110])}…")
        n += 1
        if n >= 3:
            break

print('\n== (b) id 11505 um „9,09“')
for d in ph:
    if d['id'] == 11505:
        t = d['text']
        for m in re.finditer(r'9,09', t):
            a = max(0, m.start() - 320)
            print(f"…{eng(t[a:m.end() + 320])}…")

zeilen = V8.read_text(encoding='utf-8').splitlines()
print('\n== (c) v8 Zeilen 1–81 mit Grundsatz/Grundsätze')
for i, z in enumerate(zeilen[:81], 1):
    if re.search('Grundsatz|Grundsätze', z):
        print(f'{i}: {z[:260]}')
print('\n== (d) v8 Zeilen 169–197')
for i in range(169, 198):
    print(f'{i}: {zeilen[i - 1][:260]}')
print('\n== (e) v8 überschrieben / Ende des Vorhabens / nach Phase 5 / Phase 6')
for i, z in enumerate(zeilen, 1):
    if re.search('überschrieben|Ende des Vorhabens|nach Phase 5|Phase 6', z):
        print(f'{i}: {z[:260]}')

print('\n== (f) ABNAHME Vorbedingung 1, Abschnitt 4')
ab = (D / '2026-09-20_045016_ABNAHME_Vorbedingung-1_2026-09-20.md').read_text(encoding='utf-8').splitlines()
drin = False
for i, z in enumerate(ab, 1):
    if z.startswith('## '):
        drin = z.startswith('## 4')
    if drin:
        print(f'{i}: {z[:300]}')

print('\n== (g) biotop-phase-2/BERICHT.md: Hersteller/Anbieter')
n = 0
for i, z in enumerate((L / 'biotop-phase-2/BERICHT.md').read_text(errors='replace').splitlines(), 1):
    if re.search('Hersteller|Anbieter', z) and n < 8:
        print(f'{i}: {z[:220]}')
        n += 1

print('\n== (h) Köpfe der Prüfwerk-Erzeuger')
for f in ('drahtwerk.py', 'speicherwerk.py'):
    print(f'-- /root/biotop/tests/{f}')
    for i, z in enumerate(Path('/root/biotop/tests', f).read_text(encoding='utf-8').splitlines()[:14], 1):
        print(f'{i}: {z[:200]}')
print('-- „grosswerk“ in /root/biotop/tests (*.py)')
for p in sorted(Path('/root/biotop/tests').glob('*.py')):
    for i, z in enumerate(p.read_text(encoding='utf-8', errors='replace').splitlines(), 1):
        if 'grosswerk' in z:
            print(f'{p.name}:{i}: {z.strip()[:160]}')
print('=== Ende')
