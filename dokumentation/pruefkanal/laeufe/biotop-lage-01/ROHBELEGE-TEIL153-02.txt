parse = lambda s: dt.datetime.strptime(s[:19], '%Y-%m-%d %H:%M:%S').timestamp()

# Rohbelege: Zeitfenster
fenster = []
for p in sorted((W / 'rohbelege').glob('R*.txt')):
    kopf = p.read_text(encoding='utf-8', errors='replace')[:3000]
    b = re.search(r'^Beginn: (\S+ \S+)', kopf, re.M)
    e = re.search(r'^Ende:\s+(\S+ \S+)', kopf, re.M)
    t = kopf.split('\n', 1)[0]
    if b and e:
        fenster.append((parse(b.group(1)), parse(e.group(1)), t[:110]))
con = sqlite3.connect('file:/root/.hermes/state.db?mode=ro', uri=True)
quelle = {r[0]: r[1] for r in con.execute('SELECT id, source FROM sessions')}


def zuordnung(ts):
    rb = [t for b, e, t in fenster if b - 1 <= ts <= e + 2]
    aufrufe = []
    for sid, t, tc in con.execute("SELECT session_id, timestamp, tool_calls FROM messages WHERE role='assistant' AND "
                                  "tool_calls IS NOT NULL AND timestamp BETWEEN ? AND ? ORDER BY timestamp", (ts - 10, ts + 1)):
        namen = re.findall(r'"name":\s*"([a-z_]+)"', tc or '')
        aufrufe.append(f'{fmt(t)} {sid} ({quelle.get(sid, "?")}) {",".join(sorted(set(namen)))}')
    return rb, aufrufe


zeilen = [z.rstrip('\n').split('|', 4) for z in open(liste, encoding='utf-8', errors='replace')]
treffer = [z for z in zeilen if (any(z[4].startswith(g) for g in GRUPPEN) or z[4] in EINZELN
