#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Fakten für Chronik, Entscheidungen, Tore.
1. Je Masterprompt-Fassung v1..v8 (aus arbeit/dokumente/): „Stand:"-Zeile, eigenes Änderungsverzeichnis (erster Block
   „Änderungen gegenüber Fassung n-1" bis zum nächsten Block) → arbeit/fassungen/aenderungen-vN.md; Aufzählungspunkte
   je Tor-Abschnitt.
2. Kopf (erste 6 Zeilen) jeder Einstiegsdatei der Biotop-Ordner im Prüfkanal (BERICHT.md, ABSCHLUSS-03.md …).
3. Je Lauf-Kennung 1..38: in welchen Ordnern (oberste Ebene, *.md) „Lauf <n>" vorkommt.
4. Docker-Abbilder; Inhalt von /root/w7 (Namen)."""
import re
import subprocess
from pathlib import Path

D = Path('/root/sicherung/biotop-lage-01/arbeit/dokumente')
F = Path('/root/sicherung/biotop-lage-01/arbeit/fassungen')
F.mkdir(exist_ok=True)
K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
print('== 1. Fassungen')
for p in sorted(D.glob('*INFINITE-EVOLUTION-BIOTOP-Masterprompt-v*.md')):
    t = p.read_text()
    if len(t) < 10000:
        continue  # Teilstücke aus Verdichtungen, keine Fassung
    v = re.search(r'Masterprompt-(v\d)', p.name).group(1)
    stand = next((z for z in t.splitlines()[:12] if z.startswith('Stand')), '(keine Stand-Zeile)')
    kopf = t.split('\n## 0.')[0]
    bloecke = [m.start() for m in re.finditer(r'\*\*Änderungen gegenüber Fassung', kopf)]
    eigen = kopf[bloecke[0]:bloecke[1]] if len(bloecke) > 1 else (kopf[bloecke[0]:] if bloecke else '')
    (F / f'aenderungen-{v}.md').write_text(eigen.rstrip() + '\n')
    tore = {}
    for m in re.finditer(r'\*\*Tor (\d)[^\n]*\n((?:- .*\n)+)', t):
        tore.setdefault(m.group(1), len(re.findall(r'^- ', m.group(2), re.M)))
    print(f'{v}: {p.name} | {stand} | Kopf {len(kopf)} Zeichen, Blöcke {len(bloecke)}, eigenes Verzeichnis '
          f'{len(eigen)} Zeichen | Tor-Aufzählungspunkte {tore}')
print('\n== 2. Köpfe der Einstiegsdateien')
for o in sorted(L.iterdir()):
    if not o.name.startswith('biotop') or o.name == 'biotop-lage-01':
        continue
    for name in ('BERICHT.md', 'ZWISCHENBERICHT.md', 'abschluss-03/ABSCHLUSS-03.md', 'NACHGANG-06.md',
                 'nach-reparatur-02/ZWISCHENBERICHT.md'):
        p = o / name
        if p.exists():
            z = [x for x in p.read_text(errors='replace').splitlines() if x.strip()][:5]
            print(f'-- {o.name}/{name}')
            for x in z:
                print('   ', x[:260])
print('\n== 3. Läufe → Ordner (Nennung „Lauf <n>" in *.md der obersten und zweiten Ebene)')
texte = {}
for o in sorted(L.iterdir()):
    if o.is_dir() and o.name.startswith('biotop') and o.name != 'biotop-lage-01':
        texte[o.name] = ' '.join(p.read_text(errors='replace') for p in list(o.glob('*.md')) + list(o.glob('*/*.md')))
for n in range(1, 39):
    treffer = [k for k, t in texte.items() if re.search(rf'\bL(?:auf|äufe)\s+(?:\d+[–-])?{n}\b', t)]
    print(f'Lauf {n}: {treffer}')
print('\n== 4. Docker-Abbilder; /root/w7')
print(subprocess.run(['docker', 'images', '--format', '{{.Repository}}:{{.Tag}} {{.ID}} {{.Size}} {{.CreatedAt}}'],
                     capture_output=True, text=True).stdout)
print(sorted(p.name for p in Path('/root/w7').iterdir()))
