#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil C (nur lesend): Schema der Datenbankzweige für FK28/FK29.
Regeln: STAMM (/root/biotop/daten/biotop.db) nur mit ?mode=ro&immutable=1 und nur, wenn die -wal-Datei 0 Bytes hat
(sonst NICHT geöffnet, Vermerk). Alle anderen Zweige mit ?mode=ro.
Ausgabe je Zweig: Größe von DB, -wal, -shm; Tabellen mit Zeilenzahl; Spalten der Tabellen, deren Name 'gedaechtnis',
'immun', 'protokoll', 'muster', 'art', 'projekt' oder 'behauptung' enthält."""
import os
import sqlite3

ZWEIGE = [
    ('STAMM', '/root/biotop/daten/biotop.db', 'immutable'),
    ('VB1', '/root/phase-4-vorbedingung-1/privat/biotop.db', 'ro'),
    ('R02', '/root/reparatur-02/privat-cli/privat.db', 'ro'),
    ('T3', '/root/biotop-t3/t3.db', 'ro'),
]


def groesse(p):
    return os.path.getsize(p) if os.path.exists(p) else None


for name, pfad, art in ZWEIGE:
    g, w, s = groesse(pfad), groesse(pfad + '-wal'), groesse(pfad + '-shm')
    print(f'== {name}: {pfad} | db {g} B | wal {w} | shm {s}')
    if g is None:
        print('   NICHT GEFUNDEN')
        continue
    if art == 'immutable':
        if w not in (None, 0):
            print(f'   NICHT GEÖFFNET: -wal hat {w} Bytes (Regel: nur bei 0 Bytes)')
            continue
        uri = f'file:{pfad}?mode=ro&immutable=1'
    else:
        uri = f'file:{pfad}?mode=ro'
    print(f'   URI: {uri}')
    try:
        c = sqlite3.connect(uri, uri=True)
        tabellen = [r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
        for t in tabellen:
            n = c.execute(f'SELECT count(*) FROM "{t}"').fetchone()[0]
            spalten = [r[1] for r in c.execute(f'PRAGMA table_info("{t}")')]
            merk = any(k in t for k in ('gedaechtnis', 'immun', 'protokoll', 'muster', 'art', 'projekt', 'behauptung'))
            print(f'   {t}: {n} Zeilen' + (f' | Spalten {spalten}' if merk else ''))
        c.close()
    except Exception as e:  # laut scheitern
        print(f'   FEHLER beim Lesen: {type(e).__name__}: {e}')
    print(f'   danach: wal {groesse(pfad + "-wal")} | shm {groesse(pfad + "-shm")}')

#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil A (nur lesend) — FK19: Wie wurde tatsächlich freigegeben?
Je Datenbankzweig (STAMM nur mit immutable=1 und nur bei leerer WAL-Datei; übrige mode=ro):
- generationen mit freigabe_am/freigabe_grund, gruppiert nach Werk (über laeufe.werk_id → werke.name) und Freigabeart
  (Präfix des Grundes bis zum ersten Doppelpunkt), mit Anzahl, frühester und spätester Freigabe;
- Protokollzeilen, deren Art mit 'freigabe' oder 'mentor' beginnt, je Art gezählt.
Kein Schreiben, kein Biotop-Import."""
import os
import sqlite3
from pathlib import Path

if Path('/tmp/biotop-kammer-exklusiv').exists():
    raise SystemExit('ABBRUCH: /tmp/biotop-kammer-exklusiv besteht — keine Datenbankabfragen während Zeitmessungen.')
ZWEIGE = [
    ('STAMM', '/root/biotop/daten/biotop.db', True),
    ('VB1', '/root/phase-4-vorbedingung-1/privat/biotop.db', False),
    ('R02', '/root/reparatur-02/privat-cli/privat.db', False),
    ('T3', '/root/biotop-t3/t3.db', False),
]
for name, pfad, stamm in ZWEIGE:
    print(f'\n===== {name}: {pfad}')
    if not Path(pfad).exists():
        print('NICHT GEFUNDEN')
        continue
    wal = Path(pfad + '-wal')
    print(f'WAL-Datei: {"vorhanden, " + str(wal.stat().st_size) + " Bytes" if wal.exists() else "nicht vorhanden"}')
    if stamm:
        if wal.exists() and wal.stat().st_size > 0:
            print('NICHT GEÖFFNET: WAL-Datei nicht leer (Regel 2, REGELN-FUER-TEILE).')
            continue
        uri = f'file:{pfad}?mode=ro&immutable=1'
    else:
        uri = f'file:{pfad}?mode=ro'
    c = sqlite3.connect(uri, uri=True)
    tabellen = [r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
    lsp = [r[1] for r in c.execute('PRAGMA table_info(laeufe)')]
    print('laeufe-Spalten:', lsp)
    werkspalte = 'werk_id' if 'werk_id' in lsp else None
    q = f"""SELECT COALESCE(w.name, '?') AS werk,
                   CASE WHEN g.freigabe_grund IS NULL THEN '(kein Grund)'
                        ELSE substr(g.freigabe_grund, 1, CASE WHEN instr(g.freigabe_grund, ':') > 0
                                                              THEN instr(g.freigabe_grund, ':') - 1 ELSE 80 END) END AS art,
                   count(*), min(g.freigabe_am), max(g.freigabe_am), min(g.lauf_id), max(g.lauf_id)
            FROM generationen g LEFT JOIN laeufe l ON l.id = g.lauf_id
            {'LEFT JOIN werke w ON w.id = l.' + werkspalte if werkspalte else 'LEFT JOIN werke w ON 0'}
            WHERE g.freigabe_am IS NOT NULL OR g.freigabe_grund IS NOT NULL
            GROUP BY werk, art ORDER BY min(g.freigabe_am)"""
    gesamt = c.execute('SELECT count(*), sum(freigabe_am IS NOT NULL), sum(freigabe_grund IS NOT NULL) FROM generationen').fetchone()
    print(f'generationen gesamt {gesamt[0]}, mit freigabe_am {gesamt[1]}, mit freigabe_grund {gesamt[2]}')
    print('Werk | Freigabeart (Grund bis zum ersten Doppelpunkt) | Anzahl | erste | letzte | Lauf von | Lauf bis')
    for r in c.execute(q):
        print('  ', ' | '.join(str(x) for x in r))
    for t in tabellen:
        sp = [r[1] for r in c.execute(f'PRAGMA table_info({t})')]
        if 'protokoll' in t and 'art' in sp:
            print(f'-- Protokoll ({t}), Arten freigabe*/mentor*:')
            for r in c.execute(f"SELECT art, count(*) FROM {t} WHERE art LIKE 'freigabe%' OR art LIKE 'mentor%' GROUP BY art ORDER BY art"):
                print('  ', r)
    c.close()
print('\nstat danach (Änderungszeit, Größe) STAMM und WAL/SHM:')
