def sha_datei(p) -> str:
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for block in iter(lambda: f.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def exklusiv_schreiben(p: Path, text: str) -> None:
    p.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, 'w', encoding='utf-8', errors='surrogateescape') as f:
        f.write(text)

def liste(name: str) -> None:
    wurzel = '/root/biotop'
    zeilen, n_dat, n_ver, n_son, byt = [], 0, 0, 0, 0
    for d, ds, fs in os.walk(wurzel):
        if d == wurzel:
            ds[:] = [x for x in ds if x != '.git']
        ds.sort()
        for x in list(ds):
            p = os.path.join(d, x)
            if os.path.islink(p):
                fs.append(x)
                ds.remove(x)
        for f in sorted(fs):
            p = os.path.join(d, f)
            rel = os.path.relpath(p, wurzel)
            st = os.lstat(p)
            if stat.S_ISLNK(st.st_mode):
                zeilen.append(f'VERWEIS -> {os.readlink(p)}  {rel}')
                n_ver += 1
            elif stat.S_ISREG(st.st_mode):
                zeilen.append(f'{sha_datei(p)}  {rel}')
                n_dat += 1
                byt += st.st_size
            else:
                zeilen.append(f'SONDERDATEI  {rel}')
                n_son += 1
    zeilen.sort(key=lambda z: z.split('  ', 1)[1])
    text = '\n'.join(zeilen) + '\n'
    ziel = W / 'pruefsummen' / f'{name}.txt'
    exklusiv_schreiben(ziel, text)
    print(f'Prüfsummenliste {name}: /root/biotop ohne /root/biotop/.git/ (verschachtelte .git-Ordner der Werke zählen mit)')
    print(f'Zeitpunkt: {jetzt()}')
    print(f'Dateien: {n_dat} · Verweise: {n_ver} · Sonderdateien: {n_son} · Einträge gesamt: {len(zeilen)} · Bytes der Dateien: {byt}')
    print(f'Liste: {ziel} · {len(text.encode())} Bytes · SHA-256 der Liste {hashlib.sha256(text.encode()).hexdigest()}')
    print('--- Liste (vollständig; Format: SHA-256, zwei Leerzeichen, Pfad relativ zu /root/biotop) ---')
    sys.stdout.write(text)


def vergleich(a: str, b: str) -> None:
    def lade(n):
        t = (W / 'pruefsummen' / f'{n}.txt').read_text(errors='surrogateescape')
        return {z.split('  ', 1)[1]: z.split('  ', 1)[0] for z in t.splitlines() if z}, hashlib.sha256(t.encode(errors='surrogateescape')).hexdigest()
    ma, ha = lade(a)
    mb, hb = lade(b)
    nur_a = sorted(set(ma) - set(mb))
    nur_b = sorted(set(mb) - set(ma))
    anders = sorted(k for k in set(ma) & set(mb) if ma[k] != mb[k])
    print(f'{a}: {len(ma)} Einträge, SHA-256 der Liste {ha}')
    print(f'{b}: {len(mb)} Einträge, SHA-256 der Liste {hb}')
    print(f'nur in {a}: {len(nur_a)} · nur in {b}: {len(nur_b)} · Inhalt verschieden: {len(anders)}')
    for k in nur_a:
        print(f'  NUR {a}: {k}')
    for k in nur_b:
        print(f'  NUR {b}: {k}')
    for k in anders:
        print(f'  VERSCHIEDEN: {k}  {ma[k][:16]}… → {mb[k][:16]}…')
    print('ERGEBNIS: Unterschied leer' if not (nur_a or nur_b or anders) else 'ERGEBNIS: Unterschied NICHT leer')

def laufe(wurzel: Path, teil: str):
    """Liefert (art, pfad) für alles im Umfang: 'ordner', 'datei', 'verweis', 'sonder', 'aus-ordner', 'aus-geheim', 'aus-hermes'."""
    w = str(wurzel)
    if wurzel.is_file() or wurzel.is_symlink():
        yield ('verweis' if wurzel.is_symlink() else 'datei'), w
        return
    for d, ds, fs in os.walk(w):
        yield 'ordner', d
        behalten = []
        for x in sorted(ds):
            p = os.path.join(d, x)
            if os.path.islink(p):
                yield 'verweis', p
            elif x in AUSGELASSENE_ORDNER:
                yield 'aus-ordner', p
            else:
                behalten.append(x)
        ds[:] = behalten
        for f in sorted(fs):
            p = os.path.join(d, f)
            if teil == '5-hermes-rest' and p in HERMES_AUSSER:
                yield 'aus-hermes', p
                continue
            if geheim(f) or p in GEHEIM_PFADE:
                yield 'aus-geheim', p
                continue
            try:
                st = os.lstat(p)
            except FileNotFoundError:
                yield 'verschwunden', p
                continue
            if stat.S_ISLNK(st.st_mode):
                yield 'verweis', p
            elif stat.S_ISREG(st.st_mode):
                yield 'datei', p
            else:
                yield 'sonder', p


def ist_sqlite(p: str) -> bool:
    try:
        if os.path.getsize(p) < 100:
            return False
        with open(p, 'rb') as h:
            return h.read(16) == b'SQLite format 3\x00'
    except OSError:
        return False

def sichere_db(src: str, dst: str) -> str:
    """Kopie per Connection.backup. Rückgabe: benutzter Weg."""
    wal = src + '-wal'
    if src.startswith('/root/.hermes/'):
        weg = 'mode=ro am Original (Hermes schreibt laufend; backup liest einen Schnappschuss)'
        s = sqlite3.connect(f'file:{src}?mode=ro', uri=True)
    elif os.path.exists(wal) and os.path.getsize(wal) > 0:
        # Kein Schreiber; -wal nicht leer: Datei und -wal in ein eigenes Verzeichnis kopieren und die Kopie lesen,
        # damit am Original (auch an -shm) nichts geschieht.
        td = tempfile.mkdtemp(dir=TMP, prefix='db-')
        k = os.path.join(td, 'k.db')
        shutil.copy2(src, k)
        shutil.copy2(wal, k + '-wal')
        weg = 'Kopie von Datei und -wal, Backup aus der Kopie (Original unberührt)'
        s = sqlite3.connect(k)
    else:
        weg = 'mode=ro&immutable=1 (keine -wal-Daten; Original und -shm unberührt)'
        s = sqlite3.connect(f'file:{src}?mode=ro&immutable=1', uri=True)
    d = sqlite3.connect(dst)
    with d:
        s.backup(d)
    d.close()
    s.close()
    return weg

def integritaet() -> None:
    dbs = json.loads((S / 'dbliste.json').read_text())
    ok, schlecht, fehlt = 0, [], []
    wichtig = []
    for p in sorted(dbs):
        k = STUFE / p.lstrip('/')
        if not k.exists():
            fehlt.append(p)
            continue
        c = sqlite3.connect(f'file:{k}?mode=ro&immutable=1', uri=True)
        r = [x[0] for x in c.execute('PRAGMA integrity_check').fetchall()]
        tab = c.execute("SELECT count(*) FROM sqlite_master WHERE type='table'").fetchone()[0]
        c.close()
        if r == ['ok']:
            ok += 1
        else:
            schlecht.append((p, r[:5]))
        if ('/pytest' not in p and '/tmp/' not in p and '/eltern-temp/' not in p) or dbs[p]['bytes'] > 20_000_000:
            wichtig.append(f'  {r[0] if len(r) == 1 else r[:3]} · {tab} Tabellen · Kopie {k.stat().st_size} Bytes · sha256 {sha_datei(k)[:16]}… · {p}')
    print(f'{jetzt()} · PRAGMA integrity_check über {len(dbs)} Datenbankkopien in {STUFE}')
    print(f'Ergebnis „ok“: {ok} von {len(dbs)} · nicht ok: {len(schlecht)} · Kopie fehlt: {len(fehlt)}')
    for p, r in schlecht:
        print(f'  NICHT OK: {p} {r}')
    for p in fehlt:
        print(f'  FEHLT: {p}')
    print(f'Einzeln (alle Datenbanken außer Testzwischenständen unter pytest/tmp, und alle > 20 MB): {len(wichtig)}')
    print('\n'.join(wichtig))


# ---------------------------------------------------------------- Manifest, Packen, Probe, Gegenprobe
def manifest_zeilen(root: Path) -> list:
    r = str(root)
    zeilen = []
    for d, ds, fs in os.walk(r):
        rel_d = os.path.relpath(d, r)
        zeilen.append(f'D\t\t\t{rel_d}')
        for x in list(ds):
            if os.path.islink(os.path.join(d, x)):
                fs.append(x)
                ds.remove(x)
        ds.sort()
        for f in fs:
            p = os.path.join(d, f)
            rel = os.path.relpath(p, r)
            st = os.lstat(p)
            if stat.S_ISLNK(st.st_mode):
                zeilen.append(f'L\t{os.readlink(p)}\t\t{rel}')
            elif stat.S_ISREG(st.st_mode):
                zeilen.append(f'F\t{sha_datei(p)}\t{st.st_size}\t{rel}')
            else:
                zeilen.append(f'S\t\t\t{rel}')
    zeilen.sort(key=lambda z: z.split('\t')[3])
    return zeilen


def manifest(baum: str, ziel: str) -> dict:
    t0 = time.time()
    z = manifest_zeilen(Path(baum))
    text = '\n'.join(z) + '\n'
    exklusiv_schreiben(Path(ziel), text)
    arten = {a: sum(1 for x in z if x.startswith(a + '\t')) for a in 'DFLS'}
    byt = sum(int(x.split('\t')[2]) for x in z if x.startswith('F\t'))
    info = {'baum': baum, 'ziel': ziel, 'eintraege': len(z), 'verzeichnisse': arten['D'], 'dateien': arten['F'],
            'verweise': arten['L'], 'sonder': arten['S'], 'bytes': byt,
            'sha256_manifest': hashlib.sha256(text.encode(errors='surrogateescape')).hexdigest(), 'dauer_s': round(time.time() - t0, 1)}
    print(json.dumps(info, ensure_ascii=False))
    return info


def lade_manifest(p: Path) -> dict:
    m = {}
    for z in p.read_text(errors='surrogateescape').splitlines():
        a, x, y, rel = z.split('\t')
        m[rel] = (a, x, y)
    return m


def vergleiche_manifeste(a: Path, b: Path, name_a: str, name_b: str) -> list:
    ma, mb = lade_manifest(a), lade_manifest(b)
    nur_a = sorted(set(ma) - set(mb))
    nur_b = sorted(set(mb) - set(ma))
