die Trefferzeile mit 3 Folgezeilen. Prüfkanal-Schlüssel maskiert."""
import re
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
BERICHTE = [
    L / 'biotop-phase-1/BERICHT.md', L / 'biotop-phase-1-nachlauf/BERICHT.md', L / 'biotop-phase-2/BERICHT.md',

    L / 'biotop-vor-t4-04/BERICHT.md', L / 'biotop-vor-t4-05/BERICHT.md',
    Path('/root/biotop-vor-t4-06/VERDICHTUNG-OPENROUTER.md'),
]
m = lambda p: str(p).replace(str(L), '{LAEUFE}')
for p in BERICHTE:
    z = p.read_text(errors='replace').splitlines()
    for i, s in enumerate(z):
        mm = re.match(r'^(#+) .*(Eigene Entscheidungen|Eigene Entscheidung)', s)
        if not mm:
            continue
        ebene = len(mm.group(1))
        j = i + 1
        punkte, akt = [], None
        while j < len(z) and not re.match(rf'^#{{1,{ebene}}} ', z[j]):
            t = z[j]
            if re.match(r'^\s*([-*]|\d+\.)\s', t) and not t.startswith('   '):
                if akt:
                    punkte.append(akt)
                akt = t.strip()
            elif t.strip() and akt is not None:
                akt += ' ' + t.strip()
            elif t.strip() and akt is None:
                punkte.append('(Vortext) ' + t.strip())
            j += 1
        if akt:
            punkte.append(akt)
        print(f'\n##### {m(p)}:{i + 1}: {s.strip()} — {len([x for x in punkte if not x.startswith("(Vortext)")])} Punkte')
        for x in punkte:
            print('  •', x[:330] + ('…' if len(x) > 330 else ''))
print('\n##### /root/biotop-vor-t4-06/UEBERGABE.md — Zeilen mit eigener Entscheidung')
z = Path('/root/biotop-vor-t4-06/UEBERGABE.md').read_text(errors='replace').splitlines()
for i, s in enumerate(z):
    if re.search(r'[Ee]igene Entscheidung|Entscheidung Hermes', s):
        print(f'  {i + 1}: ' + ' ⏎ '.join(x.strip() for x in z[i:i + 3])[:600])
for n in ('BAU-S1B-M2B.md', 'BAU-S4-VERDRAHTUNG.md', 'S1-ABBRUCH.md', 'sperre/SPERRE-02-S3-gateway-waechter.md',
          'sperre/SPERRE-04-S1b-formatsperre.md'):
    p = Path('/root/biotop-vor-t4-06') / n
    z = p.read_text(errors='replace').splitlines()
    for i, s in enumerate(z):
        if re.search(r'[Ee]igene Entscheidung|Entscheidung Hermes', s):
            print(f'\n##### {p}:{i + 1}: ' + ' ⏎ '.join(x.strip() for x in z[i:i + 4])[:700])
