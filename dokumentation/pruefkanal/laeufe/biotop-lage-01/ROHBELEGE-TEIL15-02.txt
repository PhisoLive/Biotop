#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Alle Dokumente, die Philipp seit 14.09. als Datei geschickt hat, aus state.db.
Je Dokument: Zeitpunkt (erste Zeile), Sitzung, Name, ob der Inhalt eingefügt ist, Länge und SHA-256 des Inhalts.
Doppelte Zeilen (dieselbe Nachricht in mehreren Generationen) werden zusammengefasst (gezählt).
Der Inhalt jedes Dokuments wird nach /root/sicherung/biotop-lage-01/arbeit/dokumente/<zeit>_<name> exklusiv
gespeichert (für Wortlaut-Abgleich und Sicherung; der Ordner liegt außerhalb des Prüfkanals)."""
import datetime as dt
import hashlib
import os
import re
import sqlite3
from pathlib import Path

ZIEL = Path('/root/sicherung/biotop-lage-01/arbeit/dokumente')
ZIEL.mkdir(parents=True, exist_ok=True)
con = sqlite3.connect('file:/root/.hermes/state.db?mode=ro', uri=True)
q = """SELECT id, session_id, timestamp, content FROM messages
       WHERE role='user' AND content LIKE '%[The user sent a%document:%' ORDER BY timestamp, id"""
kopf = re.compile(r"\[The user sent a (?:text )?document: '([^']+)'\. (Its content has been included below|It is saved at)")
inhalt_re = re.compile(r"\[Content of ([^\]]+)\]:\n", re.S)
gesehen = {}
for mid, sid, ts, c in con.execute(q):
    for m in kopf.finditer(c):
        name = m.group(1)
        eingefuegt = m.group(2).startswith('Its content')
        text = ''
        mi = inhalt_re.search(c, m.end())
        if eingefuegt and mi:
            # Inhalt bis zum nächsten Dokumentkopf oder bis zum Ende der Nachricht
            ende = kopf.search(c, mi.end())
            text = c[mi.end(): ende.start() if ende else len(c)]
            text = text.rstrip('\n')
        h = hashlib.sha256(text.encode('utf-8')).hexdigest() if text else '-'
        schluessel = (name, h)
        z = dt.datetime.fromtimestamp(ts).strftime('%Y-%m-%d %H:%M:%S')
        if schluessel in gesehen:
            gesehen[schluessel]['zeilen'] += 1
            continue
        gesehen[schluessel] = {'zeit': z, 'sitzung': sid, 'id': mid, 'name': name, 'eingefuegt': eingefuegt,
                               'laenge': len(text.encode('utf-8')), 'sha': h, 'zeilen': 1}
        if text:
            p = ZIEL / f"{z.replace(' ', '_').replace(':', '')}_{name}"
            if not p.exists():
                fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
                with os.fdopen(fd, 'w', encoding='utf-8') as f:
                    f.write(text)
print(f'Dokumente (verschieden nach Name und Inhalt): {len(gesehen)}')
print('Zeit (CEST) | Sitzung | Nachricht-id | Name | Inhalt eingefügt | Bytes | SHA-256 | Zeilen in state.db')
for d in sorted(gesehen.values(), key=lambda x: x['zeit']):
    print(f"{d['zeit']} | {d['sitzung']} | {d['id']} | {d['name']} | {'ja' if d['eingefuegt'] else 'nein'} | "
          f"{d['laenge']} | {d['sha']} | {d['zeilen']}")
