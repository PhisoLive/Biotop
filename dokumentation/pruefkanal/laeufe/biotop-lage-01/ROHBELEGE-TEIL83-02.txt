#!/usr/bin/env python3
"""Teil E (SOLL-IST), nur lesend: Protokollzeilen `freigabe.frei`, `mentor.gewaehlt`, `mass.eingefroren` je Lauf
(Feld lauf_id im JSON-Inhalt) und ob der Grund „Vollmacht“ nennt. Zweck: Masterprompt v8 Abschnitt 4 (Freigabe,
Vollmacht, Richtungswahl) und 3.5 (Maß bestätigt).
STAMM: nur mode=ro&immutable=1 und nur bei leerer WAL; T3: mode=ro. Ausgegeben werden Zählungen und die ersten
120 Zeichen des Grundes je Gruppe — keine Geheimnisse (Protokoll enthält keine Schlüssel). Schreibt nichts."""
import json
import os
import sqlite3
from collections import Counter, defaultdict

STAMM = '/root/biotop/daten/biotop.db'
wal = os.path.getsize(STAMM + '-wal') if os.path.exists(STAMM + '-wal') else None
print(f'Kammer-Sperre vorhanden: {os.path.exists("/tmp/biotop-kammer-exklusiv")}; STAMM-WAL: {wal} Bytes')
QUELLEN = [('T3', 'file:/root/biotop-t3/t3.db?mode=ro')]
if wal in (None, 0):
    QUELLEN.insert(0, ('STAMM', f'file:{STAMM}?mode=ro&immutable=1'))
else:
    print('STAMM NICHT GEÖFFNET (WAL nicht leer)')

for name, uri in QUELLEN:
    con = sqlite3.connect(uri, uri=True)
    print(f'\n===== {name}')
    for art in ('freigabe.frei', 'mentor.gewaehlt', 'mass.eingefroren'):
        je_lauf = Counter()
        vollmacht = Counter()
        gruende = defaultdict(Counter)
        for (inhalt,) in con.execute('SELECT inhalt FROM protokoll WHERE art=? ORDER BY nr', (art,)):
            try:
                d = json.loads(inhalt)
            except (json.JSONDecodeError, TypeError):
                d = {}
            lid = d.get('lauf_id', d.get('lauf'))
            je_lauf[lid] += 1
            text = json.dumps(d, ensure_ascii=False)
            if 'ollmacht' in text:
                vollmacht[lid] += 1
            g = str(d.get('grund', d.get('von', d.get('bestaetigt_von', ''))))[:120]
            gruende[lid][g] += 1
        gesamt = sum(je_lauf.values())
        print(f'\n{art}: {gesamt} Zeilen; davon mit „Vollmacht“ im Inhalt: {sum(vollmacht.values())}')
        for lid in sorted(je_lauf, key=lambda x: (x is None, x if isinstance(x, int) else 0)):
            print(f'   Lauf {lid}: {je_lauf[lid]} (Vollmacht {vollmacht[lid]})')
        if art != 'freigabe.frei':
            continue
        print('   Gründe ohne „Vollmacht“ (erste 120 Zeichen, je Lauf):')
        for lid in sorted(je_lauf, key=lambda x: (x is None, x if isinstance(x, int) else 0)):
            if je_lauf[lid] != vollmacht[lid]:
                for g, n in gruende[lid].items():
                    if 'ollmacht' not in g:
                        print(f'      Lauf {lid}: {n}× {g}')
    con.close()
