import hashlib
from pathlib import Path

D = Path('/root/sicherung/biotop-lage-01/arbeit/dokumente')
for p in sorted(D.iterdir()):
    if not any(k in p.name for k in ('PRUEFUNG', 'ABNAHME', 'WIEDEREINSTIEG', 'MASTERPROMPT_Lauf-25')):
        continue
    b = p.read_bytes()
    t = b.decode('utf-8', errors='replace')
    z = [x for x in t.splitlines() if x.strip()]
    print(f'===== {p.name} | {len(b)} Bytes | sha256 {hashlib.sha256(b).hexdigest()}')
    for x in z[:10]:
        print('  K', x[:300])
    for x in z[10:]:
        if x.startswith('#'):
            print('  H', x[:200])
