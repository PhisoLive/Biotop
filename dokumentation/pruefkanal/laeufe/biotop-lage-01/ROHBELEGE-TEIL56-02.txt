#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil B (nur lesend): Verfassungen in den Kassen-Datenbanken (Tabelle verfassung_werk, Protokoll).

Regeln: STAMM nur mit mode=ro&immutable=1 und nur, wenn biotop.db-wal 0 Bytes hat (vorher geprüft);
alle anderen Datenbanken mit mode=ro. Nichts wird geschrieben. Kein Biotop-Code wird importiert.
Ausgabe: je Zweig Tabellen mit „verfass“, Schema, Zeilen (werk_id, Werkname, Textlänge, SHA-256 von Text und Regeln,
Quelle, eingelegt_am, eingelegt_von), Abgleich mit den Dateien, Protokollzeilen „verfassung.eingelegt“,
Läufe je werk_id; Wortlaut jedes Textes und jeder Regel, der keiner bekannten Datei entspricht.
"""
import hashlib
import json
import os
import sqlite3
import sys

STAMM = '/root/biotop/daten/biotop.db'
ZWEIGE = [('STAMM', STAMM), ('VB1', '/root/phase-4-vorbedingung-1/privat/biotop.db'),
          ('R02', '/root/reparatur-02/privat-cli/privat.db'), ('T3', '/root/biotop-t3/t3.db')]
DATEIEN = {
    'lagerhaus.md': '/root/biotop/verfassungen/lagerhaus.md',
    'lagerhaus.regeln.json': '/root/biotop/verfassungen/lagerhaus.regeln.json',
    'logmuehle.md': '/root/biotop/verfassungen/logmuehle.md',
    'logmuehle.regeln.json': '/root/biotop/verfassungen/logmuehle.regeln.json',
    'VB1 privat/verfassung.md': '/root/phase-4-vorbedingung-1/privat/verfassung.md',
    'VB1 privat/verfassung.regeln.json': '/root/phase-4-vorbedingung-1/privat/verfassung.regeln.json',
    'VERFASSUNG-T3-ENTWURF.md': '/root/biotop-t3/VERFASSUNG-T3-ENTWURF.md',
    'VERFASSUNG-T3-02.md': '/root/biotop-t3/VERFASSUNG-T3-02.md',
    'VERFASSUNG-T3-02.regeln.json': '/root/biotop-t3/VERFASSUNG-T3-02.regeln.json',
}


def h(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


text_bek, regel_bek = {}, {}
for k, p in DATEIEN.items():
    b = open(p, 'rb').read()
    if p.endswith('.json'):
        kk = json.dumps(json.loads(b), sort_keys=True, ensure_ascii=False)
        regel_bek[kk] = (regel_bek[kk] + ' = ' + k) if kk in regel_bek else k
    else:
        for hh, name in ((h(b), k), (h(b.rstrip(b'\n')), k + ' (ohne Schluss-Zeilenumbruch)')):
            text_bek[hh] = (text_bek[hh] + ' = ' + name) if hh in text_bek else name
print('Bekannte Regeldateien (kanonisches JSON gleich):', sorted(regel_bek.values()))
print('Bekannte Textdateien (SHA-256 gleich):', sorted(text_bek.values()))

wal = STAMM + '-wal'
wal_groesse = os.path.getsize(wal) if os.path.exists(wal) else -1
print(f'Vorprüfung: {wal} = {wal_groesse} Bytes')

unbekannt_text, unbekannt_regel = {}, {}
for name, db in ZWEIGE:
    print(f'\n########## {name}: {db}')
    if not os.path.exists(db):
        print('  NICHT GEFUNDEN')
        continue
    if name == 'STAMM':
        if wal_groesse != 0:
            print('  NICHT GEÖFFNET: WAL nicht 0 Bytes')
            continue
        uri = f'file:{db}?mode=ro&immutable=1'
    else:
        uri = f'file:{db}?mode=ro'
    print(f'  URI: {uri}')
    c = sqlite3.connect(uri, uri=True)
    c.row_factory = sqlite3.Row
    tab = [r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
    print(f'  Tabellen gesamt: {len(tab)}; mit „verfass“: {[t for t in tab if "verfass" in t.lower()]}')
    spalten = {}
    for t in tab:
        cols = [r[1] for r in c.execute(f'PRAGMA table_info("{t}")')]
        treffer = [x for x in cols if 'verfass' in x.lower() or 'bestaet' in x.lower()]
        if treffer:
            spalten[t] = treffer
    print(f'  Spalten mit „verfass“/„bestaet“: {spalten}')
    if 'verfassung_werk' not in tab:
        print('  verfassung_werk: NICHT VORHANDEN')
        c.close()
        continue
    print('  Schema:', c.execute("SELECT sql FROM sqlite_master WHERE name='verfassung_werk'").fetchone()[0].strip())
    werke = {r['id']: r['name'] for r in c.execute('SELECT id, name FROM werke')}
    zeilen = c.execute('SELECT * FROM verfassung_werk ORDER BY werk_id').fetchall()
    print(f'  Zeilen verfassung_werk: {len(zeilen)} (Werke in Tabelle werke: {len(werke)})')
    print('  werk_id | Werk | Text Bytes | SHA-256 Text | Text = Datei | Regeln = Datei | Quelle | eingelegt_am | eingelegt_von')
    for r in zeilen:
        tb = (r['text'] or '').encode()
        ht = h(tb)
        try:
            rk = json.dumps(json.loads(r['regeln'] or 'null'), sort_keys=True, ensure_ascii=False)
        except ValueError:
            rk = 'UNLESBAR:' + str(r['regeln'])[:60]
        tdat = text_bek.get(ht, 'KEINE')
        rdat = regel_bek.get(rk, 'KEINE')
        if tdat == 'KEINE':
            unbekannt_text.setdefault(ht, (r['text'], []))[1].append(f'{name}:{r["werk_id"]}')
        if rdat == 'KEINE':
            unbekannt_regel.setdefault(rk, []).append(f'{name}:{r["werk_id"]}')
        print(f'  {r["werk_id"]} | {werke.get(r["werk_id"])} | {len(tb)} | {ht} | {tdat} | {rdat} | {r["quelle"]} | '
              f'{r["eingelegt_am"]} | {r["eingelegt_von"]}')
    werke_ohne = sorted(set(werke) - {r['werk_id'] for r in zeilen})
    print(f'  Werke ohne Verfassungszeile: {[(w, werke[w]) for w in werke_ohne]}')
    print('  Protokoll art=verfassung.eingelegt: nr | zeit | inhalt')
    n = 0
    for r in c.execute("SELECT nr, zeit, inhalt FROM protokoll WHERE art='verfassung.eingelegt' ORDER BY nr"):
        n += 1
        print(f'    {r["nr"]} | {r["zeit"]} | {r["inhalt"]}')
    print(f'  Protokollzeilen verfassung.eingelegt: {n}')
    print('  Läufe je werk_id (id-Liste):')
    lj = {}
    for r in c.execute('SELECT id, werk_id FROM laeufe ORDER BY id'):
        lj.setdefault(r['werk_id'], []).append(r['id'])
    for w in sorted(lj, key=lambda x: (x is None, x or 0)):
        print(f'    werk {w} ({werke.get(w)}): Läufe {lj[w]}')
    c.close()

print('\n########## Texte ohne passende Datei (Wortlaut):', len(unbekannt_text))
for ht, (t, wo) in unbekannt_text.items():
    print(f'--- SHA-256 {ht} · Bytes {len((t or "").encode())} · Vorkommen {wo}')
    print(t)
    print('--- Ende')
print('\n########## Regeln ohne passende Datei (kanonisches JSON):', len(unbekannt_regel))
for rk, wo in unbekannt_regel.items():
    print(f'--- Vorkommen {wo}')
    print(rk)
    print('--- Ende')
print('\nNachprüfung WAL STAMM:', os.path.getsize(wal) if os.path.exists(wal) else -1, 'Bytes')
sys.exit(0)
