#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Läufe und Kasse aus allen Biotop-Datenbankzweigen.
Zweige (Masterprompt v8 Abschnitt 6: „ohne Doppelzählung der Datenbankzweige"):
  STAMM  /root/biotop/daten/biotop.db
  VB1    /root/phase-4-vorbedingung-1/privat/biotop.db
  R02    /root/reparatur-02/privat-cli/privat.db
  T3     /root/biotop-t3/t3.db
Für jeden Zweig: Spalten der Tabellen laeufe und werke; je Lauf: Zeilen der Tabelle laeufe, Buchungen (Anzahl,
Summe kosten_eur, erste/letzte Zeit). Dann: welche Kassenzeilen eines Zweigs sind identisch mit Zeilen des Stamms
(gleiche id und gleicher Inhalt) — daraus die Zuordnung ohne Doppelzählung.
Schreibt arbeit/kasse/*.csv (Semikolon, Dezimalpunkt) und gibt eine Zusammenfassung aus."""
import csv
import hashlib
import sqlite3
from pathlib import Path

ZW = {'STAMM': '/root/biotop/daten/biotop.db', 'VB1': '/root/phase-4-vorbedingung-1/privat/biotop.db',
      'R02': '/root/reparatur-02/privat-cli/privat.db', 'T3': '/root/biotop-t3/t3.db'}
AUS = Path('/root/sicherung/biotop-lage-01/arbeit/kasse')
AUS.mkdir(parents=True, exist_ok=True)
con = {k: sqlite3.connect(f'file:{v}?mode=ro', uri=True) for k, v in ZW.items()}


def zeilen(c, sql, *a):
    cur = c.execute(sql, a)
    return [d[0] for d in cur.description], cur.fetchall()


for k, c in con.items():
    for t in ('laeufe', 'werke', 'kasse'):
        print(f'{k} {t} Spalten:', [r[1] for r in c.execute(f'PRAGMA table_info({t})')])

# Stamm-Kasse als Menge (id -> Hash des Inhalts)
def kassen_hash(c):
    d = {}
    for r in c.execute('SELECT * FROM kasse ORDER BY id'):
        d[r[0]] = hashlib.sha256(repr(r).encode()).hexdigest()
    return d


H = {k: kassen_hash(c) for k, c in con.items()}
stamm = H['STAMM']
for k in ('VB1', 'R02', 'T3'):
    gemeinsam = sum(1 for i, h in H[k].items() if stamm.get(i) == h)
    anders = sum(1 for i, h in H[k].items() if i in stamm and stamm[i] != h)
    neu = sum(1 for i in H[k] if i not in stamm)
    print(f'{k}: Kassenzeilen {len(H[k])}, gleich wie STAMM {gemeinsam}, gleiche id aber anderer Inhalt {anders}, '
          f'nur hier {neu}; STAMM-Zeilen, die hier fehlen {sum(1 for i in stamm if i not in H[k])}')

# Export: laeufe je Zweig mit Kassensumme; nur Läufe, die nicht schon im Stamm identisch stehen
with open(AUS / 'laeufe.csv', 'w', newline='') as f:
    w = csv.writer(f, delimiter=';')
    kopf_gesetzt = False
    for k, c in con.items():
        sp, rows = zeilen(c, 'SELECT * FROM laeufe ORDER BY id')
        if not kopf_gesetzt:
            w.writerow(['zweig'] + sp + ['buchungen', 'kosten_eur', 'erste_buchung', 'letzte_buchung'])
            kopf_gesetzt = True
        for r in rows:
            n, s, a, b = c.execute('SELECT count(*), coalesce(sum(kosten_eur),0), min(zeit), max(zeit) FROM kasse '
                                   'WHERE lauf_id=?', (r[0],)).fetchone()
            w.writerow([k] + list(r) + [n, f'{s:.6f}', a, b])
print('\nLäufe je Zweig (id | Spalten laut Tabelle | Buchungen | Summe):')
for k, c in con.items():
    sp, rows = zeilen(c, 'SELECT * FROM laeufe ORDER BY id')
    print(f'== {k}: {len(rows)} Läufe; Spalten {sp}')
    for r in rows:
        n, s = c.execute('SELECT count(*), coalesce(sum(kosten_eur),0) FROM kasse WHERE lauf_id=?', (r[0],)).fetchone()
        print('  ', ' | '.join(str(x)[:40] for x in r), f'| {n} Buchungen | {s:.4f} €')
    n, s = c.execute('SELECT count(*), coalesce(sum(kosten_eur),0) FROM kasse').fetchone()
    n0, s0 = c.execute('SELECT count(*), coalesce(sum(kosten_eur),0) FROM kasse WHERE lauf_id IS NULL').fetchone()
    print(f'   Summe Kasse {k}: {n} Buchungen, {s:.4f} €; davon ohne lauf_id: {n0}, {s0:.4f} €')
