"""BIOTOP-LAGE-01 (nur lesend): Ursachen der 203 nicht bestandenen Tests aus R196, berichtigt nach R202.

der erste Treffer bestimmt die Ursache. Keine Signatur → „ohne Ursache im Text“. Ausgegeben werden die Zählung je

import collections
import re
import xml.etree.ElementTree as ET
from pathlib import Path

J = Path('/root/sicherung/biotop-lage-01/testlauf/ausgabe/junit.xml')

s = w if w.tag == 'testsuite' else w.find('testsuite')

    ('S1c: rote Spezifikation test_w_g_*', None),
    ('Abschottung: Docker-Dienst nicht erreichbar', r'failed to connect to the docker API|Cannot connect to the Docker daemon|docker\.sock'),

    ('Messung ungültig (gueltig False, Grund im Text)', r"'gueltig': False"),
]

for tc in s.iter('testcase'):
    for art in ('failure', 'error'):
        e = tc.find(art)
        if e is None:
            continue
        text = f'{e.get("message", "")}\n{e.text or ""}'
        name = tc.get('name', '')
        datei = re.sub(r'^.*?(tests/)', r'\1', tc.get('file') or tc.get('classname', '').replace('.', '/') + '.py')
        ursache, stelle = 'ohne Ursache im Text', ''
        for u, m in SIG:
            if m is None:
                if name.startswith('test_w_g'):
                    ursache = u; break
                continue
            t = re.search(m, text)
            if t:
                ursache = u
                stelle = text[max(0, t.start() - 120):t.end() + 120].replace('\n', ' ⏎ ')
                break
        faelle.append((ursache, art, datei, name, text, stelle))
print(f'Fälle: {len(faelle)}')
print('== je Ursache (Anzahl · davon failure/error)')
for u, n in collections.Counter(f[0] for f in faelle).most_common():
    arten = collections.Counter(f[1] for f in faelle if f[0] == u)
    print(f'  {n:4d} · {u} · {dict(arten)}')
print('== je Ursache und Testdatei')
for (u, d), n in sorted(collections.Counter((f[0], f[2]) for f in faelle).items()):
    print(f'  {n:3d} · {u[:55]} · {d}')
print('== je Ursache ein Ausschnitt')
for u in dict.fromkeys(f[0] for f in faelle):
    f = next(f for f in faelle if f[0] == u and f[5]) if any(f[0] == u and f[5] for f in faelle) else None
    if f:
        print(f'  {u[:55]} · {f[2]}::{f[3][:60]} · …{f[5][:330]}…')
print('== ohne Ursache im Text: je Testfunktion ein Vertreter, letzte 12 Zeilen')
gesehen = set()
for f in faelle:
    if f[0] != 'ohne Ursache im Text':
        continue
    funktion = f[3].split('[')[0]

        print('   | ' + re.sub(r'0x[0-9a-f]{6,}', '0x…', z)[:200])
