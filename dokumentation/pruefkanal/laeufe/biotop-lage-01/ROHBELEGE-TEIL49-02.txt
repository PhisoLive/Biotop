#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil B (nur lesend): Treffer aus R41 nach SHA-256 gruppieren.

Liest nur den Rohbeleg R41 (Ausgabe von b1_finden.py). Gibt je Abschnitt (A Name, B Inhalt) die verschiedenen
Inhalte aus: SHA-256, Bytes, Anzahl Pfade, Endung, bis zu 4 Pfade (Vorrang: nicht unter reparatur-01/tmp/pytest).
Code-Kopien des Biotops (biotop/organe/*.py, pruefung/*.py, biotop/kern/*.py usw.) werden als Code gezählt, aber nur
summarisch ausgegeben.
"""
import re
from collections import OrderedDict, defaultdict

R = '/root/sicherung/biotop-lage-01/rohbelege/R0041.txt'
abschnitt = None
gruppen = {'A': OrderedDict(), 'B': OrderedDict()}
for z in open(R, encoding='utf-8'):
    z = z.rstrip('\n')
    if z.startswith('== A.'):
        abschnitt = 'A'
        continue
    if z.startswith('== B.'):
        abschnitt = 'B'
        continue
    if abschnitt is None or ' | ' not in z or z.startswith('Pfad |') or z.startswith('=== Ende'):
        continue
    teile = z.split(' | ')
    if len(teile) < 5:
        continue
    pfad, groesse, sha, zeit, merk = teile[0], teile[1], teile[2], teile[3], ' | '.join(teile[4:])
    g = gruppen[abschnitt].setdefault(sha, {'bytes': groesse, 'pfade': [], 'merk': set(), 'zeiten': []})
    g['pfade'].append(pfad)
    g['merk'].add(merk)
    g['zeiten'].append(zeit)


def code(p: str) -> bool:
    return bool(re.search(r'/(biotop/(organe|kern|anbieter)|pruefung|pruefung_vorbedingung1|tests|kammer)/[^/]+\.(py|mjs|js|ts)$', p)) \
        or p.endswith(('/biotop/lauf.py', '/biotop/nachspiel.py', '/biotop/dienst.py', '/biotop/oberflaeche.py',
                       '/biotop/__main__.py', '/biotop/tor4.py', '/biotop/tor3.py', '/biotop/zwischenbericht.py'))


def vorrang(p: str) -> int:
    return (('/tmp/' in p or 'pytest-of-root' in p or '/eltern-temp/' in p) * 2) + ('/reparatur-01/' in p)


for ab in 'AB':
    print(f'===== Abschnitt {ab}: {len(gruppen[ab])} verschiedene Inhalte, {sum(len(g["pfade"]) for g in gruppen[ab].values())} Pfade')
    code_n = defaultdict(int)
    for sha, g in gruppen[ab].items():
        if all(code(p) for p in g['pfade']):
            for p in g['pfade']:
                code_n[p.rsplit('/', 1)[-1]] += 1
            continue
        pf = sorted(g['pfade'], key=lambda p: (vorrang(p), p))
        endg = sorted({p.rsplit('.', 1)[-1] if '.' in p.rsplit('/', 1)[-1] else '-' for p in g['pfade']})
        print(f'- {sha} | {g["bytes"]} B | {len(pf)} Pfade | Endungen {",".join(endg)} | Merkmale {";".join(sorted(g["merk"]))}')
        for p in pf[:4]:
            print(f'    {p}')
        if len(pf) > 4:
            print(f'    … {len(pf) - 4} weitere')
    print(f'  Code-Dateien des Biotops (nur gezählt, Name → Anzahl Pfade): {dict(sorted(code_n.items()))}')
