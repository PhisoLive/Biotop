#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Zweite Stufe der Ursachenzuordnung zu R203 für die 83 Fälle „ohne Ursache im Text“.
Signatur „Kammer-Pfad im Fehlertext“ (Auslegung Hermes): Der Fehlertext zeigt einen Aufruf der Kammer
(messgeraet.messen, aegis.junit_lauf, kammer., 'Kammer abgebrochen', basis['gueltig'] nach messen). Das belegt nur,
dass der Test die Kammer braucht, nicht, dass ihr Fehlen die Ursache ist."""
import collections
import re
import xml.etree.ElementTree as ET
from pathlib import Path

J = Path('/root/sicherung/biotop-lage-01/testlauf/ausgabe/junit.xml')
w = ET.parse(J).getroot()
s = w if w.tag == 'testsuite' else w.find('testsuite')
ERST = [r'failed to connect to the docker API|docker\.sock', r'Kammer nicht bereit|Kammer-Image',
        r"No such file or directory: '(node|npx|tsc)'", r'AuftragsgrenzeFehlt', r'Kernstand seit der Eichphase']
KAMMER = r"Kammer abgebrochen|messgeraet\.messen\(|aegis\.junit_lauf\(|\bkammer\.[a-z_]+\(|_lahmlegen\(|eichproben\._"
zaehl, ohne = collections.Counter(), []
for tc in s.iter('testcase'):
    for art in ('failure', 'error'):
        e = tc.find(art)
        if e is None:
            continue
        text = f'{e.get("message", "")}\n{e.text or ""}'
        name = tc.get('name', '')
        if name.startswith('test_w_g') or any(re.search(m, text) for m in ERST):
            continue
        datei = re.sub(r'^.*?(tests/)', r'\1', tc.get('file') or tc.get('classname', '').replace('.', '/') + '.py')
        if re.search(KAMMER, text):
            zaehl[('Kammer-Pfad im Fehlertext', datei)] += 1
        else:
            zaehl[('auch ohne Kammer-Pfad', datei)] += 1
            ohne.append(f'{datei}::{name}')
print(f'Fälle ohne Ursache im Text (R203): {sum(zaehl.values())}')
for (u, d), n in sorted(zaehl.items()):
    print(f'  {n:3d} · {u} · {d}')
print(f'Summe Kammer-Pfad: {sum(n for (u, d), n in zaehl.items() if u.startswith("Kammer"))} · ohne: {len(ohne)}')
for o in ohne:
    print(f'  ohne Kammer-Pfad: {o}')
