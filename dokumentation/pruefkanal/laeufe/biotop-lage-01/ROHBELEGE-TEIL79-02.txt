#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Prüfung der eigenen Absender-Einordnung in arbeit/verlauf/wortlaut.jsonl.
Befund aus R84: Zwei Telegram-Sitzungen der Kette begannen als Review-Subagenten und trugen den Chat erst ab dem
Sitzungswechsel (end_reason session_switch der Elternsitzung):
  20260924_150913_a9326d — Elternsitzung 20260923_194003_92313ee1 endet 2026-09-24 16:10:41 (session_switch)
  20260924_203120_e6c401 — Elternsitzung 20260924_150913_a9326d endet 2026-09-24 21:51:24 (session_switch)
Vor diesem Zeitpunkt sind role=user-Nachrichten dort Aufträge von Hermes an den Reviewer (nicht Philipp) und
role=assistant-Nachrichten die des Reviewers (nicht Hermes an Philipp). Ausgabe: jede betroffene Nachricht mit id,
Zeit, bisheriger Einordnung, Länge und ersten 80 Zeichen (geschwärzte Fassung aus wortlaut.jsonl)."""
import json
from collections import Counter

GRENZE = {'20260924_150913_a9326d': '2026-09-24 16:10:41', '20260924_203120_e6c401': '2026-09-24 21:51:24'}
betroffen = []
with open('/root/sicherung/biotop-lage-01/arbeit/verlauf/wortlaut.jsonl', encoding='utf-8') as f:
    for z in f:
        e = json.loads(z)
        g = GRENZE.get(e['sitzung'])
        if g and e['zeit'] < g:
            betroffen.append(e)
c = Counter(e['absender'] for e in betroffen)
print(f'Betroffene Nachrichten (vor dem Sitzungswechsel in den zwei Sitzungen): {len(betroffen)} · bisherige Einordnung: {dict(c)}')
for e in betroffen:
    if e['absender'] in ('Philipp', 'Hermes'):
        t = e['text'].replace('\n', ' ')[:80]
        print(f"  {e['zeit']} | id {e['id']} | {e['sitzung'][-6:]} | bisher {e['absender']} | {e['bytes']} B | {t}")
print('Richtige Einordnung (Auslegung Hermes): Philipp → Hermes→Subagent (Reviewauftrag); Hermes → Subagent (Bericht des '
      'Reviewers); Hermes-Zwischentext → Subagent-Zwischentext; System-* bleibt.')
