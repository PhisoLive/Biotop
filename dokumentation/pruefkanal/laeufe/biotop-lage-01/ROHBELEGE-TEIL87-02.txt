#!/usr/bin/env python3
"""Teil E (SOLL-IST), nur lesend: je Lauf in T3 (26–38) die erste Protokollzeile `freigabe.frei` im vollen Wortlaut des
Feldes `grund`, und ob die Freigaben einer Generation im Protokoll unmittelbar auf `generation.*` folgen (Zeitabstand).
Zweck: Masterprompt v8 Abschnitt 4 Z. 287 (Vollautomatik) und Z. 288 (Vollmacht). T3: mode=ro. Schreibt nichts.
Keine Geheimnisse: Das Protokoll enthält keine Schlüssel; ausgegeben werden nur die Felder grund, von, lauf_id,
generation und die Zeitstempel."""
import json
import sqlite3
from datetime import datetime

con = sqlite3.connect('file:/root/biotop-t3/t3.db?mode=ro', uri=True)
zeilen = con.execute("SELECT nr, zeit, inhalt FROM protokoll WHERE art='freigabe.frei' ORDER BY nr").fetchall()
gesehen = set()
abstaende = []
for nr, zeit, inhalt in zeilen:
    d = json.loads(inhalt)
    lid = d.get('lauf_id')
    if lid is None or lid < 26:
        continue
    vor = con.execute("SELECT zeit, art FROM protokoll WHERE nr < ? AND art NOT LIKE 'freigabe%' "
                      "ORDER BY nr DESC LIMIT 1", (nr,)).fetchone()
    try:
        dt = (datetime.fromisoformat(zeit) - datetime.fromisoformat(vor[0])).total_seconds()
    except (ValueError, TypeError):
        dt = None
    abstaende.append((lid, d.get('nummer', d.get('generation')), dt, vor[1] if vor else None))
    if lid not in gesehen:
        gesehen.add(lid)
        print(f'Lauf {lid}, Protokoll #{nr}, {zeit}')
        print(f'   Felder: {sorted(d)}')
        print(f'   grund (voll): {d.get("grund")}')
        print(f'   von: {d.get("von")}')
print('\nAbstand jeder Freigabe (Läufe 26–38) zur vorangehenden Protokollzeile anderer Art:')
for lid, gen, dt, art in abstaende:
    print(f'   Lauf {lid} Gen {gen}: {dt} s nach „{art}“')
print(f'Anzahl: {len(abstaende)}')
