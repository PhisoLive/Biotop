Werkzeugaufrufen auf, und gab es sie vor Beginn der Runde 6? Ausgabe ohne Inhalte: Zeit, Sitzung, Quelle, Elternsitzung,
Werkzeugnamen; dazu der Anfang des ersten Auftrags jeder beteiligten Sitzung (welcher Bauauftrag)."""
import datetime as dt
import json
import sqlite3

f = lambda ts: dt.datetime.fromtimestamp(ts).strftime('%Y-%m-%d %H:%M:%S')
con = sqlite3.connect('file:/root/.hermes/state.db?mode=ro', uri=True)
sess = {r[0]: (r[1], r[2]) for r in con.execute('SELECT id, source, parent_session_id FROM sessions')}
COMMIT = dt.datetime(2026, 9, 24, 20, 24, 54).timestamp()   # Commit 0c2f058 (Runde 5)
R6 = dt.datetime(2026, 9, 24, 23, 45).timestamp()           # Beginn Runde 6 laut ihrer Übergabe


def auftrag(sid):
    r = con.execute("SELECT content FROM messages WHERE session_id=? AND role='user' ORDER BY timestamp, id LIMIT 1",
                    (sid,)).fetchone()
    return (r[0] or '')[:120].replace('\n', ' ') if r else ''


sids = set()
for name in ('stufe2.py', 'test_tor4_zeilen.py'):
    print(f'-- {name}')
    q = con.execute("SELECT id, session_id, timestamp, tool_calls FROM messages WHERE role='assistant' AND tool_calls LIKE ? "
                    "ORDER BY timestamp, id", (f'%{name}%',)).fetchall()
    print(f'   Werkzeugaufrufe (Zeilen) mit dem Namen: {len(q)}; vor dem Commit 0c2f058 ({f(COMMIT)}): '
          f'{sum(1 for x in q if x[2] < COMMIT)}; zwischen Commit und Beginn Runde 6 ({f(R6)}): '
          f'{sum(1 for x in q if COMMIT <= x[2] < R6)}')
    for mid, sid, ts, tc in q[:3]:
        try:
            namen = [c.get('function', {}).get('name') or c.get('name') for c in json.loads(tc)]
        except Exception:
            namen = ['?']
        src, eltern = sess.get(sid, ('?', '?'))
        print(f'   erste: {f(ts)} | id {mid} | Sitzung {sid} ({src}, Eltern {eltern}) | Werkzeuge {namen}')
    n = con.execute('SELECT count(*) FROM messages WHERE content LIKE ? AND timestamp < ?', (f'%{name}%', R6)).fetchone()[0]
    print(f'   Nachrichten (Textinhalt, alle Rollen) mit dem Namen vor Beginn Runde 6: {n}')
    sids |= {x[1] for x in q}
print('-- Sitzungen mit Werkzeugaufrufen, die einen der Namen enthalten, und der Anfang ihres ersten Auftrags:')
for sid in sorted(sids):
    print(f'   {sid} ({sess.get(sid, ("?",))[0]}): {auftrag(sid)!r}')
