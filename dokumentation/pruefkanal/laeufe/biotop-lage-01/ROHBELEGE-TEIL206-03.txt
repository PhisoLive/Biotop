#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Nachprüfung zu R201 für die 11 Fälle test_w_g_* und die 24 Fälle „sonst“:
enthält der Fehlertext Hinweise auf die Abschottung (docker, node, Umgebungsvariable, /opt/node)?
Gibt je Fall die letzte Meldungszeile (gekürzt) und die gefundenen Stichwörter aus."""
import re
import xml.etree.ElementTree as ET
from pathlib import Path

J = Path('/root/sicherung/biotop-lage-01/testlauf/ausgabe/junit.xml')
w = ET.parse(J).getroot()
s = w if w.tag == 'testsuite' else w.find('testsuite')
STICH = {'docker': r'docker', 'node': r"'node'|No such file or directory: 'node'|/opt/node|npx|tsc\b",
         'Umgebungsvariable': r'BIOTOP_[A-Z_]+ (fehlt|und)|AuftragsgrenzeFehlt', 'Netz': r'Errno 101|unreachable',
         'schreibgeschützt': r'Errno 30|Read-only'}
n = 0
for tc in s.iter('testcase'):
    for art in ('failure', 'error'):
        e = tc.find(art)
        if e is None:
            continue
        text = f'{e.get("message", "")}\n{e.text or ""}'
        if re.search(r'docker', text) and not tc.get('name', '').startswith('test_w_g'):
            continue  # die 168 Docker-Fälle sind in R201 belegt
        n += 1
        treffer = [k for k, m in STICH.items() if re.search(m, text)]
        zeilen = [z.strip() for z in text.splitlines() if re.match(r'\s*E\s', z)]
        letzte = (zeilen[-1] if zeilen else (e.get('message') or '').splitlines()[0] if e.get('message') else '')[:170]
        datei = re.sub(r'^.*?(tests/)', r'\1', tc.get('file') or tc.get('classname', ''))
        print(f'{n:2d} {art[:4]} · {datei}::{tc.get("name")[:70]} · Stichwörter {treffer or "keine"} · {letzte}')
print(f'geprüft: {n} Fälle (erwartet 35 = 203 − 168)')
