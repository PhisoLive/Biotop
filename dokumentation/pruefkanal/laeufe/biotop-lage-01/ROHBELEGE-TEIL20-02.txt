#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): (1) Nachricht 1892 (v1/v4-Teilstücke) einordnen; (2) Dokumente Chat-Claudes, die ohne
Dokumentkopf eingefügt wurden („[Content of …]:"), und Nennungen weiterer Dokumentnamen (…_2026-09-2x.md) in
Philipps Nachrichten; (3) Auftragsdateien auf der Platte nach Namensmuster (Pfad, Zeit, Größe, SHA-256)."""
import datetime as dt
import hashlib
import re
import sqlite3
from pathlib import Path

con = sqlite3.connect('file:/root/.hermes/state.db?mode=ro', uri=True)
print('== 1. Nachricht 1892: Rolle, Sitzung, Länge, erste 300 Zeichen, Dokumentköpfe')
r = con.execute('SELECT role, session_id, timestamp, length(content), content FROM messages WHERE id=1892').fetchone()
print(r[0], r[1], dt.datetime.fromtimestamp(r[2]), r[3])
print(repr(r[4][:300]))
print('Dokumentköpfe:', re.findall(r"\[The user sent a[^\]]{0,200}\]", r[4])[:5])
print('Content-Marken:', re.findall(r"\[Content of [^\]]+\]", r[4])[:5])

print('\n== 2. „[Content of …]" und genannte Dokumentnamen in Nachrichten der Rolle user (ab 18.09., ohne Dubletten)')
ab = dt.datetime(2026, 9, 18).timestamp()
inhalte, namen = {}, {}
for mid, ts, c in con.execute("SELECT id, timestamp, content FROM messages WHERE role='user' AND timestamp>=? ORDER BY timestamp", (ab,)):
    for m in re.finditer(r'\[Content of ([^\]]+)\]', c or ''):
        inhalte.setdefault(m.group(1), (mid, dt.datetime.fromtimestamp(ts).strftime('%Y-%m-%d %H:%M:%S')))
    if (c or '').lstrip().startswith('['):
        # System-/Dokumentnachrichten nur für Content-Marken auswerten
        pass
    for m in re.finditer(r'\b([A-Z][A-Za-z0-9-]*_[A-Za-z0-9_.-]*20\d\d-\d\d-\d\d\.md)\b', c or ''):
        namen.setdefault(m.group(1), (mid, dt.datetime.fromtimestamp(ts).strftime('%Y-%m-%d %H:%M:%S')))
print('Content-Marken:', len(inhalte))
for k, v in sorted(inhalte.items(), key=lambda x: x[1][1]):
    print('  ', v[1], 'id', v[0], k)
print('Genannte Dokumentnamen:', len(namen))
for k, v in sorted(namen.items(), key=lambda x: x[1][1]):
    print('  ', v[1], 'id', v[0], k, '| als Inhalt vorhanden:', 'ja' if k in inhalte else 'NEIN')

print('\n== 3. Auftragsdateien auf der Platte (Namensmuster), ohne node_modules/.git/venv/tmp-Testordner')
muster = re.compile(r'(AUFTRAG|PRUEFUNG|ABNAHME|WIEDEREINSTIEG|ANTWORT|NACHFRAGE|GEDAECHTNIS_|MASTERPROMPT|UEBERGABE|^BAU-)', re.I)
wurzeln = ['/root', '/var/pruefkanal', '/tmp']
gef = []
for w in wurzeln:
    for p in Path(w).rglob('*'):
        s = str(p)
        if any(x in s for x in ('/node_modules/', '/.git/', '/venv/', '/.venv/', '/pytest-of-root/', '/sicherung/biotop-lage-01/',
                                '/.hermes/skills/', '/site-packages/', '/.cache/', '/.npm/', '/.claude/', '/.codex/')):
            continue
        if p.is_file() and p.suffix in ('.md', '.txt') and muster.search(p.name):
            gef.append(p)
print('Anzahl:', len(gef))
for p in sorted(gef):
    b = p.read_bytes()
    print(f'{dt.datetime.fromtimestamp(p.stat().st_mtime):%Y-%m-%d %H:%M:%S} | {len(b)} | {hashlib.sha256(b).hexdigest()[:16]} | {p}')
