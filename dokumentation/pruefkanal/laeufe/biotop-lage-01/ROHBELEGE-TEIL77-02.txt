"""BIOTOP-LAGE-01 Teil D (nur lesend): Prüfweise je Bauteil — FK32.

Am Commit 0c2f058 (git show): Für jedes Modul unter biotop/ die Testdateien unter tests/, die es importieren oder per
Modulname ansprechen (Muster: „organe import …name…“, „organe.name“, „kern import …name…“, „kern.name“, „biotop import
…name…“, „biotop.name“). Dazu: Zahl der Einträge in register.VERLANGT, in oberflaeche.ELEMENTE und in dienst.ROUTEN.
"""
import re
import subprocess

REPO, C = '/root/biotop', '0c2f058'


def git(*a) -> str:
    return subprocess.run(['git', '-C', REPO, *a], check=True, stdout=subprocess.PIPE).stdout.decode('utf-8', 'replace')


dateien = git('ls-tree', '-r', '--name-only', C).splitlines()
tests = {p: git('show', f'{C}:{p}') for p in dateien if p.startswith('tests/') and p.endswith('.py')}
module = [p for p in dateien if p.startswith('biotop/') and p.endswith('.py') and not p.endswith('__init__.py')]
module.append('biotop/anbieter/__init__.py')
print('== 1. Modul → Testdateien, die es importieren oder ansprechen (Anzahl; Namen)')
for m in sorted(module):
    teile = m[:-3].split('/')
    name = teile[-1] if teile[-1] != '__init__' else teile[-2]
    paket = teile[-2] if teile[-1] != '__init__' else teile[-3]
    muster = re.compile(
        rf'(from\s+biotop(\.{paket})?\s+import\s+[^\n]*\b{name}\b|\bbiotop\.({paket}\.)?{name}\b|\b{paket}\.{name}\b'
        rf'|import\s+biotop\.({paket}\.)?{name}\b|from\s+\.\.?{paket}?\s+import\s+[^\n]*\b{name}\b)')
    treffer = sorted(p.split('/')[-1] for p, t in tests.items() if muster.search(t))
    print(f'{m}: {len(treffer)} — {", ".join(treffer)}')

print('\n== 2. Einträge in register.VERLANGT (Zeilen mit Tupel-Anfang innerhalb VERLANGT)')
reg = git('show', f'{C}:biotop/kern/register.py')
blk = reg.split('VERLANGT', 1)[1] if 'VERLANGT' in reg else ''
blk = blk[:blk.find('\n]') + 2] if '\n]' in blk else blk
eintraege = re.findall(r'^\s+\("([^"]+)",\s*"([^"]+)",\s*"([^"]+)"', blk, re.M)
print(len(eintraege), 'Einträge')
for e in eintraege:
    print('  ', ' ← '.join(e[:2]), '@', e[2])

print('\n== 3. Einträge in oberflaeche.ELEMENTE und dienst.ROUTEN')
ob = git('show', f'{C}:biotop/oberflaeche.py')
blk = ob.split('ELEMENTE = {', 1)[1].split('\n}', 1)[0]
print('ELEMENTE:', len(re.findall(r'^\s+"[^"]+":\s*\(', blk, re.M)))
