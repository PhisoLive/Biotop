#!/usr/bin/env python3
"""Teil E (SOLL-IST), nur lesend: Mentor- und Jury-Aufrufe je Lauf und Protokollarten `mentor.*`/`jury.*` in T3
(Läufe >= 26) gegen STAMM. Zweck: Masterprompt v8 Abschnitt 3.4/4 (Mentor, Richtungswahl) und 6 (schwere Aufrufe:
Mentor, Jury-Endurteil). STAMM nur mode=ro&immutable=1 bei leerer WAL; T3 mode=ro. Ausgegeben werden nur Zählungen,
Zwecke und Modellnamen. Kein Import von Biotop-Code. Schreibt nichts."""
import json
import os
import sqlite3

STAMM = '/root/biotop/daten/biotop.db'
wal = os.path.getsize(STAMM + '-wal') if os.path.exists(STAMM + '-wal') else None
print(f'Kammer-Sperre vorhanden: {os.path.exists("/tmp/biotop-kammer-exklusiv")}; STAMM-WAL: {wal} Bytes')
QUELLEN = [('T3', 'file:/root/biotop-t3/t3.db?mode=ro')]
if wal in (None, 0):
    QUELLEN.insert(0, ('STAMM', f'file:{STAMM}?mode=ro&immutable=1'))
for name, uri in QUELLEN:
    con = sqlite3.connect(uri, uri=True)
    print(f'\n===== {name}')
    print('(1) Kasse: Zwecke mit „mentor“ oder „jury“, je Lauf-Gruppe (<26 / >=26), Modell, Anzahl')
    for z in con.execute("SELECT CASE WHEN lauf_id >= 26 THEN '>=26' ELSE '<26' END g, zweck, modell, COUNT(*) "
                         "FROM kasse WHERE zweck LIKE '%mentor%' OR zweck LIKE '%jury%' GROUP BY 1, 2, 3 ORDER BY 1, 2"):
        print(f'   Läufe {z[0]}: zweck={z[1]} modell={z[2]}: {z[3]}')
    print('(2) Protokollarten mentor.* / jury.* / richtung*: Zeilen gesamt und mit lauf_id >= 26')
    arten = [r[0] for r in con.execute("SELECT DISTINCT art FROM protokoll WHERE art LIKE 'mentor%' OR art LIKE 'jury%' "
                                       "OR art LIKE 'richtung%' ORDER BY 1")]
    for art in arten:
        gesamt = 0
        ab26 = 0
        for (inhalt,) in con.execute('SELECT inhalt FROM protokoll WHERE art=?', (art,)):
            gesamt += 1
            try:
                d = json.loads(inhalt)
            except (json.JSONDecodeError, TypeError):
                continue
            lid = d.get('lauf_id', d.get('lauf')) if isinstance(d, dict) else None
            if isinstance(lid, int) and lid >= 26:
                ab26 += 1
        print(f'   {art}: {gesamt} gesamt, davon lauf_id >= 26: {ab26}')
    print('(3) Tabelle mentor: Zeilen je Lauf-Gruppe')
    for z in con.execute("SELECT CASE WHEN lauf_id >= 26 THEN '>=26' ELSE '<26' END, COUNT(*) FROM mentor GROUP BY 1"):
        print(f'   Läufe {z[0]}: {z[1]}')
    con.close()
