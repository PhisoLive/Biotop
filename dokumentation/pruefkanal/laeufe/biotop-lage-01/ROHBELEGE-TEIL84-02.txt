import json
import re
from pathlib import Path

P = Path('/root/sicherung/biotop-lage-01/arbeit/verlauf/v2/wortlaut.jsonl')
MUSTER = re.compile(r'Phase 5|Phase-5|erstes echtes Werk|erste echte Werk|ersten echten Werk', re.I)
n = 0
for z in P.open(encoding='utf-8'):
    d = json.loads(z)
    t = d.get('text') or ''
    for m in MUSTER.finditer(t):
        n += 1
        a = max(0, m.start() - 150)
        print(f"id {d.get('id')} · {d.get('zeit')} · {d.get('absender')} · {d.get('sitzung')} · biotop={d.get('biotop')}")
        print('   …' + t[a:m.end() + 150].replace('\n', ' ') + '…')
print(f'Treffer: {n}')
