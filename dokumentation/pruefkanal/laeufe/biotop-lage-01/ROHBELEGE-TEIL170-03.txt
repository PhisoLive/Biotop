#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Vorgehen 1 in Abschnitt 4. Misst den Plattenplatz und die Größe jedes Umfangsteils
genau so, wie arbeit/sich.py ihn sichert: dieselben Wurzeln (sich.teile()), dieselbe Auslassung (sich.laufe()).
Öffnet keine Datenbank; s2_platz.sh tat das mit bloßem mode=ro, deshalb wird es nicht wiederholt (A6).
Dazu werden Dateien genannt, deren Name auf ein Geheimnis deuten kann (nur Pfad, nie Inhalt), jeweils mit dem Hinweis,
ob sich.py sie auslässt."""
import os
import re
import shutil
import subprocess
import sys
import time
from collections import Counter

sys.path.insert(0, '/root/sicherung/biotop-lage-01/arbeit')
import sich  # noqa: E402

print(time.strftime('== Zeitpunkt %Y-%m-%d %H:%M:%S %Z (UTC%z)'))
print(subprocess.run(['df', '-h', '/'], capture_output=True, text=True).stdout.rstrip())
gesamt_b, gesamt_n = 0, 0
verdacht = re.compile(r'(token|secret|credential|passw|\.pem$|\.key$|^id_(rsa|ed25519|ecdsa|dsa)|cookie|oauth)', re.I)
funde = []
for teil, wurzeln in sich.teile().items():
    z, byt = Counter(), 0
    for wz in wurzeln:
        if not wz.exists():
            z['wurzel-fehlt'] += 1
            continue
        for art, p in sich.laufe(wz, teil):
            z[art] += 1
            name = os.path.basename(p)
            if art in ('datei', 'aus-geheim') and verdacht.search(name) and teil != '1-biotop':
                funde.append((teil, art, p))
            if art == 'datei':
                try:
                    byt += os.lstat(p).st_size
                except OSError:
                    z['verschwunden'] += 1
    gesamt_b += byt
    gesamt_n += z['datei']
    print(f'-- {teil}: {len(wurzeln)} Wurzeln · Bytes der Dateien {byt} ({byt / 2**30:.2f} GiB) · {dict(z)}')
frei = shutil.disk_usage('/').free
print(f'Summe: {gesamt_n} Dateien · {gesamt_b} Bytes ({gesamt_b / 2**30:.2f} GiB)')
print(f'Bedarf (gerechnet): Zwischenstand + entpackte Probe = {2 * gesamt_b / 2**30:.2f} GiB, dazu das Archiv (höchstens '
      f'{gesamt_b / 2**30:.2f} GiB); die Gegenprobe nutzt harte Verweise. Frei: {frei / 2**30:.2f} GiB → reicht: '
      f'{"ja" if frei > 3 * gesamt_b + 5 * 2**30 else "NEIN"} (Rechnung: frei > 3 × Umfang + 5 GiB Reserve)')
