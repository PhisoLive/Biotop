def saeubern(t, K, L):
    t = t.replace(str(L), '{LAEUFE}').replace(K, '<schlüssel>')
    for name, m in geheimnis.MUSTER.items():
        t = re.sub(m, f'[GESCHWÄRZT: {name}]', t)
    return t


def exklusiv(p, text):
    with open(p, 'x', encoding='utf-8') as f:
        f.write(text)
    return p


def schreibe(aus, name, text, K, L):
    return exklusiv(aus / name, saeubern(text, K, L))


def schreibe_md(aus, kopf, abschnitte, K, L):
    bloecke = []
    for titel, zeilen in abschnitte:
        teil, cur = [], [f'## {titel}', '']
        for x in zeilen:
            x = saeubern(x, K, L)
            if len('\n'.join(cur + [x]).encode()) > MAX_MD - 2500:
                teil.append(cur)
                cur = [f'## {titel} (Fortsetzung)', '']
                if x.startswith('|') and not x.startswith('|---'):
                    kz = next((c for c in reversed(teil[-1]) if c.startswith('|---')), None)
                    ki = teil[-1].index(kz) if kz else None
                    if ki:
                        cur += [teil[-1][ki - 1], kz]
            cur.append(x)
        teil.append(cur)
        bloecke += ['\n'.join(b) + '\n' for b in teil]
    k = saeubern('\n'.join(kopf) + '\n', K, L)
    teile, cur = [], k
    for b in bloecke:
        if len((cur + '\n' + b).encode()) > MAX_MD - 400:
            teile.append(cur)
            cur = ''
        cur = (cur + '\n' + b) if cur else b
    teile.append(cur)
    if len(teile) == 1:
        return [exklusiv(aus / 'INVENTAR.md', teile[0])]
    out = []
    for i, t in enumerate(teile, 1):
        vor = f'> INVENTAR Teil {i} von {len(teile)} (INVENTAR-TEIL1.md … INVENTAR-TEIL{len(teile)}.md)\n\n'
        out.append(exklusiv(aus / f'INVENTAR-TEIL{i}.md', vor + t))
    return out


def schreibe_csv(aus, tabellen, K, L):
    out, n = [], 0
    for name, kopf, zeilen in tabellen:
        buf = io.StringIO()
        csv.writer(buf, delimiter=';', lineterminator='\n').writerow(kopf)
        kopfz = buf.getvalue()
        cur = kopfz
        for r in zeilen:
            b = io.StringIO()
            csv.writer(b, delimiter=';', lineterminator='\n').writerow([saeubern(str(x), K, L) for x in r])
            zl = b.getvalue()
            if len((cur + zl).encode()) > MAX_ROH:
                n += 1
                out.append(exklusiv(aus / f'KASSE-TEIL{n}.csv', cur))
                cur = kopfz
            cur += zl
        n += 1
        out.append(exklusiv(aus / f'KASSE-TEIL{n}.csv', cur))
    return out


def schreibe_listen(aus, li, K, L):
    out, n, cur = [], 0, ''
    for titel, zeilen in li.items():
        kopf = f'== {titel} ({len(zeilen)} Einträge)\n'
        if len((cur + kopf).encode()) > MAX_ROH - 200:
            n += 1
            out.append(exklusiv(aus / f'LISTEN-TEIL{n}.txt', cur))
            cur = ''
        cur += kopf
        for x in zeilen:
            x = saeubern(x, K, L) + '\n'
            if len((cur + x).encode()) > MAX_ROH:
                n += 1
                out.append(exklusiv(aus / f'LISTEN-TEIL{n}.txt', cur))
                cur = f'== {titel} (Fortsetzung)\n'
            cur += x
    n += 1
    out.append(exklusiv(aus / f'LISTEN-TEIL{n}.txt', cur))
    return out

#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil F (nur lesend): Stand der Review-Dateien der Behebungsrunden 1–6 und der Übergabe Runde 6.
Je Datei: vorhanden ja/nein, Bytes, mtime (lokal, CEST), SHA-256 (erste 16 Zeichen), Zahl der Zeilen, die mit
„| <Zahl> |" beginnen (Befundzeilen der Tabellen), und die erste Überschrift. Prüfkanal-Kopien unter {LAEUFE}/…
(der Schlüssel wird intern gelesen und nicht ausgegeben). Schreibt nichts."""
import datetime as dt
import hashlib
import re
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
DATEIEN = [
    '/root/biotop-vor-t4-06/UEBERGABE.md',
    '/root/biotop-vor-t4/REVIEW.md', '/root/biotop-vor-t4-02/REVIEW.md', '/root/biotop-vor-t4-03/REVIEW.md',
    '/root/biotop-vor-t4-04/REVIEW.md', '/tmp/review-vor-t4-04/REVIEW-B.md', '/tmp/review-vor-t4-04/REVIEW.md',
    '/root/biotop-vor-t4-05/REVIEW.md', '/root/biotop-vor-t4-05/REVIEW-verworfen-erster-durchgang.md',
    '/tmp/review-vor-t4-05/REVIEW.md', '/root/biotop-vor-t4-05/ZWEI-REVIEWS.md',
]
for o in ('biotop-vor-t4', 'biotop-vor-t4-02', 'biotop-vor-t4-03', 'biotop-vor-t4-04', 'biotop-vor-t4-05',
          'biotop-vor-t4-06'):
    for n in ('REVIEW.md', 'REVIEW-B.md', 'REVIEW-verworfen-erster-durchgang.md', 'ZWEI-REVIEWS.md', 'BERICHT.md'):
        DATEIEN.append(str(L / o / n))
for d in DATEIEN:
    p = Path(d)
    name = d.replace(str(L), '{LAEUFE}')
    if not p.exists():
        print(f'{name}: NICHT VORHANDEN')
        continue
    b = p.read_bytes()
    t = b.decode('utf-8', errors='replace')
    kopf = next((z for z in t.splitlines() if z.startswith('#')), '(keine Überschrift)')
    befund = len(re.findall(r'(?m)^\| *\d+ *\|', t))
    print(f'{name}: {len(b)} Bytes | mtime {dt.datetime.fromtimestamp(p.stat().st_mtime):%Y-%m-%d %H:%M:%S} | '
          f'sha256 {hashlib.sha256(b).hexdigest()[:16]} | Tabellenzeilen mit Nummer {befund} | {kopf[:150]}')
print('\nRunde 6, Dateien REVIEW* im Arbeitsordner:', sorted(p.name for p in Path('/root/biotop-vor-t4-06').glob('REVIEW*')))
print('Runde 6, /tmp/review-vor-t4-06-*:', sorted(p.name for p in Path('/tmp').glob('review-vor-t4-06*')))
