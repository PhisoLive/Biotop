def dateien(wurzel: Path):
    for q in sorted(wurzel.rglob('*.md')):
        if any(t in AUS for t in q.relative_to(wurzel).parts[:-1]):
            continue
        yield q


quellen = []
for d in sorted(L.glob('biotop-*')):

        m = re.match(r'^(#+) .*(Eigene Entscheidungen|Entscheidungen Hermes|Eigene Entscheidung)', s)
        if m:

                    n += 1
                j += 1
            abschnitte.append((q, i, s.strip(), n, j))
m = lambda p: str(p).replace(str(L), '{LAEUFE}')
print(f'Dateien mit Treffern: {len(treffer)}; Trefferzeilen gesamt: {sum(len(v) for v in treffer.values())}')

print('\n== Trefferzeilen')
for q, v in treffer.items():
    for i, s in v:
