#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil C (nur lesend): Erkundung für FK28/FK29.
STAMM nur mit ?mode=ro&immutable=1 und nur bei -wal = 0 Bytes; VB1 und T3 mit ?mode=ro.
Ausgabe je Zweig:
 1. Protokollarten (Tabelle protokoll, Spalte art) mit 'immun', 'aufnahme', 'abweich', 'ansage', 'mentor',
    'gedaechtnis', 'muster' im Namen: Anzahl, erste und letzte Zeit.
 2. Spalten der Tabellen mentor und kandidaten.
 3. Letzte Zeile der Tabelle mentor: jede Spalte, Text auf 700 Zeichen gekürzt.
 4. Je eine Beispielzeile (die jüngste) der Protokollarten immun.aufnahme_gesperrt und immun.aufnahme_geprueft
    und aegis.abweichung (inhalt auf 700 Zeichen gekürzt)."""
import os
import sqlite3

ZWEIGE = [
    ('STAMM', '/root/biotop/daten/biotop.db', 'immutable'),
    ('VB1', '/root/phase-4-vorbedingung-1/privat/biotop.db', 'ro'),
    ('T3', '/root/biotop-t3/t3.db', 'ro'),
]
MERK = ('immun', 'aufnahme', 'abweich', 'ansage', 'mentor', 'gedaechtnis', 'muster')

for name, pfad, art in ZWEIGE:
    wal = os.path.getsize(pfad + '-wal') if os.path.exists(pfad + '-wal') else None
    print(f'==================== {name}: {pfad} (wal {wal})')
    if art == 'immutable':
        if wal not in (None, 0):
            print('NICHT GEÖFFNET: -wal nicht 0 Bytes')
            continue
        uri = f'file:{pfad}?mode=ro&immutable=1'
    else:
        uri = f'file:{pfad}?mode=ro'
    c = sqlite3.connect(uri, uri=True)
    print('-- 1. Protokollarten')
    for a, n, z0, z1 in c.execute('SELECT art, count(*), min(zeit), max(zeit) FROM protokoll GROUP BY art ORDER BY art'):
        if any(k in a for k in MERK):
            print(f'   {a}: {n} | {z0} … {z1}')
    for t in ('mentor', 'kandidaten'):
        print(f'-- 2. Spalten {t}:', [r[1] for r in c.execute(f'PRAGMA table_info({t})')])
    print('-- 3. letzte Zeile mentor')
    cur = c.execute('SELECT * FROM mentor ORDER BY rowid DESC LIMIT 1')
    namen = [d[0] for d in cur.description]
    zeile = cur.fetchone()
    if zeile:
        for k, v in zip(namen, zeile):
            s = str(v)
            print(f'   {k}: {s[:700]}{" …[" + str(len(s)) + " Zeichen]" if len(s) > 700 else ""}')
    for a in ('immun.aufnahme_gesperrt', 'immun.aufnahme_geprueft', 'aegis.abweichung'):
        r = c.execute('SELECT nr, zeit, inhalt FROM protokoll WHERE art=? ORDER BY nr DESC LIMIT 1', (a,)).fetchone()
        print(f'-- 4. Beispiel {a}:', (r[0], r[1], r[2][:700]) if r else 'keine Zeile')
    c.close()
    print(f'   danach wal: {os.path.getsize(pfad + "-wal") if os.path.exists(pfad + "-wal") else None}')
