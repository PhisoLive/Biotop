#!/usr/bin/env python3
"""Teil E (SOLL-IST), nur lesend: Prüfung der Entwurfsdateien SOLL-IST-TEIL*.md gegen Masterprompt v8.
Je Tabellenzeile, die mit `| MP` beginnt: Kennung eindeutig; genau ein Status aus FK30; jedes Wortlaut-Stück (getrennt
an „…“) steht in v8 in den angegebenen Zeilen (Vergleich ohne `*`). Dazu: jede UNKLAR-Kennung steht in Abschnitt 1 von
Teil 4; jeder Verweis auf eine Kennung trifft einen Eintrag; Bytes je Datei <= 34.000. Zählung je Status mit Rechenweg.
Liest keine Datenbank und keinen Biotop-Code. Schreibt nichts."""
import hashlib
import re
from collections import Counter
from pathlib import Path

ORDNER = Path('/root/sicherung/biotop-lage-01/entwurf/soll-ist')
V8 = Path('/root/sicherung/biotop-lage-01/arbeit/dokumente/'
          '2026-09-23_155353_INFINITE-EVOLUTION-BIOTOP-Masterprompt-v8.md')
STATUS = ['BELEGT', 'TEILWEISE', 'OFFEN', 'NICHT BEGONNEN', 'ABGEWICHEN', 'ENTFALLEN', 'UNKLAR']
KENNUNG = re.compile(r'MP(?:[KAB]|\d+a?)(?:\.\d+)+')

v8 = V8.read_text(encoding='utf-8').splitlines()
print(f'v8: {len(v8)} Zeilen, SHA-256 {hashlib.sha256(V8.read_bytes()).hexdigest()}')

teile = sorted(ORDNER.glob('SOLL-IST-TEIL*.md'))
texte = {}
eintraege = []
fehler = []
for t in teile:
    roh = t.read_bytes()
    texte[t.name] = roh.decode('utf-8')
    print(f'{t.name}: {len(roh)} Bytes, SHA-256 {hashlib.sha256(roh).hexdigest()}, '
          f'{"<= 34.000" if len(roh) <= 34000 else "UEBER 34.000"}')
    if len(roh) > 34000:
        fehler.append(f'{t.name}: {len(roh)} Bytes')
    for nr, zeile in enumerate(texte[t.name].splitlines(), 1):
        if not zeile.startswith('| MP'):
            continue
        sp = [s.strip() for s in re.split(r'(?<!\\)\|', zeile)[1:-1]]
        if len(sp) != 5:
            fehler.append(f'{t.name}:{nr}: {len(sp)} Spalten statt 5')
            continue
        kennung, zeilen, wortlaut, status, _beleg = sp
        eintraege.append((kennung, status, t.name))
        if status not in STATUS:
            fehler.append(f'{kennung}: Status {status!r} nicht in der Liste von FK30')
        z = [int(x) for x in re.findall(r'\d+', zeilen)]
        if not z:
            fehler.append(f'{kennung}: keine Zeilenangabe')
            continue
        von, bis = z[0], z[-1]
        bereich = '\n'.join(v8[von - 1:bis]).replace('*', '')
        if not (wortlaut.startswith('„') and wortlaut.endswith('“')):
            fehler.append(f'{kennung}: Wortlaut nicht in „…“ gefasst')
            continue
        stuecke = []
        for stueck in re.split(r'“ … „', wortlaut[1:-1]):
            stuecke += stueck.split(' … ')
        for s in stuecke:
            if s.replace('*', '') not in bereich:
                fehler.append(f'{kennung}: Wortlaut nicht in v8 Z. {von}–{bis}: {s[:100]!r}')

zaehl = Counter(k for k, _, _ in eintraege)
for k, n in zaehl.items():
    if n > 1:
        fehler.append(f'{k}: {n}-mal vergeben')
bekannt = set(zaehl)
for name, text in texte.items():
    for k in sorted(set(KENNUNG.findall(text))):
        if k not in bekannt:
            fehler.append(f'{name}: Verweis auf {k} ohne Eintrag')

t4 = texte.get('SOLL-IST-TEIL4.md', '')
abschnitt1 = t4.split('## 1.', 1)[-1].split('## 2.', 1)[0] if '## 1.' in t4 else ''
unklar = [k for k, s, _ in eintraege if s == 'UNKLAR']
for k in unklar:
    if not re.search(re.escape(k) + r'(?!\d)(?!\.\d)', abschnitt1):
        fehler.append(f'{k}: Status UNKLAR, fehlt in Teil 4 Abschnitt 1')

gesamt = len(eintraege)
je = Counter(s for _, s, _ in eintraege)
print(f'\nEinträge gesamt: {gesamt}')
for s in STATUS:
    print(f'   {s}: {je.get(s, 0)} von {gesamt}')
print('   Rechenweg: ' + ' + '.join(f'{je.get(s, 0)} ({s})' for s in STATUS) + f' = {sum(je.values())}')
je_teil = Counter(t for _, _, t in eintraege)
print('   je Teil: ' + ', '.join(f'{t}: {n}' for t, n in sorted(je_teil.items())))
for s in STATUS:
    ks = [k for k, st, _ in eintraege if st == s]
    if ks:
        print(f'   {s}: {", ".join(ks)}')
print(f'\nFehler: {len(fehler)}')
for f in fehler:
    print('   ' + f)
