import subprocess

def git(*a) -> str:

dateien = []
for z in git('ls-tree', '-r', '-l', C).splitlines():
    meta, pfad = z.split('\t', 1)
    _, typ, _, groesse = meta.split()
    if typ == 'blob':
        dateien.append((pfad, int(groesse)))

print('== A1. Dateien unterhalb Tiefe 3 (Pfad | Bytes)')
for p, g in dateien:
    if p.count('/') >= 3:

for p, g in sorted(dateien, key=lambda x: -x[1])[:10]:
    print(f'{g} | {p}')

print('\n== B. Kopf je Modul und öffentliche Namen')
for p, _ in dateien:
    if not p.startswith('biotop/') or not p.endswith(('.py', '.mjs', '.js', '.ts')):
        continue
    t = git('show', f'{C}:{p}')
    zeilen = t.splitlines()

    if p.endswith('.py'):

        if m:

    else:

                kopf.append(z.strip())
            elif kopf:
                break
    oeff = [re.sub(r'\s+', ' ', z.strip())[:110] for z in zeilen
            if re.match(r'^(def|class|async def) [A-Za-z]', z)]
    exp = [z.strip()[:110] for z in zeilen if z.startswith('export ')] if not p.endswith('.py') else []
    print(f'\n--- {p} ({len(zeilen)} Zeilen)')
    for k in kopf[:18]:
        print('  | ' + k)
    if len(kopf) > 18:
        print(f'  | […{len(kopf) - 18} Kopfzeilen mehr]')
    if oeff or exp:
        print('  öffentlich: ' + ' ; '.join((oeff + exp)[:40]) + (f' ; […{len(oeff) + len(exp) - 40} mehr]' if len(oeff) + len(exp) > 40 else ''))
