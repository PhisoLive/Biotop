def saeubern(t, K, L):
    t = t.replace(str(L), '{LAEUFE}').replace(K, '<schlüssel>')
    for name, m in geheimnis.MUSTER.items():
        t = re.sub(m, f'[GESCHWÄRZT: {name}]', t)
    return t


def exklusiv(p, text):
    with open(p, 'x', encoding='utf-8') as f:
        f.write(text)
    return p


def schreibe(aus, name, text, K, L):
    return exklusiv(aus / name, saeubern(text, K, L))


def schreibe_md(aus, kopf, abschnitte, K, L):
    bloecke = []
    for titel, zeilen in abschnitte:
        teil, cur = [], [f'## {titel}', '']
        for x in zeilen:
            x = saeubern(x, K, L)
            if len('\n'.join(cur + [x]).encode()) > MAX_MD - 2500:
                teil.append(cur)
                cur = [f'## {titel} (Fortsetzung)', '']
                if x.startswith('|') and not x.startswith('|---'):
                    kz = next((c for c in reversed(teil[-1]) if c.startswith('|---')), None)
                    ki = teil[-1].index(kz) if kz else None
                    if ki:
                        cur += [teil[-1][ki - 1], kz]
            cur.append(x)
        teil.append(cur)
        bloecke += ['\n'.join(b) + '\n' for b in teil]
    k = saeubern('\n'.join(kopf) + '\n', K, L)
    teile, cur = [], k
    for b in bloecke:
        if len((cur + '\n' + b).encode()) > MAX_MD - 400:
            teile.append(cur)
            cur = ''
        cur = (cur + '\n' + b) if cur else b
    teile.append(cur)
    if len(teile) == 1:
        return [exklusiv(aus / 'INVENTAR.md', teile[0])]
    out = []
    for i, t in enumerate(teile, 1):
        vor = f'> INVENTAR Teil {i} von {len(teile)} (INVENTAR-TEIL1.md … INVENTAR-TEIL{len(teile)}.md)\n\n'
        out.append(exklusiv(aus / f'INVENTAR-TEIL{i}.md', vor + t))
    return out
