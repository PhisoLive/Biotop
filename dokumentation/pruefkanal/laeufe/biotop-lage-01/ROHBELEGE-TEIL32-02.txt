#!/usr/bin/env python3
"""BIOTOP-LAGE-01 Teil C (nur lesend): Wortlaute der Torzeilen je Masterprompt-Fassung v1..v8.
Quelle: /root/sicherung/biotop-lage-01/arbeit/dokumente/*INFINITE-EVOLUTION-BIOTOP-Masterprompt-v*.md (Dateien < 10000
Zeichen sind Bruchstücke und werden übersprungen). Je Fassung: jeder Block, der mit einer Zeile '**Tor <n>' beginnt,
bis zur ersten Leerzeile; darin jede Zeile, die mit '- ' beginnt, ist eine Torzeile.
Ausgabe: (1) Anzahl Torzeilen je Tor und Fassung; (2) v8-Wortlaut jeder Torzeile (Tor 1-4) mit Nummer; (3) je frühere
Fassung jede Torzeile, deren Wortlaut vom v8-Wortlaut derselben Nummer abweicht (vollständig);
(4) die Tor-4-Zeilen aus v6 und v7 vollständig. Schreibt zusätzlich tore/wortlaute.json."""
import json
import re
from pathlib import Path

D = Path('/root/sicherung/biotop-lage-01/arbeit/dokumente')
AUS = Path('/root/sicherung/biotop-lage-01/entwurf/tore/wortlaute.json')
fassungen = {}
for p in sorted(D.glob('*INFINITE-EVOLUTION-BIOTOP-Masterprompt-v*.md')):
    t = p.read_text()
    if len(t) < 10000:
        continue
    v = re.search(r'Masterprompt-(v\d)', p.name).group(1)
    tore = {}
    zeilen = t.splitlines()
    for i, z in enumerate(zeilen):
        m = re.match(r'^\*\*Tor (\d)\b', z)
        if not m:
            continue
        n = m.group(1)
        if n in tore:
            continue
        punkte = []
        j = i + 1
        while j < len(zeilen) and zeilen[j].strip():
            if zeilen[j].startswith('- '):
                punkte.append(zeilen[j][2:].strip())
            j += 1
        tore[n] = {'kopf': z.strip(), 'zeile_nr_datei': i + 1, 'punkte': punkte}
    fassungen[v] = {'datei': p.name, 'tore': tore}

print('== 1. Anzahl Torzeilen je Tor und Fassung (Kopfzeile, Zeilennummer in der Datei)')
for v, f in fassungen.items():
    print(v, f['datei'], {n: (len(x['punkte']), x['zeile_nr_datei'], x['kopf'][:60]) for n, x in sorted(f['tore'].items())})

v8 = fassungen['v8']['tore']
print('\n== 2. Wortlaut v8')
for n in sorted(v8):
    print(f'--- Tor {n}: {v8[n]["kopf"]}')
    for k, s in enumerate(v8[n]['punkte'], 1):
        print(f'  T{n}.{k}: {s}')


def norm(s):
    return re.sub(r'\s+', ' ', s).strip()


print('\n== 3. Abweichungen früherer Fassungen vom v8-Wortlaut (gleiche Nummer), vollständig')
for v, f in fassungen.items():
    if v == 'v8':
        continue
    for n in ('1', '2', '3'):
        if n not in f['tore']:
            print(f'{v} Tor {n}: NICHT VORHANDEN')
            continue
        alt = f['tore'][n]['punkte']
        neu = v8.get(n, {}).get('punkte', [])
        gleich = sum(1 for a, b in zip(alt, neu) if norm(a) == norm(b))
        print(f'{v} Tor {n}: {len(alt)} Zeilen, gleich wie v8 (gleiche Nummer): {gleich}')
        for k, a in enumerate(alt, 1):
            b = neu[k - 1] if k <= len(neu) else None
            if b is None or norm(a) != norm(b):
                # steht der Wortlaut unter anderer Nummer in v8?
                anders = [i + 1 for i, x in enumerate(neu) if norm(x) == norm(a)]
                print(f'  {v} T{n}.{k}{" (in v8 als T" + n + "." + str(anders[0]) + ")" if anders else ""}: {a}')

print('\n== 4. Tor 4 in v6 und v7 vollständig')
for v in ('v6', 'v7'):
    t4 = fassungen[v]['tore'].get('4')
    print(f'--- {v}: {t4["kopf"] if t4 else "NICHT VORHANDEN"}')
    for k, s in enumerate((t4 or {}).get('punkte', []), 1):
        print(f'  {v} T4.{k}: {s}')
AUS.write_text(json.dumps(fassungen, ensure_ascii=False, indent=1))
print('\ngeschrieben:', AUS)

  T3.8: Archivierung: Offene Funde nach 5 **Generationen** archiviert (Fassung 5; Zahl archiviert / gesamt), 0 archivierte Funde in Kammerläufen danach. Tritt der Fall in keinem Lauf ein, wird er an einem eigens erzeugten offenen Fund geprüft und als Mechanikprüfung gekennzeichnet — eine Prüfung ohne Gegenstand besteht nicht, sie meldet.
  T3.9: Oberfläche: Für jedes Bedienelement Dateipfad und Zeile der Stelle, die es liest oder schreibt (vollständige Liste; ein Element ohne Stelle = Tor nicht bestanden); Zugriff nur mit Anmeldung (Aufruf ohne Token → 401, belegt); Kostenstand und Kurve sichtbar.
  T3.10: Kandidatenaufschlüsselung jedes Laufs geht auf die Gesamtzahl auf (Abschnitt 9).
--- Tor 4: **Tor 4 — je Testwerk (Fassung 8):**
  T4.1: Aufnahme: Werk gebaut, Bau- und Testlauf, versiegelte Schwächen mit Gegenzeugen hinterlegt (bei T4 stattdessen eine unabhängig erhobene Altlastenliste mit Bezugsgröße).
  T4.2: Generation minus eins: Rauschmessung mit Sperren je Bereich; **Maßeichung am sauberen Vergleichsbereich**, stillgelegte Regeln mit Befundzahl ausgewiesen; Eichgegenstand versiegelt; 4 von 4 Proben bestanden und 4 von 4 Gegenproben an der vorgesehenen Zusicherung gescheitert; Eichkosten getrennt.
  T4.3: Generation 0: Mutantenfang ≥ 80 von 100 nicht-äquivalenten, **je Art ausgewiesen**, mit der Verteilung dessen, was vorab verworfen wurde (Typsieb); kein Durchlauf, in dem überwiegend neutral geurteilt wurde, geht in die Zahl ein (Grundsatz 13).
  T4.4: Schwächen: ≥ 80 % der versiegelten Schwächen behoben nach „behoben heißt verschwunden"; je Schwäche der behebende Kandidat und **der Grund, aus dem er angenommen wurde** — eine Behebung, die nur über Fehlalarmpunkte oder eine Einzelmessung durchkam, zählt als nicht belastbar.
  T4.5: Kein Rückschritt: 0 von n angenommenen Generationen mit gesunkenem hartem Maß, **je Maßfeld einschließlich Laufzeit**, gepaart gemessen (Grundsatz 14).
  T4.6: **Halluzinationsprobe (streng):** Der saubere Vergleichsbereich bleibt unberührt. Jede angenommene Änderung dort lässt die Probe reißen, es sei denn, sie behebt nachweislich einen Fehler, belegt durch einen Test, der vorher fällt und nachher besteht. (Der Wortlaut von Fassung 7, „keine Änderung ohne Maßgewinn", ließ Fehlalarmbereinigung und einen fehlerhaften Zwischenspeicher durch, weil beide einen gemessenen Gewinn hatten.)
  T4.7: **Verhaltensvergleich:** jede angenommene Generation hat ihn durchlaufen; je Urteil „x von y Funktionen beobachtet"; Enthaltungen zählen nicht als bestanden; ein Sondenausfall ist ein Messgerätausfall, keine Ablehnung des Kandidaten und kein Eintrag ins Gedächtnis.
  T4.8: **Doppelungsanteil unter 20 %,** gemessen nach der Methode aus Vorbedingung 1: alle erzeugten Angriffstests, geurteilt nach ihrer geschriebenen Behauptung. Ansagefilter und Aufnahmesperre werden als Diagnosen mitgeführt, sind aber nicht die Torzeile.
  T4.9: Immunisierung (3.10 F3): für die Sprache des Werks auf dem tatsächlichen Pfad gezählt; 0 Fälle heißt nicht belegt.
  T4.10: Neutralzähler: jedes Organ mit Nenner; eine Torzeile, die an einem Organ hängt, gilt nur, wenn es in mehr als der Hälfte seiner Fälle geurteilt hat — **je Durchlauf, nicht über die Summe**.
  T4.11: Verbundausfall: als Ausfall gezählt, nie als Beitrag; belegt an mindestens einem echten oder gestellten Ausfall.
  T4.12: **Verdrahtung:** volle Regression einschließlich Verdrahtungstest grün am Kernstand des Laufs (Grundsatz 12).
  T4.13: Sättigung oder Budgetabbruch dokumentiert — der Lauf endet nie still.
  T4.14: Ein Reviewdurchgang; Abschlussordner trägt die Torzeilen. Zwischenbericht spätestens bei 120 von 150 Aufrufen einer Sitzung, mechanisch ausgelöst.
--- Tor 5: **Tor 5** wird gesetzt, wenn das Werk feststeht — Kernmaß, harte Grenzen und Bezugsgrößen hängen am Werk und werden nicht vorher erfunden.
