#!/usr/bin/env python3
"""BIOTOP-LAGE-01 (nur lesend): Hauptdatei je Biotop-Abschlussordner seit dem 21.09. (für Meldung 2 und START-03).
Ordner: laeufe/biotop-* (ohne biotop-lage-01) mit einer Datei, deren Änderungszeit ≥ 21.09.2026 00:00 CEST ist.
Hauptdatei (Auslegung Hermes), in dieser Reihenfolge, Tiefe 1 vor Tiefe 2: BERICHT*.md, ABSCHLUSS*.md, START-NN.md (höchste),
ZWISCHENBERICHT.md, README.md; sonst die jüngste .md. Gibt je Ordner die Adresse, Größe und die Zahl der Dateien aus."""
import datetime as dt
import re
from pathlib import Path

K = Path('/etc/pruefkanal/schluessel').read_text().strip()
L = Path(f'/var/pruefkanal/oeffentlich/_akte/{K}/laeufe')
BASIS = f'https://milieulux-reinigung-muenchen.de/_akte/{K}/laeufe'
AB = dt.datetime(2026, 9, 21).timestamp()
MUSTER = [r'^BERICHT.*\.md$', r'^ABSCHLUSS.*\.md$', r'^START-\d+\.md$', r'^ZWISCHENBERICHT\.md$', r'^README\.md$']
n = 0
for o in sorted(p for p in L.iterdir() if p.is_dir() and p.name.startswith('biotop-') and p.name != 'biotop-lage-01'):
    dateien = [p for p in o.rglob('*') if p.is_file()]
    if not dateien or max(p.stat().st_mtime for p in dateien) < AB:
        continue
    n += 1
    wahl = None
    for tiefe in (1, 2):
        kand = [p for p in dateien if len(p.relative_to(o).parts) == tiefe]
        for m in MUSTER:
            t = sorted((p for p in kand if re.match(m, p.name)), key=lambda p: (p.name, p.stat().st_mtime))
            if t:
                wahl = t[-1]; break
        if wahl:
            break
    if not wahl:
        md = [p for p in dateien if p.suffix == '.md']
        wahl = max(md or dateien, key=lambda p: p.stat().st_mtime)
    rel = wahl.relative_to(L).as_posix()
    print(f'{o.name} · {len(dateien)} Dateien · jüngste {dt.datetime.fromtimestamp(max(p.stat().st_mtime for p in dateien)):%d.%m. %H:%M} · '
          f'{BASIS}/{rel} · {wahl.stat().st_size} B')
print(f'Ordner seit 21.09.: {n}')

"""BIOTOP-LAGE-01 (nur lesend): Zahl der offenen Punkte in FK39 (LAGE-TEIL5-01.md): Tabellenzeilen mit einer Kennung
in der ersten Spalte, je Abschnitt; dazu „blockiert ja“ je Abschnitt nach der Abschnittsüberschrift."""
import re
from pathlib import Path

t = Path('/root/sicherung/biotop-lage-01/ausgabe/LAGE-TEIL5-01.md').read_text(encoding='utf-8')
abschnitt, zaehl = None, {}
for z in t.splitlines():
    if z.startswith('## '):
        abschnitt = z[3:60]
        zaehl[abschnitt] = 0
    elif abschnitt and z.startswith('| ') and not re.match(r'\|\s*(-+|Kennung|Nr)\s*\|', z) and not z.startswith('|---'):
        zaehl[abschnitt] += 1
for a, n in zaehl.items():
    print(f'{n:3d} · {a}')
print(f'Summe Tabellenzeilen: {sum(zaehl.values())}')
