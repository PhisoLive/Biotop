def sha(p: str) -> str:
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    return h.hexdigest()


def zeile(p: str, merkmale: str) -> str:
    st = os.stat(p)
    mz = time.strftime('%Y-%m-%d %H:%M:%S', time.localtime(st.st_mtime))
    return f'{anzeigen(p)} | {st.st_size} | {sha(p)} | {mz} | {merkmale}'

for w in WURZELN_INHALT + WURZELN_NAME:

        continue
    nur_name = w in WURZELN_NAME
    for d, dirs, files in os.walk(w):
        dirs[:] = sorted(x for x in dirs if x not in AUSLASSEN)
        for fn in sorted(files):

            if not os.path.isfile(p) or os.path.islink(p):

            if os.path.splitext(fn)[1].lower() not in ENDUNGEN:

            try:

for p, m in inhalts:
    print(zeile(p, m))
