#!/bin/bash
# Nachtrag der Behebungsrunde 3 vor T4: Commit (nur tests/test_t3_python_annahmen.py) auf d54bf2f, volle Regression am
# Nachtrag samt Verdrahtungstest, dann dieselben Fälle wie proben/vorher_6243377.py am Code von HEAD (nachher_head.py).
# Nacheinander, keine parallelen Kammerläufe. Kein Modellaufruf.
set -u
A=/root/biotop-vor-t4-03
PY=/root/biotop-t3/venv/bin/python
cd /root/biotop
echo "=== start $(date -Is) HEAD=$(git rev-parse --short HEAD)"
git status --short
git add tests/test_t3_python_annahmen.py
git commit -q -F $A/COMMIT-NACHRICHT-2.txt
echo "commit-rc=$?"
git log --oneline -4
echo "=== biotop/ gegenüber d54bf2f (leer = Code unverändert):"; git diff --stat d54bf2f HEAD -- biotop; echo "(Ende)"
git status --short -- biotop tests; echo "(Ende Status)"
git show --stat HEAD | tail -12 > $A/COMMIT-2-STAT.txt
git show HEAD > $A/KERN-COMMIT-2.patch.txt
export BIOTOP_KAMMER_IMAGE=biotop-kammer:ts1 BIOTOP_AUFTRAG_LAUF_AB=26 BIOTOP_AUFTRAG_BUDGET_EUR=10
export T3_VERDRAHTUNG_BELEG=$A/VERDRAHTUNG-BELEG-REGRESSION-2.json
echo "regression-2 start $(date -Is) HEAD=$(git rev-parse --short HEAD) fremde_container=$(docker ps -q | wc -l)"
$PY -m pytest tests/ -q -m "not integration" -p no:cacheprovider -rfEs --durations=8 \
  --junitxml=$A/REGRESSION-2.junit.xml > $A/REGRESSION-2.txt 2>&1
echo "regression-2 pytest-rc=$? $(tail -1 $A/REGRESSION-2.txt)"
git status --short -- biotop tests; echo "(Ende Status nach Regression)"
echo "nachher-head start $(date -Is)"
NEU=$(mktemp -d /tmp/biotop-head-XXXX)
git archive HEAD | tar -x -C $NEU
BIOTOP_KAMMER_IMAGE=biotop-kammer:1 $PY $A/proben/nachher_head.py $NEU "$(git rev-parse --short HEAD) (git archive)" > $A/proben/NACHHER-HEAD.txt 2>&1
echo "nachher-head rc=$? ($NEU)"
git status --short -- biotop tests; echo "(Ende Status)"
echo "ende $(date -Is)"
