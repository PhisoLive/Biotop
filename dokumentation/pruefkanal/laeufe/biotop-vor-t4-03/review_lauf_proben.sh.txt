#!/bin/bash
# Review vor T4-03: alle Kammerproben nacheinander (nur wenn kein job_commit_regression läuft). Ausgaben unter
# /tmp/review-vor-t4-03/. Kein Modell, kein Netz.
set -u
R=/tmp/review-vor-t4-03
PY=/root/biotop-t3/venv/bin/python
export TMPDIR=$R/tmp PYTHONDONTWRITEBYTECODE=1
mkdir -p $TMPDIR
cd $R
if pgrep -f job_commit_regression >/dev/null; then echo "job läuft — Abbruch"; exit 3; fi
echo "start $(date -Is) HEAD=$(git -C /root/biotop rev-parse --short HEAD) status=[$(git -C /root/biotop status --porcelain | head -3)] container=$(docker ps -q | wc -l)"
for s in sondenprobe.py sondenprobe2.py sondenprobe3.py deckprobe.py; do
  echo "=== $s $(date -Is)"
  $PY -B $s > $R/${s%.py}.out.txt 2>&1
  echo "rc=$?"
  tail -c 400 $R/${s%.py}.out.txt | tail -3
done
echo "ende $(date -Is) status=[$(git -C /root/biotop status --porcelain | head -3)]"
