function uebertragen(stand) {
  const aus = '/tmp/s1/' + stand;
  for (const rel of ['tarife/ausdruck.ts', 'tarife/modell.ts', 'zahlen/pruefen.ts']) {
    const q = `/werk/${stand}/src/${rel}`;
    if (!fs.existsSync(q)) continue;
    const js = ts.transpileModule(fs.readFileSync(q, 'utf8'),
      {compilerOptions: {module: ts.ModuleKind.ESNext, target: ts.ScriptTarget.ES2022}}).outputText;
    const z = path.join(aus, rel.replace(/\.ts$/, '.js'));
    fs.mkdirSync(path.dirname(z), {recursive: true});
    fs.writeFileSync(z, js);
  }
  fs.writeFileSync(path.join(aus, 'package.json'), '{"type":"module"}');
  return 'file://' + path.join(aus, 'tarife/ausdruck.js');
}

// ---- Vertrag (tests/vertraege/VERTRAEGE.md Z. 22) als unabhängiger Maßstab für Ausdrücke AUS DER GRAMMATIK:
// Rangfolge * / vor + -, links assoziativ; Division durch null abgewiesen; Ergebnis endlich und ≥ 0.
function vertrag(tokens, k) {
  let i = 0;
  const faktor = () => {
    const t = tokens[i++];
    if (t === '(') { const v = summe(); i++; return v; }
    if (t === 'gewicht') return k.gewicht;
    if (t === 'grund') return k.grund;
    return Number(t);
  };
  const produkt = () => {
    let v = faktor();
    while (tokens[i] === '*' || tokens[i] === '/') {
      const op = tokens[i++]; const r = faktor();
      if (op === '/') { if (r === 0) throw new Error('Division durch null'); v = v / r; } else v = v * r;
    }
    return v;
  };
  const summe = () => {
    let v = produkt();
    while (tokens[i] === '+' || tokens[i] === '-') { const op = tokens[i++]; const r = produkt(); v = op === '+' ? v + r : v - r; }
    return v;
  };
  try {
    const v = summe();
    if (!Number.isFinite(v) || v < 0) return {ok: false, e: 'abgewiesen'};
    return {ok: true, v};
  } catch { return {ok: false, e: 'abgewiesen'}; }
}

function mulberry32(a) { return () => { a |= 0; a = a + 0x6D2B79F5 | 0; let t = Math.imul(a ^ a >>> 15, 1 | a); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; }; }

const r = n => Math.floor(rnd() * n);
function zahl() {
  const g = r(4) === 0 ? '0' : String(1 + r(9999));
  if (r(3) !== 0) return g;
  const k = 1 + r(3);
  return g + '.' + String(r(10 ** k)).padStart(k, '0');
}
function erzeugen(tiefe) {
  const w = r(10);
  if (tiefe <= 0 || w < 4) { const a = r(4); return [a === 0 ? 'gewicht' : a === 1 ? 'grund' : zahl()]; }
  if (w < 6) return ['(', ...erzeugen(tiefe - 1), ')'];
  return [...erzeugen(tiefe - 1), '+-*/'[r(4)], ...erzeugen(tiefe - 1)];
}
function schreiben(tokens) {
  let s = '';
  for (const t of tokens) s += (s && r(2) ? ' '.repeat(1 + r(2)) : '') + t;
  return (r(8) === 0 ? ' ' : '') + s + (r(8) === 0 ? ' ' : '');
}
const KONTEXTE = [{gewicht: 0, grund: 0}, {gewicht: 1, grund: 2}, {gewicht: 30, grund: 250}, {gewicht: 0.5, grund: 1000},

function rufen(f, text, k) {
  try { const v = f(text, k); return {ok: true, v}; }
  catch (e) { return {ok: false, e: e?.constructor?.name ?? 'unbekannt', m: String(e?.message ?? e).slice(0, 90)}; }
}
const gleich = (a, b) => (a.ok && b.ok) ? Object.is(a.v, b.v) : (!a.ok && !b.ok) ? a.e === b.e : false;
const vertragsgleich = (x, v) => x.ok === v.ok && (!x.ok || Object.is(x.v, v.v));
const art = (a, b) => (a.ok && b.ok) ? 'wert' : (!a.ok && !b.ok) ? 'fehlerklasse' : (a.ok ? 'alt_zahl_neu_fehler' : 'alt_fehler_neu_zahl');

function kFall(text, k) {

  if (!vertragsgleich(n, v)) { K.neu_vertragswidrig++; if (K.beispiele_neu_vertragswidrig.length < 8) K.beispiele_neu_vertragswidrig.push({text, k, neu: n, vertrag: v}); }
