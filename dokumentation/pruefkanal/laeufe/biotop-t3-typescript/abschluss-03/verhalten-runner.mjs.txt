// Läuft in der Kammer, nachdem verhalten.mjs die geänderten Module eingewickelt hat.
// Installiert den Aufzeichner, führt die Referenzlast aus (beobachten() und last_<bereich>())
// und schreibt die Beobachtungen nach /tmp/ts-vv-ergebnis.json.
//
// Je Funktion werden die ersten KAPPE äußeren Aufrufe beobachtet. Nach dem echten Aufruf
// (dessen Ergebnis der Aufrufer bekommt) folgen drei Zusatzaufrufe; vor jedem wird der Zustand
// der Eingaben festgehalten und danach vollständig wiederhergestellt, so dass der Aufrufer
// genau den Zustand nach dem echten Aufruf weiterverwendet:
//   @alias               dieselben Referenzen sofort noch einmal
//   @mutation            Zahlenblätter an Ort und Stelle geändert (gleiche Referenzen, gleiche Längen)
//   @nach_ruecksetzung   wieder auf den Zustand nach dem echten Aufruf gesetzt, erneut aufgerufen
import fs from 'node:fs';
import {createHash} from 'node:crypto';

const req = JSON.parse(fs.readFileSync('/tmp/ts-vv-anfrage.json', 'utf8'));
const KAPPE = req.kappe ?? 8;
const beob = {}, zaehler = {};
let tiefe = 0;

function kan(x, anc) {
  if (x === null) return null;
  switch (typeof x) {
    case 'number':
      if (Number.isNaN(x)) return '__NaN';
      if (!Number.isFinite(x)) return x > 0 ? '__Inf' : '__-Inf';
      return Object.is(x, -0) ? '__-0' : x;
    case 'string': case 'boolean': return x;
    case 'undefined': return '__undefined';
    case 'bigint': return '__big:' + x;
    case 'function': return '__fn';
    case 'symbol': return '__sym';
  }
  if (anc.has(x)) return '__zyklus';
  anc.add(x);
  let r;
  if (Array.isArray(x)) r = x.map(v => kan(v, anc));
  else if (x instanceof Map) r = {__map: [...x.entries()].map(([k, v]) => [kan(k, anc), kan(v, anc)])};
  else if (x instanceof Set) r = {__set: [...x.values()].map(v => kan(v, anc))};
  else if (x instanceof Error) r = {__fehler: x.constructor?.name ?? 'Error', meldung: String(x.message)};
  else if (x instanceof Date) r = {__datum: x.getTime()};
  else { r = {}; for (const k of Object.keys(x).sort()) r[k] = kan(x[k], anc); }
  anc.delete(x);
  return r;
}

function wert(f) {
  try {
    const v = f();
    if (v && typeof v.then === 'function') return {text: '__promise', roh: v};
    return {text: JSON.stringify(kan(v, new Set())), roh: v};
  } catch (e) {
    return {text: 'Fehler:' + (e?.constructor?.name ?? 'unbekannt'), fehler: e};
  }
}

function knapp(t) {
  if (t.length <= 200) return t;
  return t.slice(0, 120) + '…#' + createHash('sha256').update(t).digest('hex').slice(0, 16);
}

function schlicht(x) {
  return x !== null && typeof x === 'object'
    && (Array.isArray(x) || Object.getPrototypeOf(x) === Object.prototype || Object.getPrototypeOf(x) === null);
}

// Zustand aller erreichbaren Arrays und schlichten Objekte (Tiefe ≤ 4) festhalten.
function schnappschuss(args) {
  const log = [], seen = new Set();
  function geh(x, t) {
    if (!schlicht(x) || seen.has(x) || t > 4) return;
    seen.add(x);
    if (Array.isArray(x)) { log.push([x, x.slice(), true]); for (const v of x) geh(v, t + 1); }
    else { log.push([x, Object.assign({}, x), false]); for (const k of Object.keys(x)) geh(x[k], t + 1); }
  }
  for (const a of args) geh(a, 0);
  return log;
}

function zuruecksetzen(log) {
  for (const [o, kopie, istArray] of log) {
    try {
      if (istArray) { o.length = 0; for (const v of kopie) o.push(v); }
      else {
        for (const k of Object.keys(o)) if (!Object.prototype.hasOwnProperty.call(kopie, k)) delete o[k];
        Object.assign(o, kopie);
      }
    } catch { /* eingefroren oder schreibgeschützt: bleibt, wie es ist */ }
  }
}

// Zahlenblätter an Ort und Stelle ändern — gleiche Referenzen, gleiche Längen. Rückgabe: Anzahl.
function mutieren(args) {
  let n = 0;
  const seen = new Set();
  function geh(x, t) {
    if (!schlicht(x) || seen.has(x) || t > 4) return;
    seen.add(x);
    for (const k of Object.keys(x)) {
      const v = x[k];
      if (typeof v === 'number' && Number.isFinite(v)) {
        try { x[k] = v * 10 + 7; n++; } catch { /* eingefroren */ }
      } else if (v && typeof v === 'object') geh(v, t + 1);
    }
  }
  for (const a of args) geh(a, 0);
  return n;
}

globalThis.__vvAufzeichnen = (name, args, call) => {
  if (tiefe > 0) return call(...args);
  const i = zaehler[name] = (zaehler[name] ?? -1) + 1;
  if (i >= KAPPE) return call(...args);
  tiefe++;
  try {
    const erst = wert(() => call(...args));
    beob[`${name}#${i}`] = knapp(erst.text);
    const stand = schnappschuss(args);
    beob[`${name}#${i}@alias`] = knapp(wert(() => call(...args)).text);
    zuruecksetzen(stand);
    if (mutieren(args) > 0) {
      beob[`${name}#${i}@mutation`] = knapp(wert(() => call(...args)).text);
      zuruecksetzen(stand);
      beob[`${name}#${i}@nach_ruecksetzung`] = knapp(wert(() => call(...args)).text);
      zuruecksetzen(stand);
    }
    if (erst.fehler) throw erst.fehler;
    return erst.roh;
  } finally {
    tiefe--;
  }
};

let out;
try {
  const ref = await import('file:///werk/_biotop_referenz.ts');
  const korpus = [];
  if (typeof ref.beobachten === 'function') korpus.push('beobachten');
  for (const b of req.bereiche) if (typeof ref['last_' + b] === 'function') korpus.push('last_' + b);
  for (const k of korpus) {
    try { await ref[k](); beob['korpus:' + k] = 'ok'; }
    catch (e) { beob['korpus:' + k] = 'Fehler:' + (e?.constructor?.name ?? 'unbekannt'); }
  }
  out = {gueltig: true, beobachtungen: beob, korpus, aufgerufen: zaehler, kappe: KAPPE};
} catch (e) {
  // Die Referenzlast lädt die eingewickelten Module; scheitert der Import, ist das ein Befund
  // über den gebauten Stand (z. B. Laufzeitfehler beim Modulladen), kein totes Messgerät.
  out = {gueltig: false, art: 'inhalt', grund: String(e?.stack || e).slice(0, 2000)};
}
fs.writeFileSync('/tmp/ts-vv-ergebnis.json', JSON.stringify(out));
