def _einzelprobe(con, lauf_id, werk_id, pfad, paket, bereiche, bstrassen, basis, ausgang, ref, tausch,

    """Eine vollständig ausgeführte Positiv- oder Gegenprobe an isoliertem Bestand."""

            bereich = b

        elif name == "nebenwirkung":
            bereich = a
            diff, beschreibung, belege, angriff = _nebenwirkung_vorbereiten(werk, paket, a, b, gegenprobe)
        elif name == "regression":
            bereich = a
            diff, beschreibung, belege, angriff = _regression_vorbereiten(werk, paket, a, b, gegenprobe)
        elif name == "vertrag":
            bereich = a
            diff, beschreibung, belege, angriff = _vertrag_vorbereiten(
                werk, paket, a, b, gegenstand["vertrag_pfad"], gegenprobe, ref)
        else:
            raise ValueError(f"Noch keine Probe gebaut: {name}")

function pack(x, t = 0) {
  if (t > 6 || x === undefined) return {__u: 1};
  if (x === null || typeof x === 'string' || typeof x === 'boolean') return x;
  if (typeof x === 'number') return Number.isFinite(x) ? x : {__num: String(x)};
  if (typeof x === 'bigint') return {__big: String(x)};
  if (typeof x === 'function' || typeof x === 'symbol') throw Error('nicht packbar');
  if (Array.isArray(x)) return x.slice(0, 20).map(v => pack(v, t + 1));
  if (x instanceof Map) return {__map: [...x.entries()].slice(0, 20).map(([k, v]) => [pack(k, t + 1), pack(v, t + 1)])};

def _diff(alt: str, neu: str, rel: str = "probewerk/werk/kern.py", tests: dict | None = None) -> str:
    import difflib
    d = "".join(difflib.unified_diff(alt.splitlines(True), neu.splitlines(True), f"a/{rel}", f"b/{rel}"))
    for p, t in (tests or {}).items():
        d += "".join(difflib.unified_diff([], t.splitlines(True), "/dev/null", f"b/{p}"))
    return d
