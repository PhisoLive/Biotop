import {kan} from './kanon_head.mjs';

function erfasse(x) {
  const fehlend = [];
  let t;
  try { t = JSON.stringify(kan(x, new Set(), fehlend)); } catch (e) { t = '__unkanonisch:' + e?.constructor?.name; }
  return [t, fehlend];
}
function paar(name, a, b) {
  const [ta, fa] = erfasse(a), [tb, fb] = erfasse(b);
  const blind = ta === tb && !fa.length && !fb.length;
  console.log(`B ${name}: gleich=${ta === tb} fehlend=${JSON.stringify([...fa, ...fb].slice(0, 2))} -> ${blind ? 'BLIND' : 'gesehen/enthalten'} | ${String(ta).slice(0, 100)}`);
}
function einzeln(name, x) {
  const [t, f] = erfasse(x);
  console.log(`F ${name}: fehlend=${JSON.stringify(f.slice(0, 2))} -> ${f.length ? 'ENTHALTUNG' : 'ok'} | ${String(t).slice(0, 90)}`);
}
