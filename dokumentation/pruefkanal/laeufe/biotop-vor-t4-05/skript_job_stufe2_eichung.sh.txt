rm -rf $K && mkdir -p $K

echo "kopie $(find $K -type f | wc -l) Dateien"
cd /root/biotop-vor-t4-05/proben
BIOTOP_AUFTRAG_LAUF_AB=26 BIOTOP_AUFTRAG_BUDGET_EUR=10 /root/biotop-t3/venv/bin/python -B stufe2_eichung.py $K > STUFE2-EICHUNG.txt 2>&1
