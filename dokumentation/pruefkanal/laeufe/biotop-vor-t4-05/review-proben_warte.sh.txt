#!/bin/bash
# Warten auf das Ende von job_kette (höchstens ~5 Minuten je Aufruf), dann Stand der Kette zeigen
for i in $(seq 1 20); do pgrep -f job_kette >/dev/null || break; sleep 15; done
date -Is
if pgrep -f job_kette >/dev/null; then echo "job_kette läuft noch"; else echo "job_kette beendet"; fi
tail -6 /root/biotop-vor-t4-05/JOB-KETTE.txt
tail -2 /root/biotop-vor-t4-05/REGRESSION.txt
