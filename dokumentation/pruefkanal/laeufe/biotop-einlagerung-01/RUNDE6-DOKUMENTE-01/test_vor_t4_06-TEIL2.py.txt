def _w_diff(alt: str | None, neu: str, rel: str) -> str:
    import difflib
    return "".join(difflib.unified_diff((alt or "").splitlines(True), neu.splitlines(True),
                                        "/dev/null" if alt is None else f"a/{rel}", f"b/{rel}"))
