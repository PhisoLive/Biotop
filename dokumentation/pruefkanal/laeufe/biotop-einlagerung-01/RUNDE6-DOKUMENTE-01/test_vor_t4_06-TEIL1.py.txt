def _norm_tests(faelle: dict) -> dict:
    return {f"tests/test_werk_normieren_{n}.py": R3.TEST_NORMIEREN.format(name=n, eingabe=e, erwartet=x)
            for n, (e, x) in faelle.items()}


def _vh():
    """Die Sonde des Arbeitsbaums als Modul (lokal, ohne Kammer)."""
    spec = importlib.util.spec_from_file_location("vh_r6", Path(__file__).parent.parent / "biotop/kern/py/verhalten.py")
    vh = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(vh)
    return vh


def _werk(basis: Path, quelle: str, tests: dict | None = None, name: str = "w") -> Path:
    w = basis / name
    (w / "paket" / "m").mkdir(parents=True)
    (w / "tests").mkdir()
    (w / "paket" / "__init__.py").write_text("")
    (w / "paket" / "m" / "__init__.py").write_text("")
    (w / "paket" / "m" / "a.py").write_text(quelle)
    for rel, text in (tests or {}).items():
        (w / rel).write_text(text)
    return w


def _ref(last: str) -> str:
    return "import sys\nsys.path.insert(0, '/werk')\n\n\ndef last_m():\n" + "".join("    " + z + "\n" for z in last.splitlines())


def _organ(tmp_path, monkeypatch, quelle, neu, last, tests=None, deckende=None):
    """verhaltensvergleich.pruefen in der echten Kammer; `deckende`: Kennungen deckender Tests (am Kandidaten mit
    Aufzeichner, deckende_aufrufe) — sonst keine."""
    monkeypatch.setattr(kammer, "IMAGE", "biotop-kammer:1")
    w = _werk(tmp_path, quelle, tests)
    code = {"paket/m/a.py": neu}
    d = {"tests": [], "aufrufe": []} if deckende is None else \
        (lambda abw: vv.deckende_aufrufe(w, code, tests or {}, set(deckende)))
    return vv.pruefen(w, code, _ref(last), ["m"], d)


def _organ_sicher(*a, **k):
    """Wie _organ; ein Messgerätausfall ist ein Befund der Probe (am alten Stand: Zeitlimit der Kammer)."""
    try:
        return _organ(*a, **k), None
    except vv.SondeAusgefallen as e:
        return {}, f"SondeAusgefallen: {str(e)[:300]}"


def _zeitlimit_kappen(monkeypatch, s: int):
    """Kammer-Zeitlimit höchstens s Sekunden — am alten Stand hängt die Sonde sonst 240 s je Kammerlauf."""
    orig = kammer.ausfuehren

    def gekappt(werk, befehl, zeitlimit_s=None, dateien=None):
        return orig(werk, befehl, min(zeitlimit_s or s, s), dateien)
    monkeypatch.setattr(kammer, "ausfuehren", gekappt)

@pytest.mark.parametrize("fall", ["x5", "x5b", "x3", "x3k"])
def test_t1_typpruefung_tautologie_idempotenz_halten_tab_nicht_fest(tmp_path, monkeypatch, fall):
    """Review 05 Nr. 1: der Kandidat K2 lässt Text mit Tab unverändert (Rückschritt). x5 (`isinstance(…, str)`) und x5b
    (Tautologie) „hielten“ an 0c2f058 `tab` fest — das Verfälschungsobjekt war kein str und gleich nichts; überlebt.
    Typtreu verfälscht (str → anderer str) und mit Weiterrechnen am Original bestehen beide Zeilen: der Rückschritt bleibt
    ungedeckt, abgelehnt. x3 (Idempotenz) bleibt abgelehnt, x3k (ohne Zusatzzeile) ist die Kontrolle."""
    tests = _norm_tests(X_BASIS)
    tests["tests/test_werk_normieren_x.py"] = ("from probewerk.werk.kern import normieren\n\n\ndef test_normieren_x():\n"
                                               + ZEILEN[fall] + '    assert normieren("a b") == "a_b"\n')
    u, v = R3._pruefen(tmp_path, monkeypatch, R3._diff(QUELLE, QUELLE.replace(ALT_NORMIEREN, K2)), nische="sicher",
                       werk_extra=tests)
    assert u["status"] == "abgelehnt" and u["grund"].startswith("Verhaltensvergleich:"), u["grund"]
    fest = (v.get("deckung_zaehlung") or {}).get("festhaltend") or {}
    assert not any(t for k, t in fest.items() if k.endswith("|tab")), fest
    assert any("tab" in s.split(" | ")[1].split("+") for s in v.get("ungedeckt_aufschluesselung") or {}), v

def test_t1_typtreue_verfaelschung_je_typ():
    """g(v): derselbe Typ, anderer Inhalt, deterministisch; None, leere Behälter und Fremdes: nicht verfälschbar."""
    vh = _vh()
    g = getattr(vh, "_typtreu", None)
    assert g is not None, "kern/py/verhalten.py::_typtreu fehlt (typtreue Verfälschung)"
    faelle = [True, False, 7, 1.5, float("nan"), float("inf"), 1e300, 2j, "abc", "a", b"ab", bytearray(b"ab"),
              [1, "a"], (1, 2), _Punkt(1, 2), {"a": 1, "b": "x"}, {1, 2}, frozenset({3}), decimal.Decimal("1.5"),
              fractions.Fraction(1, 3), datetime.date(2026, 1, 1), datetime.datetime(2026, 1, 1, 12),
              pathlib.PurePosixPath("a/b.txt"), _Posten("x", 2), [None, 3]]
    for v in faelle:
        ok, w = g(v)
        gleich = (w == v) if not (isinstance(v, float) and v != v) else (w != w)
        assert ok and type(w) is type(v) and not gleich, (v, ok, w)
        assert g(v) == (ok, w) or isinstance(v, (float,)) and v != v, ("nicht deterministisch", v)
    assert g("abc") == (True, "abb") and g("")[1] == "x" and g(7)[1] == 8 and g(True)[1] is False

class _Feld:
    """Daten-Deskriptor mit Zustand außerhalb der Instanz (lokal_b: deskriptor_extern)."""
    def __init__(self):
        self.werte = weakref.WeakKeyDictionary()

    def __get__(self, obj, typ=None):
        return self if obj is None else self.werte.get(obj, 0)

    def __set__(self, obj, v):
        self.werte[obj] = v
