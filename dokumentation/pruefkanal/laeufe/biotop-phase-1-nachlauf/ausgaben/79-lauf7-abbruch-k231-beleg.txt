from lagerhaus.lager.bestand import Lager, NichtGenugBestand
from lagerhaus.lager.journal import Journal, Buchung

def test_reservierung_einloesen_ohne_freigabe_fuehrt_zu_doppelter_reservierung():
    lager = Lager()
    lager.zugang('A', 10)
    lager.reservieren('A', 5)
    assert lager.reserviert('A') == 5
    assert lager.verfuegbar('A') == 5
    lager.reservierung_einloesen('A', 5)
    assert lager.bestand('A') == 5
    assert lager.reserviert('A') == 5
    assert lager.verfuegbar('A') == 0

1. Pareto-Maß 'fehlgeschlagene Tests' wird JE TEST geprüft: ein Test, der an der Vorgeneration besteht und am Kandidaten fällt, tötet den Kandidaten — unabhängig von der Summe.
